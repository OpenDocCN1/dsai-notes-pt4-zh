import time
import logging

class APIMonitor:
    def __init__(self):
        self.request_count = 0
        self.total_cost = 0
        self.error_count = 0
    
    def log_request(self, input_tokens, output_tokens, model):
        self.request_count += 1
        
        # 计算成本
        if model == "claude-haiku-4-5":
            cost = (input_tokens / 1_000_000 * 1) + (output_tokens / 1_000_000 * 5)
        elif model == "claude-sonnet-4-6":
            cost = (input_tokens / 1_000_000 * 3) + (output_tokens / 1_000_000 * 15)
        elif model == "claude-opus-4-6":
            cost = (input_tokens / 1_000_000 * 5) + (output_tokens / 1_000_000 * 25)
        
        self.total_cost += cost
        
        # 告警检查
        if self.total_cost > 100:  # 每日预算 $100
            logging.warning(f"Daily budget exceeded: ${self.total_cost:.2f}")
    
    def log_error(self, error):
        self.error_count += 1
        logging.error(f"API Error: {error}")

monitor = APIMonitor()

## 性能基准

在高并发场景下，预期性能指标：

+   **Haiku**: 处理 1000 个简单查询约需 10-15 分钟（并发 50）

+   **成本**： 约 $4.50（假设平均 2000 输入 + 500 输出 tokens）

+   **错误率**： 应低于 1%（包括超时和 API 错误）

### 5.3.3 成本敏感场景

成本敏感场景需要在有限预算内最大化 AI 能力的使用，适合初创公司、个人项目或大规模部署。

#### 典型场景

+   初创公司的 MVP 产品

+   教育和非营利项目

+   个人开发者的 Side Project

+   需要大规模部署的低利润率业务

+   实验性项目和原型开发

+   内部工具和自动化脚本

#### 模型选择策略

**核心原则： 能用 Haiku 就不用 Sonnet，能用 Sonnet 就不用 Opus**

1.  **默认使用 Haiku**

对于 70-80% 的常规任务，Claude Haiku 4.5 已经足够：

+   成本仅为 Sonnet 的 33%

+   响应速度最快

+   适合明确定义的任务

2.  **按需升级到 Sonnet**

仅在以下情况下使用 Sonnet:

+   Haiku 的输出质量不满足要求

+   需要更复杂的推理能力

+   代码生成任务（Sonnet 显著优于 Haiku）

+   需要计算机使用能力

+   需要处理办公自动化任务

3.  **极少使用 Opus**

Opus 应仅用于：

+   需要最高级别抽象推理的关键任务

+   其他模型多次尝试仍失败的场景

+   需要最高准确度的最终验证

3.  **智能上下文管理**

减少不必要的上下文传递：

``` 
class ContextManager:
    def __init__(self, max_context_tokens=10000):
        self.conversation_history = []
        self.max_context_tokens = max_context_tokens
    
    def add_message(self, role, content):
        self.conversation_history.append({"role": role, "content": content})
        self.trim_context()
    
    def trim_context(self):
        """保留最近的对话，丢弃旧消息"""
        total_tokens = sum(len(msg["content"]) // 4 for msg in self.conversation_history)
        
        while total_tokens > self.max_context_tokens and len(self.conversation_history) > 2:
            # 移除最旧的消息（保留系统消息）
            self.conversation_history.pop(1)
            total_tokens = sum(len(msg["content"]) // 4 for msg in self.conversation_history)
    
    def get_context(self):
        return self.conversation_history

# 使用示例
manager = ContextManager(max_context_tokens=5000)
manager.add_message("user", "你好")
manager.add_message("assistant", "你好！有什么可以帮助你的？")
# ... 更多对话 ...

# 仅发送必要的上下文
response = client.messages.create(
    model="claude-haiku-4-5",
    max_tokens=512,
    messages=manager.get_context()
)
```

5.  **本地预处理**

在调用 API 前进行本地预处理，减少 API 调用：

``` 
import re

def preprocess_query(query):
    """本地处理简单查询，避免 API 调用"""
    
    # FAQ 匹配
    faq_patterns = {
        r"营业时间|几点开门": "我们的营业时间是周一至周五 9:00-18:00",
        r"联系方式|电话|邮箱": "联系电话：400-123-4567，邮箱：support@example.com",
        r"地址|位置|在哪": "地址：北京市朝阳区xxx路xxx号",
    }
    
    for pattern, answer in faq_patterns.items():
        if re.search(pattern, query):
            return answer, False  # 不需要调用 API
    
    return query, True  # 需要调用 API

# 使用示例
query = "你们的营业时间是？"
processed_query, needs_api = preprocess_query(query)

if needs_api:
    response = call_claude_api(processed_query)
else:
    response = processed_query  # 直接返回预设答案
```
