from datetime import datetime, timedelta
import secrets

class KeyRotationManager:
    def __init__(self, rotation_interval_days=90):
        self.rotation_interval = timedelta(days=rotation_interval_days)
        self.keys = {}
        self.key_history = {}
    
    def create_key(self, service_name, user_id):
        """创建新密钥"""
        key_id = f"{service_name}_{user_id}_{secrets.token_hex(8)}"
        key_value = f"sk-{secrets.token_urlsafe(32)}"
        
        key_metadata = {
            'key_id': key_id,
            'key_value': key_value,
            'service': service_name,
            'user_id': user_id,
            'created_at': datetime.utcnow(),
            'expires_at': datetime.utcnow() + self.rotation_interval,
            'status': 'active',
            'rotation_count': 0
        }
        
        self.keys[key_id] = key_metadata
        
        return key_id, key_value
    
    def rotate_key(self, old_key_id, grace_period_days=7):
        """轮换密钥"""
        if old_key_id not in self.keys:
            raise ValueError(f"密钥 {old_key_id} 不存在")
        
        old_key = self.keys[old_key_id]
        
        # 创建新密钥
        new_key_id, new_key_value = self.create_key(
            old_key['service'],
            old_key['user_id']
        )
        
        # 更新新密钥的轮换计数
        self.keys[new_key_id]['rotation_count'] = old_key['rotation_count'] + 1
        
        # 将旧密钥标记为已弃用,但在宽限期内仍可使用
        old_key['status'] = 'deprecated'
        old_key['deprecated_at'] = datetime.utcnow()
        old_key['revoke_at'] = datetime.utcnow() + timedelta(days=grace_period_days)
        
        # 保存到历史记录
        if old_key_id not in self.key_history:
            self.key_history[old_key_id] = []
        self.key_history[old_key_id].append({
            'rotated_at': datetime.utcnow(),
            'new_key_id': new_key_id
        })
        
        return new_key_id, new_key_value
    
    def validate_key(self, key_value):
        """验证密钥是否有效"""
        for key_id, key_meta in self.keys.items():
            if key_meta['key_value'] == key_value:
                # 检查状态
                if key_meta['status'] == 'revoked':
                    return False, "密钥已被撤销"
                
                # 检查是否过期
                if datetime.utcnow() > key_meta['expires_at']:
                    return False, "密钥已过期"
                
                # 检查是否在宽限期内
                if key_meta['status'] == 'deprecated':
                    if datetime.utcnow() > key_meta['revoke_at']:
                        key_meta['status'] = 'revoked'
                        return False, "密钥已被撤销"
                    else:
                        return True, "密钥已弃用但仍在宽限期内"
                
                return True, "密钥有效"
        
        return False, "密钥不存在"
    
    def check_rotation_needed(self):
        """检查哪些密钥需要轮换"""
        keys_to_rotate = []
        
        for key_id, key_meta in self.keys.items():
            if key_meta['status'] != 'active':
                continue
            
            days_until_expiry = (key_meta['expires_at'] - datetime.utcnow()).days
            
            # 提前 7 天提醒轮换
            if days_until_expiry <= 7:
                keys_to_rotate.append({
                    'key_id': key_id,
                    'service': key_meta['service'],
                    'user_id': key_meta['user_id'],
                    'days_until_expiry': days_until_expiry
                })
        
        return keys_to_rotate
    
    def revoke_key(self, key_id):
        """立即撤销密钥"""
        if key_id in self.keys:
            self.keys[key_id]['status'] = 'revoked'
            self.keys[key_id]['revoked_at'] = datetime.utcnow()
            return True
        return False

# 使用示例
rotation_mgr = KeyRotationManager(rotation_interval_days=90)

# 创建密钥
key_id, key_value = rotation_mgr.create_key('openai', 'user123')
print(f"新密钥: {key_id}")

# 验证密钥
is_valid, message = rotation_mgr.validate_key(key_value)
print(f"密钥验证: {is_valid} - {message}")

# 轮换密钥
new_key_id, new_key_value = rotation_mgr.rotate_key(key_id, grace_period_days=7)
print(f"轮换后的新密钥: {new_key_id}")

# 检查需要轮换的密钥
keys_to_rotate = rotation_mgr.check_rotation_needed()
if keys_to_rotate:
    print("需要轮换的密钥:")
    for key_info in keys_to_rotate:
        print(f"  - {key_info['key_id']} (还有 {key_info['days_until_expiry']} 天过期)")
