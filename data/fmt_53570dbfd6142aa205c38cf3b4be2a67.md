        document.addEventListener('DOMContentLoaded', function() {
            // 卡片随机抽取功能
            const randomButton = document.getElementById('random-cards-button');
            const selectedCardsContainer = document.getElementById('selected-cards');
            const timerSection = document.getElementById('timer-section');
            const timerDisplay = document.getElementById('timer');
            const startTimerButton = document.getElementById('start-timer');
            const pauseTimerButton = document.getElementById('pause-timer');
            const resetTimerButton = document.getElementById('reset-timer');
            
            let cardElements = document.querySelectorAll('.card');
            let timer;
            let timeLeft = 300; // 5分钟 = 300秒
            let timerRunning = false;
            
            // 随机抽卡功能
            randomButton.addEventListener('click', function() {
                // 清空已选卡片
                selectedCardsContainer.innerHTML = '';
                
                // 随机选择3张卡片
                let cards = Array.from(cardElements);
                let selectedCards = [];
                
                // Fisher-Yates洗牌算法
                for (let i = cards.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [cards[i], cards[j]] = [cards[j], cards[i]];
                }
                
                // 取前3张
                selectedCards = cards.slice(0, 3);
                
                // 展示选中的卡片
                selectedCards.forEach(card => {
                    const clone = card.cloneNode(true);
                    const container = document.createElement('div');
                    container.className = 'selected-card-container';
                    container.appendChild(clone);
                    selectedCardsContainer.appendChild(container);
                });
                
                // 显示计时器
                timerSection.style.display = 'block';
                resetTimer();
            });
            
            // 计时器功能
            function updateTimerDisplay() {
                const minutes = Math.floor(timeLeft / 60);
                const seconds = timeLeft % 60;
                timerDisplay.textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
            }
            
            function startTimer() {
                if (!timerRunning) {
                    timerRunning = true;
                    timer = setInterval(function() {
                        timeLeft--;
                        updateTimerDisplay();
                        
                        if (timeLeft <= 0) {
                            clearInterval(timer);
                            timerRunning = false;
                            timerDisplay.textContent = "时间到!";
                        }
                    }, 1000);
                }
            }
            
            function pauseTimer() {
                clearInterval(timer);
                timerRunning = false;
            }
            
            function resetTimer() {
                clearInterval(timer);
                timerRunning = false;
                timeLeft = 300;
                updateTimerDisplay();
            }
            
            startTimerButton.addEventListener('click', startTimer);
            pauseTimerButton.addEventListener('click', pauseTimer);
            resetTimerButton.addEventListener('click', resetTimer);
        });
    </script>
