from dataclasses import dataclass
from typing import List, Dict, Optional
from enum import Enum

class StakeholderType(Enum):
    """利益相关者类型"""
    END_USER = "最终用户"
    BUSINESS_OWNER = "业务负责人"
    TECHNICAL_TEAM = "技术团队"
    COMPLIANCE = "合规团队"
    OPERATIONS = "运营团队"

@dataclass
class RequirementItem:
    """需求项"""
    id: str
    description: str
    stakeholder: StakeholderType
    priority: str  # 'must_have', 'should_have', 'nice_to_have'
    business_value: str
    technical_complexity: str  # 'low', 'medium', 'high'
    dependencies: List[str]
    acceptance_criteria: List[str]

class RequirementGathering:
    """需求调研系统"""

    def __init__(self, project_name: str):
        self.project_name = project_name
        self.requirements = []
        self.stakeholder_interviews = []
        self.pain_points = []
        self.success_metrics = []

    def conduct_stakeholder_interview(
        self,
        stakeholder: StakeholderType,
        questions: List[str]
    ) -> Dict:
        """进行利益相关者访谈"""
        interview_guide = {
            StakeholderType.END_USER: [
                "您当前的工作流程是什么?",
                "最耗时的任务是什么?",
                "您希望自动化哪些重复性工作?",
                "您对 AI 工具的期望是什么?",
                "您最担心的问题是什么?"
            ],
            StakeholderType.BUSINESS_OWNER: [
                "这个项目要解决什么业务问题?",
                "成功的量化指标是什么?",
                "预算和时间限制是什么?",
                "ROI 预期是什么?",
                "有哪些竞争产品或替代方案?"
            ],
            StakeholderType.TECHNICAL_TEAM: [
                "现有技术栈是什么?",
                "有哪些技术约束?",
                "集成点在哪里?",
                "数据可用性如何?",
                "安全和合规要求是什么?"
            ]
        }

        interview = {
            'stakeholder': stakeholder,
            'questions': interview_guide.get(stakeholder, questions),
            'responses': {},
            'pain_points': [],
            'opportunities': []
        }

        self.stakeholder_interviews.append(interview)
        return interview

    def identify_pain_points(self) -> List[Dict]:
        """识别痛点"""
        pain_point_template = {
            'description': '',
            'frequency': '',  # 'daily', 'weekly', 'monthly'
            'impact': '',     # 'high', 'medium', 'low'
            'current_solution': '',
            'cost_of_inaction': '',
            'affected_users': 0
        }

        # 从访谈中提取痛点
        for interview in self.stakeholder_interviews:
            for pain_point in interview.get('pain_points', []):
                self.pain_points.append(pain_point)

        # 按影响和频率排序
        self.pain_points.sort(
            key=lambda x: (
                {'high': 3, 'medium': 2, 'low': 1}[x.get('impact', 'low')],
                {'daily': 3, 'weekly': 2, 'monthly': 1}[x.get('frequency', 'monthly')]
            ),
            reverse=True
        )

        return self.pain_points

    def define_success_metrics(self) -> List[Dict]:
        """定义成功指标"""
        metric_categories = {
            'efficiency': [
                '任务完成时间减少 X%',
                '自动化率达到 X%',
                '人工干预次数减少 X%'
            ],
            'quality': [
                '错误率低于 X%',
                '用户满意度达到 X 分',
                '准确率高于 X%'
            ],
            'business': [
                '成本节省 $X',
                'ROI 达到 X%',
                '用户增长 X%'
            ],
            'adoption': [
                '活跃用户数达到 X',
                '日均使用次数达到 X',
                '留存率达到 X%'
            ]
        }

        for category, metrics in metric_categories.items():
            for metric in metrics:
                self.success_metrics.append({
                    'category': category,
                    'metric': metric,
                    'baseline': None,
                    'target': None,
                    'measurement_method': ''
                })

        return self.success_metrics

    def prioritize_requirements(self) -> List[RequirementItem]:
        """需求优先级排序"""
        # 使用 MoSCoW 方法
        must_have = [r for r in self.requirements if r.priority == 'must_have']
        should_have = [r for r in self.requirements if r.priority == 'should_have']
        nice_to_have = [r for r in self.requirements if r.priority == 'nice_to_have']

        # 按业务价值和技术复杂度评分
        def score_requirement(req: RequirementItem) -> float:
            value_score = {'high': 3, 'medium': 2, 'low': 1}
            complexity_score = {'low': 3, 'medium': 2, 'high': 1}

            value = value_score.get(req.business_value, 1)
            complexity = complexity_score.get(req.technical_complexity, 1)

            return value * complexity

        must_have.sort(key=score_requirement, reverse=True)
        should_have.sort(key=score_requirement, reverse=True)
        nice_to_have.sort(key=score_requirement, reverse=True)

        return must_have + should_have + nice_to_have

    def generate_requirement_doc(self, filepath: str) -> None:
        """生成需求文档"""
        doc = f"""
# {self.project_name} - 需求文档

## 1. 项目概述

## 2. 利益相关者访谈总结
访谈数量: {len(self.stakeholder_interviews)}

## 3. 核心痛点
"""
        for i, pain_point in enumerate(self.pain_points[:5], 1):
            doc += f"\n### 痛点 {i}: {pain_point.get('description', 'N/A')}\n"
            doc += f"- 频率: {pain_point.get('frequency', 'N/A')}\n"
            doc += f"- 影响: {pain_point.get('impact', 'N/A')}\n"
            doc += f"- 受影响用户: {pain_point.get('affected_users', 0)}\n"

        doc += "\n## 4. 功能需求\n"
        prioritized = self.prioritize_requirements()
        for req in prioritized:
            doc += f"\n### {req.id}: {req.description}\n"
            doc += f"- 优先级: {req.priority}\n"
            doc += f"- 业务价值: {req.business_value}\n"
            doc += f"- 技术复杂度: {req.technical_complexity}\n"

        doc += "\n## 5. 成功指标\n"
        for metric in self.success_metrics:
            doc += f"- [{metric['category']}] {metric['metric']}\n"

        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(doc)

        print(f"需求文档已生成: {filepath}")

# 使用示例
project = RequirementGathering("AI 写作助手")

# 进行利益相关者访谈
project.conduct_stakeholder_interview(
    StakeholderType.END_USER,
    []
)

project.conduct_stakeholder_interview(
    StakeholderType.BUSINESS_OWNER,
    []
)

# 添加需求
project.requirements.append(RequirementItem(
    id="REQ-001",
    description="自动生成文章大纲",
    stakeholder=StakeholderType.END_USER,
    priority="must_have",
    business_value="high",
    technical_complexity="medium",
    dependencies=[],
    acceptance_criteria=[
        "用户输入主题后 10 秒内生成大纲",
        "大纲包含 3-5 个主要章节",
        "每个章节包含 2-3 个子主题"
    ]
))

# 识别痛点
project.pain_points.append({
    'description': '手动撰写大纲耗时 30-60 分钟',
    'frequency': 'daily',
    'impact': 'high',
    'current_solution': '手动头脑风暴',
    'cost_of_inaction': '每天浪费 1 小时',
    'affected_users': 500
})

# 定义成功指标
project.define_success_metrics()

# 生成需求文档
project.generate_requirement_doc("requirements.md")
