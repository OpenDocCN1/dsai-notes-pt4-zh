from enum import Enum
from dataclasses import dataclass
from datetime import datetime
from typing import List, Optional

class IncidentSeverity(Enum):
    """事件严重程度"""
    LOW = 1        # 低 - 轻微影响,无数据泄露
    MEDIUM = 2     # 中 - 有限影响,可能的数据暴露
    HIGH = 3       # 高 - 重大影响,确认的数据泄露
    CRITICAL = 4   # 严重 - 灾难性影响,大规模数据泄露或系统妥协

class IncidentType(Enum):
    """事件类型"""
    PROMPT_INJECTION = "prompt_injection"
    DATA_LEAK = "data_leak"
    TOXIC_CONTENT = "toxic_content_generation"
    HALLUCINATION = "critical_hallucination"
    BIAS_INCIDENT = "bias_discrimination"
    UNAUTHORIZED_ACCESS = "unauthorized_access"
    MALICIOUS_SKILL = "malicious_skill_execution"
    SYSTEM_COMPROMISE = "system_compromise"

@dataclass
class SecurityIncident:
    """安全事件"""
    incident_id: str
    incident_type: IncidentType
    severity: IncidentSeverity
    description: str
    timestamp: datetime
    affected_systems: List[str]
    affected_users: List[str]
    detected_by: str
    status: str = "open"  # open, investigating, contained, resolved
    response_actions: List[str] = None
    root_cause: Optional[str] = None
    lessons_learned: Optional[str] = None
    
    def __post_init__(self):
        if self.response_actions is None:
            self.response_actions = []

class IncidentResponseSystem:
    """事件响应系统"""
    
    def __init__(self):
        self.incidents = []
        self.response_playbooks = self._initialize_playbooks()
    
    def _initialize_playbooks(self) -> Dict[IncidentType, List[str]]:
        """初始化事件响应手册"""
        return {
            IncidentType.PROMPT_INJECTION: [
                "1. 立即隔离受影响的 agent 实例",
                "2. 审查最近的 agent 操作日志",
                "3. 识别注入向量(邮件/网页/文档)",
                "4. 回滚任何未经授权的更改",
                "5. 加强输入验证规则",
                "6. 通知受影响的用户",
                "7. 更新系统提示和安全策略",
            ],
            IncidentType.DATA_LEAK: [
                "1. 立即停止数据传输",
                "2. 识别泄露的数据范围和敏感度",
                "3. 确定数据接收方和暴露时长",
                "4. 通知数据保护官(DPO)和法务团队",
                "5. 在72小时内向监管机构报告(如适用GDPR)",
                "6. 通知受影响的数据主体",
                "7. 实施补救措施防止再次发生",
                "8. 文档化事件和响应过程",
            ],
            IncidentType.TOXIC_CONTENT: [
                "1. 立即下架有毒内容",
                "2. 识别内容生成的提示和参数",
                "3. 审查内容审核系统的失效点",
                "4. 加强毒性检测阈值",
                "5. 审查类似内容",
                "6. 向受影响方道歉",
                "7. 更新内容政策和过滤器",
            ],
            IncidentType.MALICIOUS_SKILL: [
                "1. 立即从 ClawHub 下架恶意 Skill",
                "2. 识别所有安装了该 Skill 的用户",
                "3. 发送紧急安全通知",
                "4. 提供卸载和系统扫描指南",
                "5. 分析 Skill 的恶意行为",
                "6. 加强 Skill 审核流程",
                "7. 考虑对上传者采取法律行动",
            ],
        }
    
    def report_incident(
        self,
        incident_type: IncidentType,
        severity: IncidentSeverity,
        description: str,
        affected_systems: List[str],
        affected_users: List[str],
        detected_by: str
    ) -> SecurityIncident:
        """报告安全事件"""
        incident_id = f"INC-{datetime.now().strftime('%Y%m%d-%H%M%S')}"
        
        incident = SecurityIncident(
            incident_id=incident_id,
            incident_type=incident_type,
            severity=severity,
            description=description,
            timestamp=datetime.now(),
            affected_systems=affected_systems,
            affected_users=affected_users,
            detected_by=detected_by
        )
        
        self.incidents.append(incident)
        
        # 自动触发响应流程
        self._initiate_response(incident)
        
        return incident
    
    def _initiate_response(self, incident: SecurityIncident) -> None:
        """启动事件响应"""
        print(f"
{'='*60}")
        print(f"🚨 安全事件报告: {incident.incident_id}")
        print(f"{'='*60}")
        print(f"类型: {incident.incident_type.value}")
        print(f"严重程度: {incident.severity.name}")
        print(f"描述: {incident.description}")
        print(f"时间: {incident.timestamp}")
        print(f"受影响系统: {', '.join(incident.affected_systems)}")
        print(f"受影响用户: {len(incident.affected_users)} 人")
        print(f"检测者: {incident.detected_by}")
        
        # 获取响应手册
        playbook = self.response_playbooks.get(incident.incident_type, [])
        if playbook:
            print(f"
📋 响应手册:")
            for step in playbook:
                print(f"  {step}")
        
        # 根据严重程度触发不同级别的响应
        if incident.severity in [IncidentSeverity.CRITICAL, IncidentSeverity.HIGH]:
            print(f"
⚠️ 高严重程度事件 - 已通知:")
            print(f"  - 安全团队")
            print(f"  - 事件响应团队")
            print(f"  - 高级管理层")
            if incident.severity == IncidentSeverity.CRITICAL:
                print(f"  - CEO/CTO")
                print(f"  - 法务团队")
                print(f"  - 公关团队")
        
        print(f"{'='*60}
")
    
    def update_incident(
        self,
        incident_id: str,
        status: Optional[str] = None,
        response_action: Optional[str] = None,
        root_cause: Optional[str] = None,
        lessons_learned: Optional[str] = None
    ) -> bool:
        """更新事件状态"""
        incident = self._find_incident(incident_id)
        if not incident:
            return False
        
        if status:
            incident.status = status
        if response_action:
            incident.response_actions.append(response_action)
        if root_cause:
            incident.root_cause = root_cause
        if lessons_learned:
            incident.lessons_learned = lessons_learned
        
        return True
    
    def _find_incident(self, incident_id: str) -> Optional[SecurityIncident]:
        """查找事件"""
        for incident in self.incidents:
            if incident.incident_id == incident_id:
                return incident
        return None
    
    def get_open_incidents(self) -> List[SecurityIncident]:
        """获取未解决的事件"""
        return [i for i in self.incidents if i.status != "resolved"]
    
    def generate_incident_report(self, incident_id: str) -> str:
        """生成事件报告"""
        incident = self._find_incident(incident_id)
        if not incident:
            return "事件未找到"
        
        report = f"""
安全事件报告
{'='*60}

事件 ID: {incident.incident_id}
类型: {incident.incident_type.value}
严重程度: {incident.severity.name}
状态: {incident.status}

时间线:
- 发生时间: {incident.timestamp}
- 检测者: {incident.detected_by}

影响范围:
- 受影响系统: {', '.join(incident.affected_systems)}
- 受影响用户: {len(incident.affected_users)} 人

事件描述:
{incident.description}

响应措施:
"""
        for i, action in enumerate(incident.response_actions, 1):
            report += f"{i}. {action}
"
        
        if incident.root_cause:
            report += f"
根本原因:
{incident.root_cause}
"
        
        if incident.lessons_learned:
            report += f"
经验教训:
{incident.lessons_learned}
"
        
        report += f"
{'='*60}
"
        
        return report

# 使用示例
irs = IncidentResponseSystem()

# 报告 Prompt Injection 事件
incident = irs.report_incident(
    incident_type=IncidentType.PROMPT_INJECTION,
    severity=IncidentSeverity.HIGH,
    description="检测到通过电子邮件的间接 prompt injection 攻击,agent 尝试访问敏感文件",
    affected_systems=["openclaw-gateway-prod-01", "email-processor"],
    affected_users=["user_12345"],
    detected_by="AgentBehaviorMonitor"
)

# 更新事件
irs.update_incident(
    incident_id=incident.incident_id,
    status="contained",
    response_action="已隔离受影响的 agent 实例",
)

irs.update_incident(
    incident_id=incident.incident_id,
    response_action="已加强邮件内容清洗规则",
)

irs.update_incident(
    incident_id=incident.incident_id,
    status="resolved",
    root_cause="邮件处理器未对外部内容进行充分清洗,允许嵌入的指令被模型执行",
    lessons_learned="实施两阶段处理:使用只读 agent 总结邮件,主 agent 仅处理摘要"
)

# 生成报告
report = irs.generate_incident_report(incident.incident_id)
print(report)
