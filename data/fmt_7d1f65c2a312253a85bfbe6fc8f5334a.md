from typing import List, Dict
from dataclasses import dataclass
from datetime import datetime

@dataclass
class TestParticipant:
    """测试参与者"""
    id: str
    name: str
    role: str  # 'target_user', 'power_user', 'novice'
    background: str
    recruited_date: datetime

@dataclass
class TestTask:
    """测试任务"""
    id: str
    description: str
    expected_outcome: str
    success_criteria: List[str]
    difficulty: str  # 'easy', 'medium', 'hard'

@dataclass
class TestResult:
    """测试结果"""
    participant_id: str
    task_id: str
    completed: bool
    time_taken_seconds: int
    errors_encountered: List[str]
    satisfaction_score: int  # 1-5
    feedback: str

class UserTesting:
    """用户测试系统"""
    
    def __init__(self, product_name: str):
        self.product_name = product_name
        self.participants = []
        self.tasks = []
        self.results = []
        self.insights = []
    
    def recruit_participants(self, target_count: int = 5) -> List[TestParticipant]:
        """招募测试参与者"""
        # 遵循 Nielsen 的 5 用户测试原则
        # 5 个用户可以发现 85% 的可用性问题
        
        participant_profiles = [
            {
                'role': 'target_user',
                'count': 3,
                'criteria': [
                    '符合目标用户画像',
                    '有相关使用经验',
                    '愿意提供反馈'
                ]
            },
            {
                'role': 'power_user',
                'count': 1,
                'criteria': [
                    '技术背景强',
                    '使用过类似产品',
                    '能提供深度反馈'
                ]
            },
            {
                'role': 'novice',
                'count': 1,
                'criteria': [
                    '首次使用此类产品',
                    '代表新手用户',
                    '能发现入门问题'
                ]
            }
        ]
        
        print(f"招募 {target_count} 名测试参与者:")
        for profile in participant_profiles:
            print(f"- {profile['role']}: {profile['count']} 人")
            for criterion in profile['criteria']:
                print(f"  * {criterion}")
        
        return self.participants
    
    def design_test_tasks(self) -> List[TestTask]:
        """设计测试任务"""
        tasks = [
            TestTask(
                id="T001",
                description="注册新账户并完成首次登录",
                expected_outcome="用户成功创建账户并登录",
                success_criteria=[
                    "5 分钟内完成注册",
                    "无需外部帮助",
                    "理解每个步骤"
                ],
                difficulty="easy"
            ),
            TestTask(
                id="T002",
                description="使用核心功能完成一个完整任务",
                expected_outcome="用户成功使用 AI 生成内容",
                success_criteria=[
                    "10 分钟内完成任务",
                    "生成的内容符合预期",
                    "理解 AI 的工作方式"
                ],
                difficulty="medium"
            ),
            TestTask(
                id="T003",
                description="编辑和优化 AI 生成的内容",
                expected_outcome="用户能够自定义和改进输出",
                success_criteria=[
                    "找到编辑功能",
                    "成功修改内容",
                    "保存修改"
                ],
                difficulty="medium"
            ),
            TestTask(
                id="T004",
                description="遇到问题时寻找帮助",
                expected_outcome="用户能够自助解决问题",
                success_criteria=[
                    "找到帮助文档",
                    "理解错误信息",
                    "成功解决问题"
                ],
                difficulty="hard"
            )
        ]
        
        self.tasks = tasks
        return tasks
    
    def conduct_test_session(
        self,
        participant: TestParticipant,
        tasks: List[TestTask]
    ) -> List[TestResult]:
        """进行测试会话"""
        print(f"\n开始测试会话: {participant.name} ({participant.role})")
        print("=" * 60)
        
        session_results = []
        
        for task in tasks:
            print(f"\n任务: {task.description}")
            print(f"难度: {task.difficulty}")
            
            # 观察用户完成任务
            # 实际测试中,这里会记录用户的实际操作
            
            result = TestResult(
                participant_id=participant.id,
                task_id=task.id,
                completed=True,  # 实际测试中由观察者判断
                time_taken_seconds=0,  # 实际测试中计时
                errors_encountered=[],  # 实际测试中记录
                satisfaction_score=0,  # 实际测试中询问
                feedback=""  # 实际测试中收集
            )
            
            session_results.append(result)
            self.results.append(result)
        
        return session_results
    
    def analyze_results(self) -> Dict:
        """分析测试结果"""
        if not self.results:
            return {}
        
        # 任务完成率
        task_completion = {}
        for task in self.tasks:
            task_results = [r for r in self.results if r.task_id == task.id]
            completed = sum(1 for r in task_results if r.completed)
            task_completion[task.id] = {
                'task': task.description,
                'completion_rate': (completed / len(task_results) * 100) if task_results else 0,
                'avg_time': sum(r.time_taken_seconds for r in task_results) / len(task_results) if task_results else 0
            }
        
        # 总体满意度
        avg_satisfaction = sum(r.satisfaction_score for r in self.results) / len(self.results)
        
        # 常见问题
        all_errors = []
        for result in self.results:
            all_errors.extend(result.errors_encountered)
        
        error_frequency = {}
        for error in all_errors:
            error_frequency[error] = error_frequency.get(error, 0) + 1
        
        top_issues = sorted(error_frequency.items(), key=lambda x: x[1], reverse=True)[:5]
        
        return {
            'task_completion': task_completion,
            'avg_satisfaction': avg_satisfaction,
            'top_issues': top_issues,
            'total_participants': len(set(r.participant_id for r in self.results)),
            'total_tasks': len(self.tasks)
        }
    
    def generate_insights(self, analysis: Dict) -> List[str]:
        """生成洞察"""
        insights = []
        
        # 基于完成率生成洞察
        for task_id, data in analysis['task_completion'].items():
            if data['completion_rate'] < 80:
                insights.append(
                    f"⚠️ 任务 '{data['task']}' 完成率较低 ({data['completion_rate']:.1f}%),需要优化"
                )
        
        # 基于满意度生成洞察
        if analysis['avg_satisfaction'] < 3.5:
            insights.append(
                f"⚠️ 总体满意度较低 ({analysis['avg_satisfaction']:.1f}/5),需要改进用户体验"
            )
        elif analysis['avg_satisfaction'] >= 4.0:
            insights.append(
                f"✅ 总体满意度良好 ({analysis['avg_satisfaction']:.1f}/5)"
            )
        
        # 基于常见问题生成洞察
        if analysis['top_issues']:
            insights.append(f"🔍 发现 {len(analysis['top_issues'])} 个高频问题:")
            for issue, count in analysis['top_issues']:
                insights.append(f"  - {issue} (出现 {count} 次)")
        
        self.insights = insights
        return insights
    
    def generate_test_report(self) -> Dict:
        """生成测试报告"""
        analysis = self.analyze_results()
        insights = self.generate_insights(analysis)
        
        return {
            'product': self.product_name,
            'test_date': datetime.now().strftime('%Y-%m-%d'),
            'participants': len(self.participants),
            'tasks': len(self.tasks),
            'analysis': analysis,
            'insights': insights,
            'recommendations': self._generate_recommendations(insights)
        }
    
    def _generate_recommendations(self, insights: List[str]) -> List[str]:
        """生成改进建议"""
        recommendations = []
        
        for insight in insights:
            if '完成率较低' in insight:
                recommendations.append("简化任务流程,添加更多引导")
            elif '满意度较低' in insight:
                recommendations.append("改进 UI/UX,优化用户体验")
            elif '高频问题' in insight:
                recommendations.append("优先修复高频问题,改进错误提示")
        
        return recommendations

# 使用示例
testing = UserTesting("AI 写作助手 MVP")

# 招募参与者
testing.recruit_participants(5)

# 设计测试任务
tasks = testing.design_test_tasks()

# 生成测试报告
report = testing.generate_test_report()
print(f"\n测试报告:")
print(f"产品: {report['product']}")
print(f"参与者: {report['participants']} 人")
print(f"任务数: {report['tasks']} 个")
