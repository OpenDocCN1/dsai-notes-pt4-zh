```python
from typing import TypedDict, List, Dict, Union

# 定义输入参数类型
class Input(TypedDict):
    input: Union[List[str], None]  # 修改：允许 input 为 None

# 定义输出参数类型
class Output(TypedDict):
    array_num: int
    is_valid_input: int

async def main(args) -> Output:
    params = args['params']
    input_data = params['input']

    if input_data is None:
        is_valid_input = 0
        array_num = 0  #  当 input 为 None 时，array_num 设为 0
    else:
        is_valid_input = 1
        array_num = len(input_data)

    return {
        'array_num': array_num,
        'is_valid_input': is_valid_input
    }
```

```
"""
功能说明：通过阿里云的服务，将音频文件转换为文本内容。
"""

from typing import TypedDict
import dashscope
import requests

class Input(TypedDict):
    api_key: str
    file_url: str

class Output(TypedDict):
    data: str
    error_message: str

def handler(args) -> Output:
    """
    示例用法:
    ```python

    args = {

        "input": {

            "api_key": "your_api_key",

            "file_url": "https://example.com/audio.mp3"

        }

    }

    result = handler(args)

    print(result)

    ```
    """
    try:
        api_key = args.input.api_key
        file_url = args.input.file_url

        dashscope.api_key = api_key

        task_response = dashscope.audio.asr.Transcription.async_call(
            model='paraformer-v1',
            file_urls=[file_url]
        )

        transcribe_response = dashscope.audio.asr.Transcription.wait(task=task_response.output.task_id)

        if transcribe_response.status_code == requests.codes.ok:
            transcription_data = transcribe_response.output
            transcription_url = transcription_data['results'][0]['transcription_url']

            response = requests.get(transcription_url)
            if response.status_code == requests.codes.ok:
                result = response.json()
                text = result['transcripts'][0]['text']
                return {"answer": text, "error_message": ""}
            else:
                raise Exception(f"Failed to fetch transcription content from {transcription_url}")
        else:
            raise Exception("Transcription task failed")

    except Exception as e:
        error_message = f"发生错误: {str(e)}"
        return {"answer": "", "error_message": error_message}
```

#### 3.6 文案纠错

这里我插入了一个工作流，主要目的是文案纠错

![](img/3f4bd99b412e232fa827aba3d1a476ef.png)

![](img/6c7d9842db49316789e952690662d61c.png)

主要流程就是把长文案根据标点符号或者换行符，拆分成一段一段的数组，然后每段分别取用AI模型进行纠错

> 原先写这个的子工作流的时候，主要调用的claude模型支持一次输出的token数量有限，遇到千字长文案甚至万字长文案的时候就会报错，所以采用这个方案，理论上采用这个方案，10万字的文案也不会报错

当然现在coze内置了deepseek-v3，可以直接调整回复长度最大16384token，约等于25000个文字，足以应对99.99%的情况了

![](img/61955b606009cd09201c7ec56238f301.png)

因此我们可以直接插入大模型节点来代替原来的子流程
