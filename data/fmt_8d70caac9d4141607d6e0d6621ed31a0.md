class GDPRCompliance:
    def __init__(self):
        self.data_inventory = {}
        self.consent_records = {}
        self.processing_activities = []
    
    def register_data_processing(self, purpose, legal_basis, 
                                 data_categories, retention_period):
        """注册数据处理活动"""
        activity = {
            'purpose': purpose,
            'legal_basis': legal_basis,  # consent, contract, legal_obligation, etc.
            'data_categories': data_categories,
            'retention_period': retention_period,
            'registered_at': datetime.utcnow()
        }
        
        self.processing_activities.append(activity)
        return activity
    
    def collect_consent(self, user_id, purposes, consent_text):
        """收集用户同意"""
        consent_record = {
            'user_id': user_id,
            'purposes': purposes,
            'consent_text': consent_text,
            'consented_at': datetime.utcnow(),
            'ip_address': self._get_client_ip(),
            'user_agent': self._get_user_agent(),
            'consent_method': 'explicit',
            'version': '1.0'
        }
        
        self.consent_records[user_id] = consent_record
        return consent_record
    
    def withdraw_consent(self, user_id):
        """撤回同意"""
        if user_id in self.consent_records:
            self.consent_records[user_id]['withdrawn_at'] = datetime.utcnow()
            self.consent_records[user_id]['status'] = 'withdrawn'
            
            # 触发数据删除流程
            self._trigger_data_deletion(user_id)
    
    def handle_dsar(self, user_id, request_type):
        """处理数据主体访问请求 (DSAR)"""
        if request_type == 'access':
            return self._export_user_data(user_id)
        elif request_type == 'delete':
            return self._delete_user_data(user_id)
        elif request_type == 'rectify':
            return self._rectify_user_data(user_id)
        elif request_type == 'portability':
            return self._export_portable_data(user_id)
    
    def conduct_dpia(self, processing_description, risks):
        """进行数据保护影响评估 (DPIA)"""
        dpia = {
            'processing_description': processing_description,
            'necessity_assessment': self._assess_necessity(processing_description),
            'risks_identified': risks,
            'mitigation_measures': self._identify_mitigations(risks),
            'conducted_at': datetime.utcnow(),
            'conducted_by': 'DPO'
        }
        
        return dpia

        return "127.0.0.1"  # 实际实现中从请求获取

    def _trigger_data_deletion(self, user_id):

    def _export_user_data(self, user_id):
        return {"user_id": user_id, "data": "..."}

## 2026 年 GDPR 重点关注领域

根据最新指导，2026 年 GDPR 执法重点包括：

1. **AI 系统的自主性**：[技术层面的自主性](https://www.insideprivacy.com/artificial-intelligence/spanish-supervisory-authority-issues-detailed-guidance-on-agentic-ai-and-gdpr-compliance/)不改变 GDPR 下的法律责任分配。虽然 AI Agent 可能在实践中自主执行数据处理操作，但处理在法律上仍归属于部署系统并确定其目的和基本手段的控制者（或处理者）。citation

2. **LLM 匿名化**：[EDPB 明确](https://secureprivacy.ai/blog/gdpr-compliance-2026)指出，大语言模型很少达到匿名化标准。部署第三方 LLM 的控制者必须进行全面的合法利益评估。citation

3. **影子 AI**：[组织内用户以不](https://techgdpr.com/blog/data-protection-digest-03012026-improvements-are-being-made-to-gdpr-enforcement-us-consumer-privacy-and-emerging-shadow-ai/)受监管、不透明、不协调的方式使用 AI 系统的现象正在成为新的关注点。必须应用适当的保护措施。citation
