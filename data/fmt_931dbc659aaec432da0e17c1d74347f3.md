
```
import numpy as np

a = np.array([[1,2],[3,4]])
b = np.array([[11,12],[13,14]])
print(np.dot(a, b))
# [[1*11+2*13, 1*12+2*14],[3*11+4*13, 3*12+4*14]]
# [[37  40] 
#  [85  92]]

print(np.matmul(a, b)) # 同上
print(np.vdot(a, b)) # 130
# 1*11 + 2*12 + 3*13 + 4*14 = 130

print(np.linalg.det(a)) # -2.0
print(np.linalg.inv(a))
# [[-2.   1. ]
#  [ 1.5 -0.5]]
```
