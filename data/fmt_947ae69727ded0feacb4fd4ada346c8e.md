import json
async def main(args: Args) -> Output:
    params = args.params
    scene_prompt = params["scene_prompt"]
    pic_array = params["pictures"]
    titles = params["titles"]
    row_array = []
    row_array.append(titles)

    for i in range(len(pic_array)):
