```
import requests
from typing import Dict, List, Optional

class ContentModerator:
    """内容审核系统"""
    
    TOXICITY_CATEGORIES = [
        'hate_speech',       # 仇恨言论
        'harassment',        # 骚扰
        'violence',          # 暴力
        'sexual_content',    # 性内容
        'self_harm',         # 自残
        'profanity',         # 亵渎
        'spam',              # 垃圾信息
        'misinformation',    # 错误信息
    ]
    
    def __init__(self, api_key: str, policy_id: str):
        self.api_key = api_key
        self.policy_id = policy_id
        self.api_endpoint = "https://api.vettly.com/v1/moderate"
    
    def moderate_content(
        self,
        content: str,
        content_type: str = 'text',
        threshold: float = 0.7
    ) -> Dict:
        """
        审核内容
        
        Args:
            content: 要审核的内容
            content_type: 内容类型 ('text', 'image', 'video')
            threshold: 毒性阈值 (0-1)
        
        Returns:
            审核结果字典
        """
        payload = {
            'policy_id': self.policy_id,
            'content': content,
            'content_type': content_type
        }
        
        headers = {
            'Authorization': f'Bearer {self.api_key}',
            'Content-Type': 'application/json'
        }
        
        try:
            response = requests.post(
                self.api_endpoint,
                json=payload,
                headers=headers,
                timeout=10
            )
            response.raise_for_status()
            result = response.json()
            
            # 解析结果
            return self._parse_moderation_result(result, threshold)
            
        except requests.exceptions.RequestException as e:
            return {
                'error': str(e),
                'is_safe': False,
                'action': 'block'
            }
    
    def _parse_moderation_result(self, result: Dict, threshold: float) -> Dict:
        """解析审核结果"""
        scores = result.get('category_scores', {})
        
        # 检查是否有任何类别超过阈值
        violations = {
            category: score
            for category, score in scores.items()
            if score > threshold
        }
        
        is_safe = len(violations) == 0
        action = result.get('action', 'allow' if is_safe else 'block')
        
        return {
            'is_safe': is_safe,
            'action': action,
            'violations': violations,
            'scores': scores,
            'provider': result.get('provider'),
            'latency_ms': result.get('latency_ms'),
            'cost': result.get('cost')
        }
    
    def batch_moderate(self, contents: List[str]) -> List[Dict]:
        """批量审核"""
        return [self.moderate_content(content) for content in contents]

# 使用示例
moderator = ContentModerator(
    api_key="your_vettly_api_key",
    policy_id="your_policy_id"
)

test_content = "这是一段需要审核的用户生成内容..."
result = moderator.moderate_content(test_content, threshold=0.7)

if result['is_safe']:
    print("✅ 内容安全,可以发布")
else:
    print(f"⚠️ 内容包含违规: {result['violations']}")
    print(f"建议操作: {result['action']}")
```
