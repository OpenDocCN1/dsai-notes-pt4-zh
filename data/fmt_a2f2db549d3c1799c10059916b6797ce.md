```python
from itertools import permutations, product

def calc(a, b, op):
    """根据运算符计算两个数字"""
    if op == '+':
        return a + b
    elif op == '-':
        return a - b
    elif op == '*':
        return a * b
    elif op == '/':
        if b != 0:
            return a / b
        else:
            return None

def try_operations(numbers, target=24):
    """尝试所有可能的运算符组合来达到目标值"""
    if len(numbers) == 1:
        return abs(numbers[0] - target) < 1e-6

    for perm in permutations(numbers):
        for ops in product('+-*/', repeat=len(numbers)-1):
            stack = [perm[0]]
            for num, op in zip(perm[1:], ops):
                if op in '+*' and (stack[-1], op) not in [('*', '+'), ('+', '*')]:
                    stack.append(num)
                    stack.append(op)
                else:
                    b = stack.pop()
                    a = stack.pop() if stack else 0
                    stack.append(calc(a, b, op))
                    stack.append(num)
            result = stack.pop()
            if try_operations([result], target):
                return True
    return False

def can_reach_24(numbers):
    """检查是否可以从给定的数字中计算出24"""
    if not numbers or len(numbers) != 4:
        print("需要提供四个数字。")
        return False
    return try_operations(numbers)

# 测试
numbers = [3, 3, 8, 8]  # 例子输入
if can_reach_24(numbers):
    print(f"{numbers} 可以组合成24。")
else:
    print(f"{numbers} 不能组合成24。")
```
