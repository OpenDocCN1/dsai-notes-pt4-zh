
## server端代码：
```
from __future__ import annotations
import time
from typing import Sequence
from absl import app
from absl import flags
from absl import logging
from vizier.service import vizier_server
from vizier.service import vizier_service_pb2_grpc
import grpc
from concurrent import futures

flags.DEFINE_string(
    'host',
    'localhost',
    'Host location for the server. For distributed cases, use the IP address.',
)

FLAGS = flags.FLAGS

_ONE_DAY_IN_SECONDS = 60 * 60 * 24

def main(argv: Sequence[str]) -> None:
  if len(argv) > 1:
    raise app.UsageError('Too many command-line arguments.')
  hostname = 'localhost'  # Example ; usually user - specified
  port = 6006  # Example ; usually user - specified
  address = f'{hostname}:{port}'
  servicer = vizier_server.vizier_service.VizierServicer()
  server = grpc.server(futures.ThreadPoolExecutor(max_workers=100))
  vizier_service_pb2_grpc.add_VizierServiceServicer_to_server(servicer, server)
  server.add_secure_port(address, grpc.local_server_credentials())
  server.start()
  # prevent the main thread from exiting
  try:
    while True:
      time.sleep(_ONE_DAY_IN_SECONDS)
  except KeyboardInterrupt:
    del server

if __name__ == '__main__':
  app.run(main)
```

```
from __future__ import annotations

import argparse

MAX_TRIALS_PER_CLIENT = 25
MAX_PROCS = 4

def get_scale_type(sType):
    if sType == 'LOG':
        return vz.ScaleType.LOG
    if sType == 'LINEAR':
        return vz.ScaleType.LINEAR
    if sType == 'REVERSE_LOG':
        return vz.ScaleType.REVERSE_LOG
    if sType == 'UNIFORM_DISCRETE':
        return vz.ScaleType.UNIFORM_DISCRETE
    return vz.ScaleType.LINEAR

def creat_client(config, server='localhost:45437', owner='a', study_id='gmnet', client_id='0'):
    clients.environment_variables.server_endpoint = server
    study_config = vz.StudyConfig()  # Search space, metrics, and algorithm.
    root = study_config.search_space.root
    if config == None:
        study = clients.Study.from_owner_and_id(owner, study_id)
        return study

    for key in config.keys():
        sType = get_scale_type(config[key]['scale_type']),
        if config[key]['type'] == 'float':
            root.add_float_param(
                key,
                min_value=config[key]['min'],
                max_value=config[key]['max'],
                scale_type=sType[0],
            )
        elif config[key]['type'] == 'int':
            root.add_int_param(
                key,
                min_value=config[key]['min'],
                max_value=config[key]['max'],
                scale_type=sType[0],
            )

    study_config.metric_information.append(
        vz.MetricInformation(
            name='mIoU',
            goal=vz.ObjectiveMetricGoal.MAXIMIZE,
            min_value=0.0,
            max_value=1.0,
        )
    )

    study_config.algorithm = vz.Algorithm.QUASI_RANDOM_SEARCH

    client = vizier_client.create_or_load_study(
        server,
        owner_id=owner,
        client_id=client_id,
        study_id=study_id,
        study_config=study_config,
    )
    return client

def main_process(config, search_keys, client):
    print(client)
    for _ in range(MAX_TRIALS_PER_CLIENT):
        trials = client.get_suggestions(suggestion_count=1)
        for trial in trials:
            materialized_trial = trial
            print(materialized_trial)
            for key in search_keys:
                print(key)
                value = materialized_trial.parameters.get_value(key)
                print(value)
                config[key] = value
            config['trial_id'] = trial.id
            ioU = main_train(config)
            m = vz.Measurement()
            m.metrics = {'mIoU': ioU}
            client.complete_trial(trial.id, m)
            logging.info(
            'Trial %d completed with metrics: %s', trial.id, m.metrics
            )

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Train MFNet with pytorch')
    parser.add_argument('--config', '-c', type=str, default='./config/tokyo.config')
    parser.add_argument('--gpu', '-g', type=int, default=-1)
    args = parser.parse_args()

    if args.config:
        file_address = open(args.config)
        config = yaml.safe_load(file_address)
    else:
        print('no config in cmd failed.!')
    search_sp = config['search_space']
    base_cfg = config['base_cfg']
    if base_cfg:
        file_address = open(base_cfg)
        base_cfg = yaml.safe_load(file_address)
    else:
        print('no base config.!')
    if args.gpu >= 0:
        base_cfg['gpu_ids'] = [args.gpu]
    for key in base_cfg.keys():
        if isinstance(base_cfg[key], str):
            if ',' in base_cfg[key]:
                base_cfg[key] = get_conf_list(base_cfg[key])
    client = creat_client(search_sp, server='ip:port', owner='e', client_id='4')
    main_process(base_cfg, search_sp.keys(), client)
    trial_filter = vz.TrialFilter(status=[vz.TrialStatus.COMPLETED])
    all_trials = client.list_trials()
    trials = []
    for t in filter(trial_filter, all_trials):
        trials.append(t)
    if len(trials) >= MAX_TRIALS_PER_CLIENT * MAX_PROCS:
        if client.get_study_state() == vz.StudyState.COMPLETED:
            client.set_study_state(vz.StudyState.COMPLETED)
        optimal_trials = client.list_optimal_trials()
        print(optimal_trials)
        for optimal_trial in optimal_trials:
            print(f"Pareto-optimal trial found so far has parameters {optimal_trial.parameters} and metrics {optimal_trial.final_measurement.metrics}")
```

```
base_cfg: './config/tokyo_gmnet_train.config'

  backbone_lr:
    type: 'float'
    min: 0.00001
    max: 0.001
    scale_type: 'LINEAR' # LINEAR = 'LINEAR' LOG = 'LOG' REVERSE_LOG = 'REVERSE_LOG'  UNIFORM_DISCRETE = 'UNIFORM_DISCRETE'
```

> 原计划直接用多进程编程。但是多进程加载`cuda`库的时候会报错。而且如果在多进程的话，所有`vizier`的初始化代码都需要再进程在实现，否则会报错。

故直接使用`bash`脚本进行调用：

```
#!/bin/sh
export OMP_NUM_THREADS=32
#python -u -m torch.distributed.run --rdzv_backend=c10d --rdzv_endpoint=localhost:0 --nnodes=1 --nproc_per_node=4 trainV2.py -c $1
if [ 0 -gt $2 ] ; then
  echo "start on multi gpus"
  (python  run_vizier_client.py -c $1 -g 0 > search0.log) & (python  run_vizier_client.py -c $1 -g 1 > search1.log) & (python  run_vizier_client.py -c $1 -g 2 > search2.log) &(python  run_vizier_client.py -c $1 -g 3 > search3.log)
else
  echo "start on gpu $2"
  python  run_vizier_client.py -c $1 -g $2
fi
```
