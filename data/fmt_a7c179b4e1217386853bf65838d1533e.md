
# 简单 RAG 示例代码与说明

```python
import os
import json # json 模块在此代码片段中未被直接使用，但保留以防未来扩展
from openai import OpenAI, APIError, AuthenticationError
import numpy as np
import traceback

# --- 安全地获取 API Key ---
AOAI_API_KEY = os.getenv("AOAI_API_KEY") # 使用大写更符合环境变量的习惯

if not AOAI_API_KEY:
    raise ValueError("请设置环境变量 AOAI_API_KEY")

# --- 初始化 OpenAI 客户端 ---
# 由于 embeddings 和 chat API 都指向同一个 base_url 并使用相同的 key，
# 我们可以使用同一个客户端实例。
try:
    client = OpenAI(
        api_key=AOAI_API_KEY,
        base_url="https://api.huaqloud.com/v1"
    )
    print("OpenAI 客户端初始化成功。")
except Exception as e:
    print(f"初始化 OpenAI 客户端时出错: {e}")
    # 如果客户端初始化失败，后续依赖客户端的代码将无法执行，
    # 可以选择在这里抛出异常或让程序在后续调用时失败。
    # 为了简单起见，如果初始化失败，后续代码会因为 client 未定义或为 None 而出错。
    # 更健壮的做法是确保 client 初始化成功，否则退出或进行错误处理。
    # 此处假设环境变量已正确设置且 API 可达，以便专注于 RAG 逻辑。
    # 如果要让程序在此处停止，可以取消下一行的注释：
    # raise ConnectionError(f"无法初始化 OpenAI 客户端: {e}")


print("\n\n--- 场景八：实现简单的检索增强生成 (RAG) ---")

# --- 1. 准备一个简单的知识库 (几段文本) ---
knowledge_base_texts = [
    "Luminaris 水晶是一种罕见的矿石，据说蕴含着纯净的星辰能量，夜晚会发出柔和的蓝光。",
    "这种水晶主要被发现在极北之地的冰封洞穴深处，或是高耸入云的山脉之心。",
    "古代文献记载，Luminaris 水晶的能量可以被用来驱动精密的魔法装置，也可用于制作具有治疗效果的护身符。",
    "采集 Luminaris 水晶极其危险，不仅因为环境恶劣，更因为守护水晶的元素生物会攻击入侵者。"
]
print(f"知识库包含 {len(knowledge_base_texts)} 条信息。")

# --- 定义余弦相似度计算函数 ---
def cosine_similarity(v1, v2):
    vec1 = np.array(v1)
    vec2 = np.array(v2)
    dot_product = np.dot(vec1, vec2)
    norm_v1 = np.linalg.norm(vec1)
    norm_v2 = np.linalg.norm(vec2)
    if norm_v1 == 0 or norm_v2 == 0:
        return 0.0
    else:
        similarity = dot_product / (norm_v1 * norm_v2)
        return np.clip(similarity, -1.0, 1.0)

# --- 辅助函数：获取嵌入向量 ---
def get_embeddings(texts, current_client, model="text-embedding-ada-002"): # Renamed client to current_client to avoid conflict if client is global
    """使用指定客户端和模型获取文本列表的嵌入向量"""
    try:
        response = current_client.embeddings.create(model=model, input=texts)
        embeddings = [item.embedding for item in sorted(response.data, key=lambda x: x.index)]
        return embeddings
    except APIError as e:
        print(f"获取嵌入向量时出错 (API Error): {e.status_code} - {e.message}")
    except Exception as e:
        print(f"获取嵌入向量时发生意外错误: {e}")
    return None

# --- RAG 核心逻辑 ---
def perform_rag(user_query):
    print(f"\n用户查询: {user_query}")

    # --- 步骤 1 & 2: 获取知识库和查询的嵌入向量 ---
    # 使用统一的 client 实例
    print("--- 正在获取知识库嵌入向量 (使用 text-embedding-ada-002)... ---")
    kb_embeddings = get_embeddings(knowledge_base_texts, client) # 使用合并后的 client
    if not kb_embeddings:
        print("无法获取知识库嵌入，RAG 中止。")
        return

    print("--- 正在获取查询嵌入向量 (使用 text-embedding-ada-002)... ---")
    query_embedding = get_embeddings([user_query], client) # 使用合并后的 client
    if not query_embedding:
        print("无法获取查询嵌入，RAG 中止。")
        return
    query_embedding = query_embedding[0]

    # --- 步骤 3: 检索最相关的知识片段 ---
    print("--- 正在计算相似度并检索相关信息... ---")
    similarities = [cosine_similarity(query_embedding, kb_emb) for kb_emb in kb_embeddings]

    most_similar_index = np.argmax(similarities)
    retrieved_context_text = knowledge_base_texts[most_similar_index]
    max_similarity = similarities[most_similar_index]

    print(f"--- 检索到的最相关信息 (索引 {most_similar_index}, 相似度: {max_similarity:.4f}): ---")
    print(retrieved_context_text)

    similarity_threshold = 0.5
    if max_similarity < similarity_threshold:
        print(f"--- 相似度 ({max_similarity:.4f}) 低于阈值 ({similarity_threshold})，可能没有找到强相关信息。将不使用 RAG，直接提问。---")
        retrieved_context_text = None

    # --- 步骤 4: 构建增强提示 ---
    system_prompt = "你是一个乐于助人的AI助手。请根据下面提供的背景知识来回答用户的问题。如果背景知识与问题无关或不足以回答，请说明你依赖了背景知识但无法完全回答，或者直接回答（如果问题很通用）。"

    messages = [
        {"role": "system", "content": system_prompt}
    ]

    if retrieved_context_text:
        augmented_prompt = f"背景知识：\n\"\"\"\n{retrieved_context_text}\n\"\"\"\n\n用户问题：{user_query}"
        messages.append({"role": "user", "content": augmented_prompt})
    else:
        messages.append({"role": "user", "content": user_query})

    print("\n--- 准备发送给 Chat API 的最终消息 (已增强): ---")
    print(messages[-1]['content'])

    # --- 步骤 5: 调用 Chat API 生成最终答案 ---
    # 使用统一的 client 实例
    print("\n--- 正在调用 Chat API 生成答案... ---")
    try:
        chat_response = client.chat.completions.create( # 使用合并后的 client
            model="gpt-4o-mini",
            messages=messages,
            temperature=0.3
        )
        final_answer = chat_response.choices[0].message.content
        print("\n--- AI 生成的最终答案: ---")
        print(final_answer)

    except APIError as e:
        print(f"调用 Chat API 时出错 (API Error): {e.status_code} - {e.message}")
    except Exception as e:
        print(f"调用 Chat API 时发生意外错误: {e}")


# --- 执行 RAG 流程 ---
if __name__ == "__main__": # 确保在直接运行时执行
    # 检查客户端是否成功初始化
    if 'client' not in globals() or client is None:
        print("OpenAI 客户端未成功初始化。请检查 API Key 和网络连接。RAG 流程中止。")
    else:
        try:
            question = "Luminaris 水晶有什么用途？"
            perform_rag(question)

            print("\n" + "="*50 + "\n")

            question_2 = "Luminaris 水晶晚上会怎样？"
            perform_rag(question_2)

        except AuthenticationError as e:
            # 修正了异常信息中的重复变量名
            print(f"\n认证错误: API Key 可能无效或权限不足。请检查 AOAI_API_KEY。 {e}")
        except Exception as e:
            print(f"\n发生意外错误: {type(e).__name__} - {e}")
            traceback.print_exc()
```

## 代码解释

1. API Key 与客户端：获取 `AOAI_API_KEY`，并分别初始化两个 `OpenAI` 客户端（`client_sf` 用于 Embeddings，`client_ds` 用于 Chat）。

1. 知识库：定义了一个包含 4 段关于虚构 "Luminaris 水晶" 文本的列表 `knowledge_base_texts`。

1. `cosine_similarity()` 函数：复用了之前的余弦相似度计算函数。

1. `get_embeddings()` 函数：封装了调用 Embeddings API 的逻辑，接收文本列表和对应的客户端，返回向量列表，并包含基本的错误处理。

1. `perform_rag()` 函数（核心逻辑）：
   + 接收 `user_query`。
   + 步骤 1 & 2：调用 `get_embeddings()` 分别获取知识库（`kb_embeddings`）和用户查询（`query_embedding`）的向量。增加了对获取嵌入失败的处理。
   + 步骤 3：计算查询向量与所有知识库向量的余弦相似度。使用 `np.argmax()` 找到相似度最高的知识片段的索引 `most_similar_index`，并获取对应的文本 `retrieved_context_text`。打印检索到的内容和相似度分数。增加了可选的相似度阈值判断。
   + 步骤 4：构建发送给 AOAI Chat 的 `messages` 列表。
   + 包含一个 `system_prompt` 指导模型如何使用背景知识。
   + 如果检索到了相关且足够相似的上下文（`retrieved_context_text` 非空且相似度高于阈值），则构建一个包含“背景知识”和“用户问题”的 `user` 消息。
   + 否则，只包含原始的 `user` 消息。
   + 打印最终构建的用户消息内容，以便观察。
   + 步骤 5：调用 `client_ds.chat.completions.create()`，传入构建好的 `messages`。设置较低的 `temperature`（如 `0.3`）鼓励模型更忠实于提供的上下文。打印模型生成的最终答案。包含了对 Chat API 调用的错误处理。

1. 执行：调用 `perform_rag()` 函数两次，分别测试不同的问题，观察 RAG 流程是否能找到正确的上下文并生成相应的答案。增加了全局的 `try...except` 块处理认证错误和其他意外。

## RAG 的价值与应用

+ 提高准确性：基于检索到的事实回答，减少捏造信息的可能。

+ 知识时效性：可以接入实时更新的数据库或文档库。

+ 私有数据利用：让模型能回答关于特定领域或内部文档的问题。

+ 可解释性：可以追溯答案来源于哪些检索到的信息。

+ 常见应用：企业智能问答机器人、基于文档的摘要生成、客服助手、个性化内容推荐等。

## 简化与局限

这个示例非常基础，实际的 RAG 系统通常更复杂，涉及：

+ 文档预处理与分块（Chunking）：将长文档切分成合适大小、包含连贯信息的块。
