from unsloth import FastLanguageModel
import torch
```

```
max_seq_length = 2048
dtype = None
load_in_4bit = True
```

```

    model_name = "./Qwen3-32B-unsloth-bnb-4bit",
    max_seq_length = max_seq_length,
    dtype = dtype,
    load_in_4bit = load_in_4bit,
)
