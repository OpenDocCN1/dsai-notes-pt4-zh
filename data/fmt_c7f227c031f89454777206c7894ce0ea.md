
### 多因素认证 (MFA)

```
import pyotp
import qrcode

class MFAManager:
    def __init__(self, user_id):
        self.user_id = user_id
        # 生成密钥
        self.secret = pyotp.random_base32()
    
    def generate_qr_code(self, issuer="OpenClaw"):
        """生成 QR 码供用户扫描"""
        totp = pyotp.TOTP(self.secret)
        uri = totp.provisioning_uri(
            name=self.user_id,
            issuer_name=issuer
        )
        
        qr = qrcode.QRCode(version=1, box_size=10, border=5)
        qr.add_data(uri)
        qr.make(fit=True)
        
        img = qr.make_image(fill_color="black", back_color="white")
        img.save(f"mfa_{self.user_id}.png")
        
        return self.secret
    
    def verify_token(self, token):
        """验证用户输入的 TOTP 令牌"""
        totp = pyotp.TOTP(self.secret)
        return totp.verify(token, valid_window=1)

# 使用示例
mfa = MFAManager("user@example.com")
secret = mfa.generate_qr_code()
print(f"密钥: {secret}")

# 验证用户输入的令牌
user_token = input("输入 6 位验证码: ")
if mfa.verify_token(user_token):
    print("✅ 认证成功")
else:
    print("❌ 认证失败")
```

### API 密钥认证

```
import secrets
import hashlib
import time

class APIKeyAuth:
    def __init__(self):
        self.keys = {}  # 存储哈希后的密钥
    
    def generate_key(self, user_id, permissions):
        """生成新的 API 密钥"""
        # 生成随机密钥
        key = f"sk-{secrets.token_urlsafe(32)}"
        
        # 哈希存储
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        
        self.keys[key_hash] = {
            "user_id": user_id,
            "permissions": permissions,
            "created_at": time.time(),
            "last_used": None
        }
        
        # 只返回一次密钥
        return key
    
    def validate_key(self, key):
        """验证 API 密钥"""
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        
        if key_hash not in self.keys:
            return None
        
        # 更新最后使用时间
        self.keys[key_hash]["last_used"] = time.time()
        
        return self.keys[key_hash]
    
    def revoke_key(self, key):
        """撤销 API 密钥"""
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        if key_hash in self.keys:
            del self.keys[key_hash]
            return True
        return False

# 使用示例
auth = APIKeyAuth()
api_key = auth.generate_key("user123", ["read", "write"])
print(f"API 密钥: {api_key}")

# 验证密钥
key_info = auth.validate_key(api_key)
if key_info:
    print(f"用户: {key_info['user_id']}")
    print(f"权限: {key_info['permissions']}")
```

### OAuth 2.0 集成

对于企业部署，集成 OAuth 2.0:

```
from authlib.integrations.flask_client import OAuth
from flask import Flask, redirect, url_for, session

app = Flask(__name__)
app.secret_key = 'your-secret-key'

oauth = OAuth(app)

# 配置 OAuth 提供商 (如 Google)
oauth.register(
    name='google',
    client_id='your-client-id',
    client_secret='your-client-secret',
    server_metadata_url='https://accounts.google.com/.well-known/openid-configuration',
    client_kwargs={'scope': 'openid email profile'}
)

@app.route('/login')
def login():
    redirect_uri = url_for('authorize', _external=True)
    return oauth.google.authorize_redirect(redirect_uri)

@app.route('/authorize')
def authorize():
    token = oauth.google.authorize_access_token()
    user_info = oauth.google.parse_id_token(token)
    session['user'] = user_info
    return redirect('/dashboard')

@app.route('/logout')
def logout():
    session.pop('user', None)
    return redirect('/')
```

### 13.3.2 权限分级设计

实施最小权限原则，为不同用户和 Agent 分配适当的权限。

#### 基于角色的访问控制 (RBAC)

```
from enum import Enum
from typing import Set, Dict

class Permission(Enum):
    READ_DATA = "read_data"
    WRITE_DATA = "write_data"
    DELETE_DATA = "delete_data"
    MANAGE_USERS = "manage_users"
    MANAGE_AGENTS = "manage_agents"
    EXECUTE_SKILLS = "execute_skills"
    VIEW_LOGS = "view_logs"
    MANAGE_API_KEYS = "manage_api_keys"

class Role(Enum):
    VIEWER = "viewer"
    USER = "user"
    ADMIN = "admin"
    SUPER_ADMIN = "super_admin"

class RBACManager:
    def __init__(self):
        # 定义角色权限映射
        self.role_permissions: Dict[Role, Set[Permission]] = {
            Role.VIEWER: {
                Permission.READ_DATA,
                Permission.VIEW_LOGS
            },
            Role.USER: {
                Permission.READ_DATA,
                Permission.WRITE_DATA,
                Permission.EXECUTE_SKILLS,
                Permission.VIEW_LOGS
            },
            Role.ADMIN: {
                Permission.READ_DATA,
                Permission.WRITE_DATA,
                Permission.DELETE_DATA,
                Permission.MANAGE_USERS,
                Permission.MANAGE_AGENTS,
                Permission.EXECUTE_SKILLS,
                Permission.VIEW_LOGS
            },
            Role.SUPER_ADMIN: set(Permission)  # 所有权限
        }
        
        self.user_roles: Dict[str, Role] = {}
    
    def assign_role(self, user_id: str, role: Role):
        """为用户分配角色"""
        self.user_roles[user_id] = role
    
    def check_permission(self, user_id: str, permission: Permission) -> bool:
        """检查用户是否有特定权限"""
        role = self.user_roles.get(user_id)
        if not role:
            return False
        
        return permission in self.role_permissions[role]
    
    def get_user_permissions(self, user_id: str) -> Set[Permission]:
        """获取用户的所有权限"""
        role = self.user_roles.get(user_id)
        if not role:
            return set()
        
        return self.role_permissions[role]

# 使用示例
rbac = RBACManager()

# 分配角色
rbac.assign_role("alice", Role.ADMIN)
rbac.assign_role("bob", Role.USER)
rbac.assign_role("charlie", Role.VIEWER)

# 检查权限
if rbac.check_permission("alice", Permission.MANAGE_USERS):
    print("Alice 可以管理用户")

if not rbac.check_permission("bob", Permission.DELETE_DATA):
    print("Bob 不能删除数据")

# 获取用户权限
print(f"Charlie 的权限: {rbac.get_user_permissions('charlie')}")
```
