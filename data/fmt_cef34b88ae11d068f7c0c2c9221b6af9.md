```
def estimate_tokens(text: str) -> int:
    """估算文本的 token 数"""
    chinese chars = len(re.findall(r'[u4e00-\u9fff)', text))
    other chars = len(text) - chinese chars
    return chinese chars + int(other chars * 0.25)
```
