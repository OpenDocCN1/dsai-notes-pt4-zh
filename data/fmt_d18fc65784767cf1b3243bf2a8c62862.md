    def generate_security_report(self) -> Dict:
        """生成安全报告"""
        total_items = 0
        completed_items = 0
        critical_pending = []
        high_pending = []
        
        for category, data in self.security_checklist.items():
            for item in data['items']:
                total_items += 1
                if item['status']:
                    completed_items += 1
                else:
                    if item['priority'] == 'critical':
                        critical_pending.append(f"[{category}] {item['check']}")
                    elif item['priority'] == 'high':
                        high_pending.append(f"[{category}] {item['check']}")
        
        completion_rate = (completed_items / total_items * 100) if total_items > 0 else 0
        
        return {
            'completion_rate': completion_rate,
            'total_items': total_items,
            'completed': completed_items,
            'pending': total_items - completed_items,
            'critical_pending': critical_pending,
            'high_pending': high_pending,
            'security_score': self._calculate_security_score(completion_rate, critical_pending, high_pending)
        }
    
    def _calculate_security_score(
        self,
        completion_rate: float,
        critical_pending: List[str],
        high_pending: List[str]
    ) -> str:
        """计算安全评分"""
        if len(critical_pending) > 0:
            return 'F - 严重风险'
        elif completion_rate < 50:
            return 'D - 高风险'
        elif completion_rate < 70:
            return 'C - 中等风险'
        elif completion_rate < 90:
            return 'B - 低风险'
        else:
            return 'A - 安全'
