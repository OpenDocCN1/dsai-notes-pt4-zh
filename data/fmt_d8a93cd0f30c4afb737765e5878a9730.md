sales_agent = Agent(
    name="销售智能体",
    model = "deepseek-chat")

def transfer_to_sales():
   return sales_agent

agent = Agent(
    functions=[transfer_to_sales],
    model = "deepseek-chat")

response = swarm_client.run(agent, [{"role":"user", "content":"请转接到销售。"}])
