import re
import requests

class KeyLeakDetector:
    def __init__(self):
        # API 密钥模式
        self.patterns = {
            'openai': r'sk-[A-Za-z0-9]{48}',
            'anthropic': r'sk-ant-[A-Za-z0-9\-]{95}',
            'aws': r'AKIA[0-9A-Z]{16}',
            'github': r'ghp_[A-Za-z0-9]{36}',
            'google': r'AIza[0-9A-Za-z\-_]{35}'
        }
    
    def scan_text(self, text):
        """扫描文本中的 API 密钥"""
        findings = []
        
        for service, pattern in self.patterns.items():
            matches = re.finditer(pattern, text)
            for match in matches:
                findings.append({
                    'service': service,
                    'key': match.group(),
                    'position': match.span()
                })
        
        return findings
    
    def scan_file(self, filepath):
        """扫描文件中的 API 密钥"""
        try:
            with open(filepath, 'r', encoding='utf-8') as f:
                content = f.read()
            return self.scan_text(content)
        except Exception as e:
            return []
    
    def scan_directory(self, directory):
        """扫描目录中的所有文件"""
        import os
        
        all_findings = {}
        
        for root, dirs, files in os.walk(directory):
            # 跳过 .git 目录
            if '.git' in dirs:
                dirs.remove('.git')
            
            for file in files:
                # 只扫描文本文件
                if file.endswith(('.py', '.js', '.env', '.yaml', '.json', '.md')):
                    filepath = os.path.join(root, file)
                    findings = self.scan_file(filepath)
                    
                    if findings:
                        all_findings[filepath] = findings
        
        return all_findings
    
    def check_github_exposure(self, github_token, repo_owner, repo_name):
        """检查 GitHub 仓库中的密钥泄露"""
        # 使用 GitHub API 搜索
        url = f"https://api.github.com/search/code"
        
        findings = []
        
        for service, pattern in self.patterns.items():
            params = {
                'q': f'{pattern} repo:{repo_owner}/{repo_name}',
                'per_page': 100
            }
            
            headers = {
                'Authorization': f'token {github_token}',
                'Accept': 'application/vnd.github.v3+json'
            }
            
            try:
                response = requests.get(url, params=params, headers=headers)
                if response.status_code == 200:
                    results = response.json()
                    if results['total_count'] > 0:
                        findings.append({
                            'service': service,
                            'count': results['total_count'],
                            'files': [item['path'] for item in results['items']]
                        })
            except Exception as e:
                print(f"检查 {service} 时出错: {e}")
        
        return findings

# 使用示例
detector = KeyLeakDetector()

# 扫描文本
text = "My OpenAI key is sk-abc123... and AWS key is AKIAIOSFODNN7EXAMPLE"
findings = detector.scan_text(text)
if findings:
    print("⚠️ 检测到泄露的 API 密钥:")
    for finding in findings:
        print(f"  - {finding['service']}: {finding['key'][:10]}...")

# 扫描目录
findings = detector.scan_directory('~/.openclaw')
if findings:
    print("⚠️ 在以下文件中检测到 API 密钥:")
    for filepath, keys in findings.items():
        print(f"  - {filepath}: {len(keys)} 个密钥")
