roadmap = ProductFeatureRoadmap("AI 写作助手")
feature_plan = roadmap.define_feature_roadmap()

print("产品功能路线图:")
for quarter, data in feature_plan.items():
    print(f"\n{quarter}: {data['theme']}")
    for feature in data['features']:
        print(f"  - {feature['name']} ({feature['effort']} effort)")
        print(f"    {feature['description']}")
