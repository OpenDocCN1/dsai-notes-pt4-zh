from datetime import datetime, timedelta
from typing import List, Dict
from dataclasses import dataclass
from enum import Enum

class SprintStatus(Enum):
    &quot;&quot;&quot;Sprint 状态&quot;&quot;&quot;
    PLANNED = &quot;计划中&quot;
    IN_PROGRESS = &quot;进行中&quot;
    COMPLETED = &quot;已完成&quot;
    BLOCKED = &quot;阻塞&quot;

@dataclass
class Task:
    &quot;&quot;&quot;任务&quot;&quot;&quot;
    id: str
    title: str
    description: str
    assignee: str
    story_points: int
    status: str
    dependencies: List[str]
    acceptance_criteria: List[str]

@dataclass
class Sprint:
    &quot;&quot;&quot;Sprint&quot;&quot;&quot;
    number: int
    goal: str
    start_date: datetime
    end_date: datetime
    tasks: List[Task]
    status: SprintStatus
    velocity: int = 0

class AgileProjectPlan:
    &quot;&quot;&quot;敏捷项目计划&quot;&quot;&quot;
    
    def __init__(self, project_name: str, team_size: int):
        self.project_name = project_name
        self.team_size = team_size
        self.sprints = []
        self.backlog = []
        self.team_velocity = 0
    
    def create_product_backlog(self, requirements: List[Dict]) -&gt; None:
        &quot;&quot;&quot;创建产品待办列表&quot;&quot;&quot;
        for req in requirements:
            task = Task(
                id=req[&#x27;id&#x27;],
                title=req[&#x27;title&#x27;],
                description=req[&#x27;description&#x27;],
                assignee=&quot;&quot;,
                story_points=self._estimate_story_points(req),
                status=&quot;backlog&quot;,
                dependencies=req.get(&#x27;dependencies&#x27;, []),
                acceptance_criteria=req.get(&#x27;acceptance_criteria&#x27;, [])
            )
            self.backlog.append(task)
        
        # 按优先级排序
        self.backlog.sort(key=lambda x: x.story_points, reverse=True)
    
    def _estimate_story_points(self, requirement: Dict) -&gt; int:
        &quot;&quot;&quot;估算故事点&quot;&quot;&quot;
        complexity_map = {
            &#x27;low&#x27;: 2,
            &#x27;medium&#x27;: 5,
            &#x27;high&#x27;: 8
        }
        return complexity_map.get(requirement.get(&#x27;complexity&#x27;, &#x27;medium&#x27;), 5)
    
    def plan_sprint(
        self,
        sprint_number: int,
        sprint_goal: str,
        duration_weeks: int = 2
    ) -&gt; Sprint:
        &quot;&quot;&quot;规划 Sprint&quot;&quot;&quot;
        start_date = datetime.now() + timedelta(weeks=(sprint_number - 1) * duration_weeks)
        end_date = start_date + timedelta(weeks=duration_weeks)
        
        # 根据团队速度选择任务
        if self.team_velocity == 0:
            # 首个 Sprint,保守估计
            capacity = self.team_size * duration_weeks * 5  # 每人每周 5 点
        else:
            capacity = self.team_velocity
        
        sprint_tasks = []
        total_points = 0
        
        for task in self.backlog:
            if total_points + task.story_points &lt;= capacity:
                sprint_tasks.append(task)
                total_points += task.story_points
                task.status = &quot;sprint&quot;
        
        # 从待办列表移除
        self.backlog = [t for t in self.backlog if t not in sprint_tasks]
        
        sprint = Sprint(
            number=sprint_number,
            goal=sprint_goal,
            start_date=start_date,
            end_date=end_date,
            tasks=sprint_tasks,
            status=SprintStatus.PLANNED,
            velocity=total_points
        )
        
        self.sprints.append(sprint)
        return sprint
    
    def generate_project_timeline(self) -&gt; Dict:
        &quot;&quot;&quot;生成项目时间线&quot;&quot;&quot;
        timeline = {
            &#x27;project&#x27;: self.project_name,
            &#x27;start_date&#x27;: self.sprints[0].start_date if self.sprints else None,
            &#x27;end_date&#x27;: self.sprints[-1].end_date if self.sprints else None,
            &#x27;total_sprints&#x27;: len(self.sprints),
            &#x27;sprints&#x27;: []
        }
        
        for sprint in self.sprints:
            timeline[&#x27;sprints&#x27;].append({
                &#x27;number&#x27;: sprint.number,
                &#x27;goal&#x27;: sprint.goal,
                &#x27;start&#x27;: sprint.start_date.strftime(&#x27;%Y-%m-%d&#x27;),
                &#x27;end&#x27;: sprint.end_date.strftime(&#x27;%Y-%m-%d&#x27;),
                &#x27;tasks&#x27;: len(sprint.tasks),
                &#x27;story_points&#x27;: sprint.velocity
            })
        
        return timeline
    
    def generate_gantt_chart_data(self) -&gt; List[Dict]:
        &quot;&quot;&quot;生成甘特图数据&quot;&quot;&quot;
        gantt_data = []
        
        for sprint in self.sprints:
            for task in sprint.tasks:
                gantt_data.append({
                    &#x27;task&#x27;: task.title,
                    &#x27;start&#x27;: sprint.start_date.strftime(&#x27;%Y-%m-%d&#x27;),
                    &#x27;end&#x27;: sprint.end_date.strftime(&#x27;%Y-%m-%d&#x27;),
                    &#x27;assignee&#x27;: task.assignee,
                    &#x27;status&#x27;: task.status,
                    &#x27;sprint&#x27;: sprint.number
                })
        
        return gantt_data

# 使用示例
project = AgileProjectPlan(&quot;AI 写作助手&quot;, team_size=5)

# 创建产品待办列表
requirements = [
    {
        &#x27;id&#x27;: &#x27;US-001&#x27;,
        &#x27;title&#x27;: &#x27;用户可以输入主题生成大纲&#x27;,
        &#x27;description&#x27;: &#x27;作为写作者,我希望输入主题后自动生成文章大纲&#x27;,
        &#x27;complexity&#x27;: &#x27;medium&#x27;,
        &#x27;dependencies&#x27;: [],
        &#x27;acceptance_criteria&#x27;: [
            &#x27;输入主题后 10 秒内生成大纲&#x27;,
            &#x27;大纲包含 3-5 个章节&#x27;
        ]
    },
    {
        &#x27;id&#x27;: &#x27;US-002&#x27;,
        &#x27;title&#x27;: &#x27;用户可以编辑和调整大纲&#x27;,
        &#x27;description&#x27;: &#x27;作为写作者,我希望可以编辑 AI 生成的大纲&#x27;,
        &#x27;complexity&#x27;: &#x27;low&#x27;,
        &#x27;dependencies&#x27;: [&#x27;US-001&#x27;]

    {
        &#x27;id&#x27;: &#x27;US-003&#x27;,
        &#x27;title&#x27;: &#x27;基于大纲生成段落内容&#x27;,
        &#x27;description&#x27;: &#x27;作为写作者,我希望基于大纲自动生成段落&#x27;,
        &#x27;complexity&#x27;: &#x27;high&#x27;,
        &#x27;dependencies&#x27;: [&#x27;US-001&#x27;]
    }

project.create_product_backlog(requirements)

# 规划 Sprint
sprint1 = project.plan_sprint(1, &quot;实现核心大纲生成功能&quot;)

# 生成时间线
timeline = project.generate_project_timeline()
print(f&quot;项目时间线:&quot;)
print(f&quot;开始日期: {timeline[&#x27;start_date&#x27;].strftime(&#x27;%Y-%m-%d&#x27;)}&quot;)
print(f&quot;结束日期: {timeline[&#x27;end_date&#x27;].strftime(&#x27;%Y-%m-%d&#x27;)}&quot;)
print(f&quot;总 Sprint 数: {timeline[&#x27;total_sprints&#x27;]}&quot;)
