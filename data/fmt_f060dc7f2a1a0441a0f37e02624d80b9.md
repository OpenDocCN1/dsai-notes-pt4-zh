from typing import List, Dict
from dataclasses import dataclass
from enum import Enum

class Role(Enum):
    """团队角色"""
    PRODUCT_OWNER = "产品负责人"
    SCRUM_MASTER = "Scrum Master"
    AI_ENGINEER = "AI 工程师"
    BACKEND_DEV = "后端开发"
    FRONTEND_DEV = "前端开发"
    DEVOPS = "DevOps 工程师"
    QA = "测试工程师"
    UX_DESIGNER = "UX 设计师"
    DATA_SCIENTIST = "数据科学家"

@dataclass
class TeamMember:
    """团队成员"""
    name: str
    role: Role
    skills: List[str]
    availability: float  # 0-1, 1 表示全职
    cost_per_month: float

class ResourcePlanning:
    """资源规划"""
    
    def __init__(self, project_name: str, budget: float):
        self.project_name = project_name
        self.budget = budget
        self.team = []
        self.tools = []
        self.infrastructure = {}
    
    def build_team(self, project_phase: str) -> List[TeamMember]:
        """组建团队"""
        team_templates = {
            'mvp': [
                TeamMember("", Role.PRODUCT_OWNER, ["需求分析", "产品设计"], 0.5, 8000),
                TeamMember("", Role.AI_ENGINEER, ["LLM", "OpenClaw", "Python"], 1.0, 12000),
                TeamMember("", Role.FRONTEND_DEV, ["React", "TypeScript"], 1.0, 10000),
            ],
            'production': [
                TeamMember("", Role.PRODUCT_OWNER, ["需求分析", "产品设计"], 1.0, 15000),
                TeamMember("", Role.SCRUM_MASTER, ["敏捷", "项目管理"], 0.5, 10000),
                TeamMember("", Role.AI_ENGINEER, ["LLM", "OpenClaw", "Python"], 2.0, 24000),
                TeamMember("", Role.BACKEND_DEV, ["Node.js", "PostgreSQL"], 1.0, 11000),
                TeamMember("", Role.FRONTEND_DEV, ["React", "TypeScript"], 1.0, 10000),
                TeamMember("", Role.DEVOPS, ["Docker", "Kubernetes", "AWS"], 0.5, 6000),
                TeamMember("", Role.QA, ["自动化测试", "性能测试"], 1.0, 9000),
                TeamMember("", Role.UX_DESIGNER, ["Figma", "用户研究"], 0.5, 5000),
            ]
        }
        
        self.team = team_templates.get(project_phase, team_templates['mvp'])
        return self.team
    
    def calculate_team_cost(self, duration_months: int) -> Dict:
        """计算团队成本"""
        total_cost = 0
        cost_breakdown = []
        
        for member in self.team:
            member_cost = member.cost_per_month * member.availability * duration_months
            total_cost += member_cost
            cost_breakdown.append({
                'role': member.role.value,
                'monthly_cost': member.cost_per_month * member.availability,
                'total_cost': member_cost
            })
        
        return {
            'total': total_cost,
            'breakdown': cost_breakdown,
            'average_monthly': total_cost / duration_months if duration_months > 0 else 0
        }
    
    def plan_tools_and_services(self) -> List[Dict]:
        """规划工具和服务"""
        tools = [
            {
                'category': 'AI/LLM',
                'items': [
                    {'name': 'OpenAI API', 'cost_monthly': 500, 'required': True},
                    {'name': 'Anthropic Claude', 'cost_monthly': 400, 'required': False},
                    {'name': 'LangChain', 'cost_monthly': 0, 'required': True},
                ]
            },
            {
                'category': '开发工具',
                'items': [
                    {'name': 'GitHub', 'cost_monthly': 50, 'required': True},
                    {'name': 'Cursor/Windsurf', 'cost_monthly': 40, 'required': True},
                    {'name': 'Figma', 'cost_monthly': 45, 'required': True},
                ]
            },
            {
                'category': '基础设施',
                'items': [
                    {'name': 'AWS/云服务', 'cost_monthly': 1000, 'required': True},
                    {'name': 'MongoDB Atlas', 'cost_monthly': 200, 'required': True},
                    {'name': 'Redis Cloud', 'cost_monthly': 100, 'required': True},
                ]
            },
            {
                'category': '监控和分析',
                'items': [
                    {'name': 'Datadog', 'cost_monthly': 300, 'required': True},
                    {'name': 'Sentry', 'cost_monthly': 80, 'required': True},
                    {'name': 'Mixpanel', 'cost_monthly': 200, 'required': False},
                ]
            }
        ]
        
        self.tools = tools
        return tools
    
    def calculate_total_budget(self, duration_months: int) -> Dict:
        """计算总预算"""
        team_cost = self.calculate_team_cost(duration_months)
        
        tools_cost = 0
        for category in self.tools:
            for item in category['items']:
                if item['required']:
                    tools_cost += item['cost_monthly']
        
        total_tools_cost = tools_cost * duration_months
        
        # 预留 20% 应急预算
        contingency = (team_cost['total'] + total_tools_cost) * 0.2
        
        total = team_cost['total'] + total_tools_cost + contingency
        
        return {
            'team_cost': team_cost['total'],
            'tools_cost': total_tools_cost,
            'contingency': contingency,
            'total': total,
            'monthly_burn_rate': total / duration_months,
            'budget_status': 'within_budget' if total <= self.budget else 'over_budget',
            'budget_utilization': (total / self.budget * 100) if self.budget > 0 else 0
        }
    
    def generate_resource_plan(self, duration_months: int) -> Dict:
        """生成资源计划"""
        self.build_team('production')
        self.plan_tools_and_services()
        budget = self.calculate_total_budget(duration_months)
        
        return {
            'project': self.project_name,
            'duration_months': duration_months,
            'team_size': len(self.team),
            'budget': budget,
            'team': [
                {
                    'role': m.role.value,
                    'availability': f"{m.availability * 100}%",
                    'skills': m.skills
                }
                for m in self.team
            ],
            'tools': self.tools
        }

# 使用示例
resource_plan = ResourcePlanning("AI 写作助手", budget=500000)
plan = resource_plan.generate_resource_plan(duration_months=6)

print(f"项目: {plan['project']}")
print(f"团队规模: {plan['team_size']} 人")
print(f"总预算: ${plan['budget']['total']:,.2f}")
print(f"预算状态: {plan['budget']['budget_status']}")
print(f"预算利用率: {plan['budget']['budget_utilization']:.1f}%")
