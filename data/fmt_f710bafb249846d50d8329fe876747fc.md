
```python
class AgentLoopController:
    def init(self, max_iterations=10, max_cost=0.1):
        self.max_iterations = max_iterations  # 最大迭代次数
        self.max_cost = max_cost  # 最大成本（美元）
        self.iteration_count = 0
        self.total_cost = 0.0  # 累计成本

    def can_continue(self, current_step_cost=0.0) -> bool:
        """判断是否可以继续循环"""
        self.iteration_count += 1
        self.total_cost += current_step_cost
        # 检查是否超过最大次数
        if self.iteration_count >= self.max_iterations:
            print(f"已达到最大迭代次数（{self.max_iterations}），终止循环")
            return False
        # 检查是否超过成本上限
        if self.total_cost >= self.max_cost:
            print(f"已超过成本上限（${self.max_cost}），当前花费：${self.total_cost:.4f}")
            return False
        return True
```

def simplified_autogpt(goal: str):
    # 初始化循环控制器（最多8步，成本上限$0.05）
    controller = AgentLoopController(max_iterations=8, max_cost=0.05)
    progress = "未开始"
    history = []  # 记录已完成步骤
    final_result = ""
    while True:
        # 检查是否继续循环
        if not controller.can_continue(current_step_cost=0.005):  # 假设每步成本$0.005
            final_result = f"任务未完成，已执行{controller.iteration_count}步，结果如下：{progress}"
            break
        # 1. 计划分解：确定下一步任务
        plan_chain = plan_prompt | llm
        plan_result = plan_chain.invoke({"goal": goal, "progress": progress, "history": history}).content
        # 解析任务和工具（实际项目需更严谨的解析）
        task = plan_result.split("任务：")[1].split("\n")[0]
        tool = plan_result.split("工具：")[1].split("\n")[0]
        # 2. 行动执行：完成当前任务
        action_result = execute_action(task, tool)
        history.append(f"步骤{len(history)+1}：{task} → {action_result[:30]}...")
        # 3. 反思评估：判断是否完成
        reflect_chain = reflect_prompt | llm
        reflect_result = reflect_chain.invoke({"goal": goal, "current_result": action_result}).content
        # 解析状态
        if "已完成" in reflect_result.split("状态：")[1].split("\n")[0]:
            final_result = reflect_result.split("反馈：")[1]
            break
        # 更新进度
        progress = action_result
    return final_result

# 测试：制定Python三天学习计划
