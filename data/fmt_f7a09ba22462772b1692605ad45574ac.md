        // 控制灵敏度参数 - 减小值使控制更平滑
        const PITCH_SENSITIVITY = 0.008; // 降低上下俯仰灵敏度
        const YAW_SENSITIVITY = 0.012;   // 降低左右转向灵敏度
        const SPEED_CHANGE_RATE = 0.1;   // 降低速度变化率
        const DAMPING = 0.98;            // 增加阻尼使动作更平滑

        // Animation loop
        function animate() {
            requestAnimationFrame(animate);

            // 处理输入，确保正确检测所有按键，使用更低的灵敏度
            // 上下控制
            if (keys['w'] || keys['arrowup']) {
                pitch += PITCH_SENSITIVITY;
            }
            if (keys['s'] || keys['arrowdown']) {
                pitch -= PITCH_SENSITIVITY;
            }
            // 左右控制
            if (keys['a'] || keys['arrowleft']) {
                yaw += YAW_SENSITIVITY;
            }
            if (keys['d'] || keys['arrowright']) {
                yaw -= YAW_SENSITIVITY;
            }
            
            // 速度控制
            if (keys[' ']) {
                speed = Math.min(speed + SPEED_CHANGE_RATE, 20);
                updateSpeedDisplay();
            }
            if (keys['shift']) {
                speed = Math.max(speed - SPEED_CHANGE_RATE, 1);
                updateSpeedDisplay();
            }

            // Apply damping - 增加阻尼使动作更平滑
            pitch *= DAMPING;
            yaw *= DAMPING;
            
            // Update airplane rotation
            airplane.rotation.x += pitch;
            airplane.rotation.y += yaw;
            
            // Banking effect - 减小侧倾效果
            roll = -yaw * 8; // 减少倾斜程度
            airplane.rotation.z = THREE.MathUtils.lerp(airplane.rotation.z, roll, 0.05); // 更平滑的过渡

            // Move airplane forward
            const direction = new THREE.Vector3(0, 0, -1);
            direction.applyQuaternion(airplane.quaternion);
            airplane.position.add(direction.multiplyScalar(speed * 0.5));

            // Keep airplane above ground
            if (airplane.position.y < 20) airplane.position.y = 20;

            // Rotate propeller
            propellerGroup.rotation.x += speed * 0.2;

            // Update camera to follow airplane
            const desiredCameraPosition = airplane.position.clone();
            const offset = cameraOffset.clone();
            offset.applyQuaternion(airplane.quaternion);
            desiredCameraPosition.add(offset);
            
            camera.position.lerp(desiredCameraPosition, 0.1);
            camera.lookAt(airplane.position);

            // 更新仪表盘
            updateDashboard();

            renderer.render(scene, camera);
        }

            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);

        animate();
    </script>
