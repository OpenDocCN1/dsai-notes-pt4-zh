# --- 初始化函数 ---
def initialize_clients():

    global client_ds, client_sf
    if DEEPSEEK_API_KEY:
        client_ds = OpenAI(api_key=DEEPSEEK_API_KEY, base_url=DEEPSEEK_BASE_URL)

    else:
        print("警告：未配置 DEEPSEEK_API_KEY，无法调用 DeepSeek 模型。")

    if SILICONFLOW_API_KEY:
        client_sf = OpenAI(api_key=SILICONFLOW_API_KEY, base_url=SILICONFLOW_BASE_URL)

    else:
        print("警告：未配置 SILICONFLOW_API_KEY，无法生成嵌入向量。")

def initialize_chromadb():
    """初始化 ChromaDB 客户端并加载或创建 Collection"""
    global collection
    if not pathlib.Path(CHROMA_DB_PATH).exists():
        print(f"ChromaDB 路径 '{CHROMA_DB_PATH}' 不存在，将自动创建。")
    # 使用新的 Settings 参数指定持久化目录
    client = chromadb.PersistentClient(path=CHROMA_DB_PATH)
    try:
        # 尝试获取 Collection，如果不存在会抛出异常
        collection = client.get_collection(name=COLLECTION_NAME)
        print(f"已加载现有的 ChromaDB Collection '{COLLECTION_NAME}'.")
    except Exception as e:
        print(f"Collection '{COLLECTION_NAME}' 未找到或加载时出错: {e}. 尝试创建...")
        try:
            collection = client.create_collection(
                name=COLLECTION_NAME,
                metadata={"hnsw:space": DISTANCE_METRIC} # 指定距离度量
            )
            print(f"已创建新的 ChromaDB Collection '{COLLECTION_NAME}'.")
        except Exception as create_e:
            print(f"致命错误：无法创建 Collection '{COLLECTION_NAME}'. 错误: {create_e}")
            collection = None # 保证失败时 collection 为 None

# --- 文件加载与分块函数 ---
def load_and_chunk_data(data_dir: pathlib.Path):
    """加载、解析、分块支持的文件"""
    all_chunks = []
    all_metadatas = []
    all_ids = []
    chunk_id_counter = 0

    if not data_dir.is_dir():
        print(f"错误: 数据目录 '{data_dir}' 不存在或不是一个目录。")
        return [], [], []

    print(f"开始扫描目录 '{data_dir}' 下的 {SUPPORTED_FILE_EXTENSIONS} 文件...")
    source_files_processed = set()

    for file_path in data_dir.rglob('*'):
        if file_path.is_file() and file_path.suffix.lower() in SUPPORTED_FILE_EXTENSIONS:
            print(f"  处理文件: {file_path.relative_to(data_dir)}")
            file_content = None
            try:
                if file_path.suffix.lower() in ['.txt', '.md']:
                    with open(file_path, 'r', encoding='utf-8') as f:
                        file_content = f.read()
                elif file_path.suffix.lower() == '.pdf':
                    doc_text = ""
                    with fitz.open(file_path) as pdf_doc:
                        for page_num, page in enumerate(pdf_doc):
                            doc_text += f"[Page {page_num + 1}]\n" + page.get_text("text") + "\n\n" # 添加页码信息
                    file_content = doc_text
                elif file_path.suffix.lower() == '.docx':
                    doc = docx.Document(file_path)
                    # 保留段落结构，过滤空段落
                    paragraphs = [para.text for para in doc.paragraphs if para.text.strip()]
                    file_content = "\n\n".join(paragraphs) # 使用双换行符连接段落

                if file_content is None or not file_content.strip():
                    print(f"  警告: 未能从 {file_path.name} 提取有效文本内容。")
                    continue

                source_files_processed.add(str(file_path.relative_to(data_dir)))

                # 使用双换行符作为主要的段落/逻辑分隔符进行切分
                potential_chunks = file_content.split('\n\n')
                current_chunk_text = ""

                for chunk_part in potential_chunks:
                    trimmed_part = chunk_part.strip()
                    if trimmed_part: # 如果部分非空
                        # 简单策略：如果当前块加上新部分不太长，就合并；否则形成新块
                        # (这里可以替换为更复杂的 chunking 逻辑)
                        if not current_chunk_text:
                           current_chunk_text = trimmed_part
                        elif len(current_chunk_text) + len(trimmed_part) < 800: # 示例：简单合并短段落
                           current_chunk_text += "\n" + trimmed_part
                        else:
                            # 保存之前的块 (如果满足长度)
                            if len(current_chunk_text) >= CHUNK_MIN_LENGTH:
                                metadata = {"source": str(file_path.relative_to(data_dir))}
                                chunk_unique_id = f"{file_path.stem}_{chunk_id_counter}"
                                all_chunks.append(current_chunk_text)
                                all_metadatas.append(metadata)
                                all_ids.append(chunk_unique_id)
                                chunk_id_counter += 1
                            # 开始新块
                            current_chunk_text = trimmed_part

                # 处理最后一个块
                if current_chunk_text and len(current_chunk_text) >= CHUNK_MIN_LENGTH:
                    metadata = {"source": str(file_path.relative_to(data_dir))}
                    chunk_unique_id = f"{file_path.stem}_{chunk_id_counter}"
                    all_chunks.append(current_chunk_text)
                    all_metadatas.append(metadata)
                    all_ids.append(chunk_unique_id)
                    chunk_id_counter += 1

            except Exception as e:
                print(f"  处理文件 {file_path.name} 时出错: {e}")
                traceback.print_exc() # 打印详细错误

    if not all_chunks:
        print(f"警告: 未能从目录 '{data_dir}' 加载任何有效的文本块。")
    else:
        print(f"成功加载 {len(all_chunks)} 个文本块，来源 {len(source_files_processed)} 个文件。")

    return all_chunks, all_metadatas, all_ids

# --- 嵌入函数 (使用硅基流动) ---
def get_embeddings_from_silibase(texts):
    """使用硅基流动 API 获取文本列表的嵌入向量"""
    global client_sf
    if not client_sf:
        print("错误: 硅基流动客户端未初始化。")
        return [None] * len(texts) # 返回对应数量的 None

    if isinstance(texts, str):
        texts = [texts]

    embeddings = []
    try:
        # 硅基流动 API 通常与 OpenAI 兼容
        response = client_sf.embeddings.create(model=EMBEDDING_MODEL, input=texts)
        # 确保按输入顺序返回嵌入
        embedding_map = {item.index: item.embedding for item in response.data}
        for i in range(len(texts)):
            embeddings.append(embedding_map.get(i)) # 如果某个索引缺失，会是 None
        print(f"  成功获取 {len(texts)} 个文本的嵌入向量。")

    except AuthenticationError as e:
         print(f"错误: 硅基流动 API Key 无效或权限不足。 {e}")
         return [None] * len(texts)
    except APIError as e:
        print(f"错误: 调用硅基流动 Embeddings API 失败: {e.status_code} - {e.message}")
        # 尝试解析错误体
        try:
            error_details = json.loads(e.response.content.decode('utf-8'))
            print(f"  错误详情: {error_details}")
        except: pass
        return [None] * len(texts)
    except Exception as e:
        print(f"错误: 获取嵌入向量时发生未知错误: {e}")
        traceback.print_exc()
        return [None] * len(texts)

    # 检查是否有 None (表示获取失败)
    if any(emb is None for emb in embeddings):
        print(f"警告: 未能获取所有 {len(texts)} 个文本的嵌入向量。")

    return embeddings


# --- 数据索引函数 ---
def index_knowledge_base(chunks, metadatas, ids):
    """获取嵌入并索引数据到 ChromaDB"""
    global collection
    if not collection:
        print("错误: ChromaDB Collection 未初始化，无法索引。")
        return False
    if not chunks:
        print("错误: 没有可供索引的文本块。")
        return False

    print(f"开始为 {len(chunks)} 个文本块生成嵌入向量 (使用硅基流动)...")
    embeddings = get_embeddings_from_silibase(chunks)

    # 严格检查：如果任何一个嵌入获取失败，则不进行索引
    valid_indices = [i for i, emb in enumerate(embeddings) if emb is not None and len(emb) == EMBEDDING_DIMENSION]

    if len(valid_indices) != len(chunks):
        failed_count = len(chunks) - len(valid_indices)
        print(f"错误: 获取嵌入向量失败 {failed_count} 次。索引操作中止。")
        print("请检查硅基流动 API Key、网络连接和模型名称。")
        return False
    else:
        print("所有嵌入向量获取成功。开始添加到 ChromaDB...")

    try:
        # 添加数据到 ChromaDB
        collection.add(
            embeddings=embeddings, # 使用获取到的完整列表
            documents=[chunks[i] for i in valid_indices], # 虽然现在 valid_indices == range(len(chunks))
            metadatas=[metadatas[i] for i in valid_indices],
            ids=[ids[i] for i in valid_indices]
        )
        print(f"成功将 {len(valid_indices)} 个条目索引到 Collection '{COLLECTION_NAME}'.")
        return True
    except Exception as e:
        print(f"错误: 添加数据到 ChromaDB 时出错: {e}")
        traceback.print_exc()
        return False

# --- RAG 核心回答函数 (Gradio 调用) ---
def get_rag_answer(query: str) -> str:
    """处理用户查询，执行 RAG 流程并返回答案"""
    global collection, client_ds
    if not query or not query.strip():
        return "请输入您的问题。"
    if collection is None:
        return "错误：知识库向量数据库未初始化。"
    if client_ds is None:
        return "错误：DeepSeek 语言模型客户端未初始化 (请检查 DEEPSEEK_API_KEY)。"
    if client_sf is None: # 检查硅基流动客户端
        return "错误：嵌入模型客户端未初始化 (请检查 SILICONFLOW_API_KEY)。"


    print(f"\n--- 收到查询: {query} ---")

    # 1. & 2. 查询嵌入
    print("  生成查询嵌入向量 (硅基流动)...")
    query_embedding_list = get_embeddings_from_silibase(query) # 单个查询也用列表
    if not query_embedding_list or query_embedding_list[0] is None:
        print("  错误: 未能生成查询嵌入。")
        return "抱歉，处理您的问题时遇到错误（无法生成查询嵌入）。"
    query_embedding = query_embedding_list[0]
    print("  查询嵌入向量已生成。")

    # 3. 检索
    print(f"  在 ChromaDB 中检索 Top {N_RESULTS_RETRIEVAL} 相关文档...")
    try:
        results = collection.query(
            query_embeddings=[query_embedding],
            n_results=N_RESULTS_RETRIEVAL,
            include=['documents', 'metadatas', 'distances'] # 明确请求包含距离
        )
        # print(f"  ChromaDB 原始查询结果: {results}") # 调试时取消注释
    except Exception as e:
        print(f"  错误: 查询 ChromaDB 时出错: {e}")
        traceback.print_exc()
        return "抱歉，在知识库中检索信息时遇到错误。"

    # 处理和过滤检索结果
    retrieved_chunks = []
    retrieved_sources = set() # 使用 set 自动去重
    context_str = "根据现有知识库，未找到与您问题直接相关的信息。" # 默认信息

    if results and results.get('documents') and results['documents'][0]:
        docs = results['documents'][0]
        metadatas = results['metadatas'][0] if results.get('metadatas') else [{}] * len(docs)
        distances = results['distances'][0] if results.get('distances') else [float('inf')] * len(docs)

        print(f"  初步检索到 {len(docs)} 个文档。应用距离阈值 (< {DISTANCE_THRESHOLD})...")

        for i in range(len(docs)):
            doc = docs[i]
            dist = distances[i]
            meta = metadatas[i]
            source = meta.get('source', '未知来源')

            print(f"    - 文档 {i}: 距离={dist:.4f}, 来源='{source}'") # 打印每个结果的距离
            if dist < DISTANCE_THRESHOLD:
                print(f"      -> 相关 (距离 < {DISTANCE_THRESHOLD})")
                retrieved_chunks.append(doc)
                retrieved_sources.add(source)
            else:
                print(f"      -> 不够相关 (距离 >= {DISTANCE_THRESHOLD})")

        if retrieved_chunks:
            context_str = "\n\n---\n\n".join(retrieved_chunks)
            print(f"  最终筛选出 {len(retrieved_chunks)} 个相关文本块。来源: {', '.join(sorted(list(retrieved_sources)))}")
            # print("  筛选后的上下文:\n" + context_str) # 调试时取消注释
        else:
            print("  警告: 没有文档满足距离阈值。")
            context_str = "虽然检索到一些信息，但它们与问题的相关性不够高。"
    else:
        print("  警告: ChromaDB 未返回任何文档。")

    # 4. 构建 Prompt
    prompt = f"""
你是一个专业的“易速鲜花”客服助手。请根据下面提供的“背景知识”来回答“用户问题”。
请主要依据背景知识作答。如果背景知识中没有足够的信息来回答问题，请明确说明你的回答并非完全基于提供的背景知识，或者说明无法从背景知识中找到答案。

背景知识:
---
{context_str}
---

用户问题: {query}

请给出清晰、准确、友好的回答:
"""
    print("  已构建发送给 DeepSeek 的 Prompt。")

    # 5. 调用 DeepSeek LLM 生成答案
    print("  调用 DeepSeek API 生成回答...")
    final_answer = "抱歉，我在思考时遇到了一些问题，暂时无法回答。" # 默认错误回复
    try:
        response = client_ds.chat.completions.create(
            model=DEEPSEEK_MODEL,
            messages=[{"role": "user", "content": prompt}],
            max_tokens=1000, # 适当增加 max_tokens
            temperature=0.3,  # 较低的温度使回答更基于事实
            stream=False # 课堂演示简化，不使用流式
        )
        if response.choices and response.choices[0].message:
            final_answer = response.choices[0].message.content.strip()
            print("  成功从 DeepSeek 获取回答。")
        else:
            print("  警告: DeepSeek API 返回的响应中没有有效的回答。")
            final_answer = "抱歉，模型响应异常，未能生成回答。"
        # print(f"  DeepSeek 原始响应: {response}") # 调试时取消注释

    except AuthenticationError as e:
         print(f"错误: DeepSeek API Key 无效或权限不足。 {e}")
         final_answer = "错误：DeepSeek API Key 配置无效。"
    except APIError as e:
        print(f"错误: 调用 DeepSeek API 失败: {e.status_code} - {e.message}")
        if e.status_code == 429:
            final_answer = "抱歉，请求过于频繁，请稍后再试。"
        else:
            final_answer = f"抱歉，与语言模型服务通信时出错 (代码: {e.status_code})。"
    except Exception as e:
        print(f"错误: 调用 DeepSeek API 时发生未知错误: {e}")
        traceback.print_exc()
        final_answer = "抱歉，处理您的请求时发生内部错误。"

    print(f"--- 回答已生成 ---")
    return final_answer

if __name__ == "__main__":
    print("--- 启动智能客服 RAG 系统 ---")

    # 1. 初始化 API 客户端
    initialize_clients()

    # 2. 初始化 ChromaDB
    initialize_chromadb()

    # 3. 检查是否需要索引数据
    should_index = False
    if collection: # 确保 collection 初始化成功
        try:
            current_count = collection.count()
            print(f"当前 Collection '{COLLECTION_NAME}' 中已有 {current_count} 个文档。")
            if current_count == 0:
                print("Collection 为空，需要进行数据索引。")
                should_index = True
            else:
                print("Collection 中已存在数据，跳过索引。")
        except Exception as e:
            print(f"错误: 检查 Collection 文档数量时出错: {e}")
            print("假设需要重新索引。")
            should_index = True
    else:
        print("致命错误: ChromaDB Collection 未能初始化。程序退出。")
        exit(1) # 退出程序

    if should_index:
        if client_sf is None: # 检查嵌入客户端是否可用
             print("致命错误：硅基流动客户端未初始化，无法进行索引。请设置 SILICONFLOW_API_KEY。")
             exit(1)

        print("开始加载和分块知识库文档...")
        chunks, metadatas, ids = load_and_chunk_data(DATA_DIR)
        if not chunks:
            print("致命错误: 未能从数据目录加载任何文本块。无法进行索引。程序退出。")
            exit(1)

        print("开始索引知识库 (这可能需要一些时间)...")
        indexing_success = index_knowledge_base(chunks, metadatas, ids)
        if not indexing_success:
            print("致命错误: 知识库索引失败。请检查之前的错误日志。程序退出。")
            exit(1)
        print("知识库索引完成。")

    # 4. 启动 Gradio 界面
    print("\n--- 启动 Gradio Web 界面 ---")
    if not client_ds:
         print("警告：DeepSeek 客户端未初始化，问答功能将不可用！")
    if not client_sf:
         print("警告：硅基流动客户端未初始化，无法处理新问题！ RAG 将失败。")

    interface = gr.Interface(
        fn=get_rag_answer,
        inputs=gr.Textbox(lines=3, placeholder="请输入您关于易速鲜花的问题...", label="您的问题"),
        outputs=gr.Textbox(label="易速鲜花智能客服 回答", lines=15),
        title="易速鲜花 - 智能客服助手 (RAG Demo)",
        description="输入您的问题，系统将尝试从本地知识库中查找信息并通过 DeepSeek AI 回答。\n知识库路径: {}\n嵌入模型: {}\n语言模型: {}".format(DATA_DIR.resolve(), EMBEDDING_MODEL, DEEPSEEK_MODEL),
        allow_flagging='never',
        examples=[ # 添加一些示例问题
            ["你们送货范围是哪些？"],
            ["鲜花收到后如何保养？"],
            ["如果对花不满意怎么办？"],
            ["Luminaris 水晶晚上会怎样？"] # 测试知识库外的知识
        ]
    )

    # 运行 Gradio 应用
    # share=True 会创建一个公共链接，方便临时分享，但要注意安全和 API Key 消耗
    interface.launch()

    print("Gradio 界面已关闭。程序结束。")
