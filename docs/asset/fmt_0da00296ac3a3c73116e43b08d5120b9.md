import pygame
import random
import math
import time
from collections import deque

# Initialize Pygame
pygame.init()

# Constants
WINDOW_WIDTH = 800
WINDOW_HEIGHT = 600
GRID_SIZE = 20
CELL_SIZE = WINDOW_WIDTH // GRID_SIZE

# Colors
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GREEN = (0, 255, 0)
RED = (255, 0, 0)
BLUE = (0, 0, 255)
YELLOW = (255, 255, 0)
PURPLE = (255, 0, 255)
CYAN = (0, 255, 255)
ORANGE = (255, 165, 0)

# Setup display
screen = pygame.display.set_mode((WINDOW_WIDTH, WINDOW_HEIGHT))
pygame.display.set_caption("Enhanced Visual Snake")
clock = pygame.time.Clock()

class Particle:
    def __init__(self, x, y, color, velocity, size=3):
        self.x = x
        self.y = y
        self.color = color
        self.velocity = velocity
        self.size = size
        self.lifetime = 60
        self.max_lifetime = 60
        
    def update(self):
        self.x += self.velocity[0]
        self.y += self.velocity[1]
        self.velocity = (self.velocity[0] * 0.95, self.velocity[1] * 0.95)
        self.lifetime -= 1
        
    def draw(self, surface):
        if self.lifetime > 0:
            alpha = int(255 * (self.lifetime / self.max_lifetime))
            size = int(self.size * (self.lifetime / self.max_lifetime))
            if size > 0:
                pygame.draw.circle(surface, (*self.color[:3], alpha), (int(self.x), int(self.y)), size)

class Trail:
    def __init__(self, x, y, color):
        self.x = x
        self.y = y
        self.color = color
        self.lifetime = 30
        self.max_lifetime = 30
        
    def update(self):
        self.lifetime -= 1
        
    def draw(self, surface):
        if self.lifetime > 0:
            alpha = int(150 * (self.lifetime / self.max_lifetime))
            size = int(CELL_SIZE * 0.8 * (self.lifetime / self.max_lifetime))
            if size > 0:
                glow_surf = pygame.Surface((size * 2, size * 2), pygame.SRCALPHA)
                pygame.draw.circle(glow_surf, (*self.color[:3], alpha), (size, size), size)
                surface.blit(glow_surf, (self.x * CELL_SIZE + CELL_SIZE // 2 - size, 
                                       self.y * CELL_SIZE + CELL_SIZE // 2 - size))

class Food:
    def __init__(self, x, y, food_type):
        self.x = x
        self.y = y
        self.type = food_type
        self.animation_offset = random.random() * math.pi * 2
        self.pulse = 0
        
        # Food types and their effects
        self.food_types = {
            'normal': {'color': RED, 'effect': None, 'duration': 0},
            'speed': {'color': YELLOW, 'effect': 'speed_boost', 'duration': 150},
            'reverse': {'color': PURPLE, 'effect': 'reverse_controls', 'duration': 200},
            'double': {'color': CYAN, 'effect': 'double_vision', 'duration': 250},
            'slow': {'color': BLUE, 'effect': 'time_slow', 'duration': 200}
        }
        
    def draw(self, surface, time_factor=1.0):
        food_data = self.food_types[self.type]
        color = food_data['color']
        
        # Animated pulsing effect
        self.pulse += 0.1 * time_factor
        size = CELL_SIZE // 2 + int(math.sin(self.pulse + self.animation_offset) * 3)
        
        # Draw glow effect
        for i in range(3):
            glow_size = size + (i * 4)
            glow_alpha = 50 - (i * 15)
            glow_surf = pygame.Surface((glow_size * 2, glow_size * 2), pygame.SRCALPHA)
            pygame.draw.circle(glow_surf, (*color[:3], glow_alpha), (glow_size, glow_size), glow_size)
            surface.blit(glow_surf, (self.x * CELL_SIZE + CELL_SIZE // 2 - glow_size,
                                   self.y * CELL_SIZE + CELL_SIZE // 2 - glow_size))
        
        # Draw main food
        pygame.draw.circle(surface, color, 
                         (self.x * CELL_SIZE + CELL_SIZE // 2, self.y * CELL_SIZE + CELL_SIZE // 2), 
                         size)

class Obstacle:
    def __init__(self, x, y, obstacle_type='wall'):
        self.x = x
        self.y = y
        self.type = obstacle_type
        self.animation_phase = random.random() * math.pi * 2
        self.rotation = 0
        
    def update(self, time_factor=1.0):
        if self.type == 'rotating':
            self.rotation += 0.05 * time_factor
        self.animation_phase += 0.03 * time_factor
        
    def draw(self, surface):
        center_x = self.x * CELL_SIZE + CELL_SIZE // 2
        center_y = self.y * CELL_SIZE + CELL_SIZE // 2
        
        if self.type == 'wall':
            # Pulsating wall
            pulse = math.sin(self.animation_phase) * 2
            size = CELL_SIZE // 2 + int(pulse)
            
            # Glow effect
            glow_surf = pygame.Surface((CELL_SIZE * 2, CELL_SIZE * 2), pygame.SRCALPHA)
            pygame.draw.rect(glow_surf, (255, 100, 100, 80), 
                           (CELL_SIZE // 2, CELL_SIZE // 2, CELL_SIZE, CELL_SIZE))
            surface.blit(glow_surf, (self.x * CELL_SIZE - CELL_SIZE // 2, 
                      self.y * CELL_SIZE - CELL_SIZE // 2))
            
            # Main obstacle
            pygame.draw.rect(surface, (200, 50, 50), 
                           (center_x - size, center_y - size, size * 2, size * 2))
                           
        elif self.type == 'rotating':
            # Rotating obstacle
            points = []
            for i in range(4):
                angle = self.rotation + (i * math.pi / 2)
                x = center_x + math.cos(angle) * CELL_SIZE // 2
                y = center_y + math.sin(angle) * CELL_SIZE // 2
                points.append((x, y))
            
            # Glow
            pygame.draw.circle(surface, (150, 50, 200, 100), (center_x, center_y), CELL_SIZE)
            
            # Shape
            pygame.draw.polygon(surface, (200, 100, 255), points)

class TerrainZone:
    def __init__(self, x, y, width, height, zone_type):
        self.rect = pygame.Rect(x * CELL_SIZE, y * CELL_SIZE, 
                               width * CELL_SIZE, height * CELL_SIZE)
        self.type = zone_type
        self.animation = 0
        
        self.zone_effects = {
            'slow': {'color': (100, 100, 200, 50), 'speed_mult': 0.5},
            'warp': {'color': (200, 100, 200, 50), 'speed_mult': 2.0},
            'friction': {'color': (200, 150, 100, 50), 'speed_mult': 0.7}
        }
        
    def update(self):
        self.animation += 0.02
        
    def draw(self, surface):
        effect = self.zone_effects[self.type]
        
        # Animated pattern
        pattern_surf = pygame.Surface((self.rect.width, self.rect.height), pygame.SRCALPHA)
        
        for x in range(0, self.rect.width, 20):
            for y in range(0, self.rect.height, 20):
                offset = math.sin(self.animation + x * 0.01 + y * 0.01) * 5
                alpha = int(effect['color'][3] + offset)
                pygame.draw.circle(pattern_surf, (*effect['color'][:3], min(alpha, 100)), 
                                 (x + 10, y + 10), 8)
        
        surface.blit(pattern_surf, self.rect.topleft)

class Snake:
    def __init__(self):
        self.body = [(GRID_SIZE // 2, GRID_SIZE // 2)]
        self.direction = (1, 0)
        self.grow_count = 0
        self.effects = {}
        self.decoy_positions = []
        self.segment_colors = []
        
    def move(self, reverse=False):
        head = self.body[0]
        direction = self.direction
        
        if reverse:
            direction = (-direction[0], -direction[1])
            
        new_head = (head[0] + direction[0], head[1] + direction[1])
        
        # Wrap around screen
        new_head = (new_head[0] % GRID_SIZE, new_head[1] % (WINDOW_HEIGHT // CELL_SIZE))
        
        self.body.insert(0, new_head)
        
        if self.grow_count > 0:
            self.grow_count -= 1
            self.segment_colors.insert(0, self.get_segment_color(len(self.body)))
        else:
            self.body.pop()
            if self.segment_colors:
                self.segment_colors.pop()
                
    def change_direction(self, new_direction):
        if (new_direction[0] * -1, new_direction[1] * -1) != self.direction:
            self.direction = new_direction
            
    def grow(self, amount=1):
        self.grow_count += amount
        
    def get_segment_color(self, index):
        # Biomechanical evolution based on length
        base_color = GREEN
        evolution_factor = min(index / 20, 1.0)
        
        # Shift from green to cyan as snake grows
        r = int(base_color[0] * (1 - evolution_factor))
        g = int(base_color[1])
        b = int(base_color[2] * evolution_factor + 255 * evolution_factor)
        
        return (r, g, min(b, 255))
        
    def draw(self, surface, time_factor=1.0):
        # Draw decoy if double vision is active
        if 'double_vision' in self.effects:
            for i, pos in enumerate(self.decoy_positions):
                alpha = 100 - (i * 2)
                segment_surf = pygame.Surface((CELL_SIZE, CELL_SIZE), pygame.SRCALPHA)
                pygame.draw.rect(segment_surf, (*CYAN[:3], alpha), (0, 0, CELL_SIZE, CELL_SIZE))
                surface.blit(segment_surf, (pos[0] * CELL_SIZE, pos[1] * CELL_SIZE))
        
        # Draw main snake
        for i, segment in enumerate(self.body):
            # Get color based on evolution
            if i < len(self.segment_colors):
                color = self.segment_colors[i]
            else:
                color = self.get_segment_color(i)
                self.segment_colors.append(color)
            
            # Animated texture effect for evolved segments
            if i > 10:
                pulse = math.sin(time.time() * 5 + i * 0.5) * 20
                color = (min(color[0] + int(pulse), 255), 
                        min(color[1] + int(pulse), 255), 
                        min(color[2] + int(pulse), 255))
            
            # Draw segment with gradient effect
            segment_rect = pygame.Rect(segment[0] * CELL_SIZE, segment[1] * CELL_SIZE, 
                                     CELL_SIZE, CELL_SIZE)
            pygame.draw.rect(surface, color, segment_rect)
            
            # Inner highlight for 3D effect
            if i == 0:  # Head
                pygame.draw.rect(surface, WHITE, segment_rect.inflate(-4, -4))
            else:
                inner_color = tuple(min(c + 50, 255) for c in color)
                pygame.draw.rect(surface, inner_color, segment_rect.inflate(-4, -4))

class Game:
    def __init__(self):
        self.snake = Snake()
        self.foods = []
        self.obstacles = []
        self.terrain_zones = []
        self.particles = []
        self.trails = deque(maxlen=100)
        self.score = 0
        self.game_over = False
        self.background_animation = 0
        self.time_factor = 1.0
        
        # Initialize game elements
        self.spawn_food()
        self.generate_obstacles()
        self.generate_terrain()
        
    def spawn_food(self):
        while len(self.foods) < 3:
            x = random.randint(0, GRID_SIZE - 1)
            y = random.randint(0, WINDOW_HEIGHT // CELL_SIZE - 1)
            
            # Check if position is valid
            if (x, y) not in self.snake.body and not any(obs.x == x and obs.y == y for obs in self.obstacles):
                # Random food type
                food_types = ['normal', 'normal', 'normal', 'speed', 'reverse', 'double', 'slow']
                food_type = random.choice(food_types)
                self.foods.append(Food(x, y, food_type))
                
    def generate_obstacles(self):
        for _ in range(8):
            x = random.randint(2, GRID_SIZE - 3)
            y = random.randint(2, WINDOW_HEIGHT // CELL_SIZE - 3)
            obstacle_type = random.choice(['wall', 'rotating'])
            self.obstacles.append(Obstacle(x, y, obstacle_type))
            
    def generate_terrain(self):
        # Create random terrain zones
        for _ in range(3):
            x = random.randint(0, GRID_SIZE - 5)
            y = random.randint(0, WINDOW_HEIGHT // CELL_SIZE - 5)
            width = random.randint(3, 6)
            height = random.randint(3, 6)
            zone_type = random.choice(['slow', 'warp', 'friction'])
            self.terrain_zones.append(TerrainZone(x, y, width, height, zone_type))
            
    def create_particle_explosion(self, x, y, color):
        for _ in range(20):
            angle = random.uniform(0, math.pi * 2)
            speed = random.uniform(2, 8)
            velocity = (math.cos(angle) * speed, math.sin(angle) * speed)
            size = random.randint(2, 6)
            self.particles.append(Particle(x * CELL_SIZE + CELL_SIZE // 2, 
                                         y * CELL_SIZE + CELL_SIZE // 2, 
                                         color, velocity, size))
            
    def update(self):
        if self.game_over:
            return
            
        # Update time factor for slow motion effect
        if 'time_slow' in self.snake.effects:
            self.time_factor = 0.3
        else:
            self.time_factor = 1.0
            
        # Update background animation
        self.background_animation += 0.01 * self.time_factor
        
        # Move snake
        reverse = 'reverse_controls' in self.snake.effects
        
        # Apply terrain effects
        head = self.snake.body[0]
        head_rect = pygame.Rect(head[0] * CELL_SIZE, head[1] * CELL_SIZE, CELL_SIZE, CELL_SIZE)
        
        speed_mult = 1.0
        for zone in self.terrain_zones:
            if zone.rect.colliderect(head_rect):
                speed_mult = zone.zone_effects[zone.type]['speed_mult']
                
        # Speed boost effect
        if 'speed_boost' in self.snake.effects:
            speed_mult *= 1.5
            
        # Move based on speed multiplier
        if random.random() < speed_mult:
            self.snake.move(reverse)
            
            # Add trail
            self.trails.append(Trail(self.snake.body[-1][0], self.snake.body[-1][1], 
                                   self.snake.segment_colors[-1] if self.snake.segment_colors else GREEN))
        
        # Update double vision decoy
        if 'double_vision' in self.snake.effects:
            self.snake.decoy_positions = [(seg[0] + 2, seg[1] + 2) for seg in self.snake.body[:10]]
        
        # Check collision with self
        if self.snake.body[0] in self.snake.body[1:]:
            self.game_over = True
            
        # Check collision with obstacles
        head = self.snake.body[0]
        for obstacle in self.obstacles:
            if head[0] == obstacle.x and head[1] == obstacle.y:
                self.game_over = True
                
        # Check food collision
        for food in self.foods[:]:
            if head[0] == food.x and head[1] == food.y:
                self.snake.grow(3)
                self.score += 10
                
                # Apply food effect
                food_data = food.food_types[food.type]
                if food_data['effect']:
                    self.snake.effects[food_data['effect']] = food_data['duration']
                    
                # Create particle explosion
                self.create_particle_explosion(food.x, food.y, food_data['color'])
                
                self.foods.remove(food)
                self.spawn_food()
                
        # Update effects duration
        for effect in list(self.snake.effects.keys()):
            self.snake.effects[effect] -= 1 * self.time_factor
            if self.snake.effects[effect] <= 0:
                del self.snake.effects[effect]
                if effect == 'double_vision':
                    self.snake.decoy_positions = []
                    
        # Update particles
        for particle in self.particles[:]:
            particle.update()
            if particle.lifetime <= 0:
                self.particles.remove(particle)
                
        # Update trails
        for trail in list(self.trails):
            trail.update()
            if trail.lifetime <= 0:
                self.trails.remove(trail)
                
        # Update obstacles
        for obstacle in self.obstacles:
            obstacle.update(self.time_factor)
            
        # Update terrain zones
        for zone in self.terrain_zones:
            zone.update()
            
    def draw_background(self, surface):
        # Dynamic gradient background
        for y in range(0, WINDOW_HEIGHT, 10):
            progress = y / WINDOW_HEIGHT
            
            # Pulsing color transitions
            r = int(20 + math.sin(self.background_animation + progress * 3) * 20)
            g = int(10 + math.sin(self.background_animation * 1.3 + progress * 2) * 10)
            b = int(30 + math.sin(self.background_animation * 0.7 + progress * 4) * 30)
            
            pygame.draw.rect(surface, (r, g, b), (0, y, WINDOW_WIDTH, 10))
            
    def draw(self, surface):
        # Draw background
        self.draw_background(surface)
        
        # Draw terrain zones
        for zone in self.terrain_zones:
            zone.draw(surface)
        
        # Draw trails
        for trail in self.trails:
            trail.draw(surface)
        
        # Draw obstacles
        for obstacle in self.obstacles:
            obstacle.draw(surface)
        
        # Draw food
        for food in self.foods:
            food.draw(surface, self.time_factor)
        
        # Draw snake
        self.snake.draw(surface, self.time_factor)
        
        # Draw particles
        for particle in self.particles:
            particle.draw(surface)
        
        # Draw UI
        self.draw_ui(surface)
        
    def draw_ui(self, surface):
        # Score
        font = pygame.font.Font(None, 36)
        score_text = font.render(f"Score: {self.score}", True, WHITE)
        surface.blit(score_text, (10, 10))
        
        # Active effects
        y_offset = 50
        for effect, duration in self.snake.effects.items():
            effect_text = font.render(f"{effect.replace('_', ' ').title()}: {int(duration/30)}s", 
                                    True, YELLOW)
            surface.blit(effect_text, (10, y_offset))
            y_offset += 30
            
        # Game over
        if self.game_over:
            game_over_font = pygame.font.Font(None, 72)
            game_over_text = game_over_font.render("GAME OVER", True, RED)
            text_rect = game_over_text.get_rect(center=(WINDOW_WIDTH // 2, WINDOW_HEIGHT // 2))
            
            # Glow effect
            for i in range(3):
                glow_surf = pygame.Surface((text_rect.width + i*20, text_rect.height + i*10), pygame.SRCALPHA)
                glow_surf.fill((255, 0, 0, 50-i*15))
                surface.blit(glow_surf, (text_rect.x - i*10, text_rect.y - i*5))
                
            surface.blit(game_over_text, text_rect)
            
            restart_font = pygame.font.Font(None, 36)
            restart_text = restart_font.render("Press SPACE to restart", True, WHITE)
            restart_rect = restart_text.get_rect(center=(WINDOW_WIDTH // 2, WINDOW_HEIGHT // 2 + 50))
            surface.blit(restart_text, restart_rect)

def main():
    game = Game()
    running = True
    move_timer = 0
    move_delay = 5
    
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_SPACE and game.game_over:
                    game = Game()
                elif not game.game_over:
                    if event.key == pygame.K_UP:
                        game.snake.change_direction((0, -1))
                    elif event.key == pygame.K_DOWN:
                        game.snake.change_direction((0, 1))
                    elif event.key == pygame.K_LEFT:
                        game.snake.change_direction((-1, 0))
                    elif event.key == pygame.K_RIGHT:
                        game.snake.change_direction((1, 0))
        
        # Update game at controlled intervals
        move_timer += clock.get_time()
        if move_timer >= move_delay * (10 / game.time_factor):
            move_timer = 0
            game.update()
        
        # Draw everything
        game.draw(screen)
        pygame.display.flip()
        clock.tick(60)
    
    pygame.quit()

if __name__ == "__main__":
    main()
