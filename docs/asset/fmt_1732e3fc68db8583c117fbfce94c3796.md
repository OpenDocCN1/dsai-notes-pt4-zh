from typing import List, Dict
from dataclasses import dataclass
from enum import Enum

class FeaturePriority(Enum):
    """功能优先级"""
    CORE = "核心功能"  # MVP 必须有
    IMPORTANT = "重要功能"  # 尽快添加
    NICE_TO_HAVE = "锦上添花"  # 可以延后

@dataclass
class Feature:
    """功能"""
    id: str
    name: str
    description: str
    user_story: str
    priority: FeaturePriority
    estimated_effort: int  # 人天
    value_score: int  # 1-10
    risk_score: int  # 1-10

class MVPDefinition:
    """MVP 定义"""

    def __init__(self, product_name: str):
        self.product_name = product_name
        self.features = []
        self.core_user_journey = []
        self.success_criteria = []

    def define_core_features(self) -> List[Feature]:
        """定义核心功能"""
        # 使用 Kano 模型分析
        features = []

        # 基本型需求(Must-be):用户认为理所当然的功能
        basic_features = [
            Feature(
                id="F001",
                name="用户注册/登录",
                description="用户可以创建账户并登录系统",
                user_story="作为用户,我需要创建账户以保存我的数据",
                priority=FeaturePriority.CORE,
                estimated_effort=3,
                value_score=8,
                risk_score=2
            ),
            Feature(
                id="F002",
                name="基础 AI 对话",
                description="用户可以与 AI 进行基本对话",
                user_story="作为用户,我需要与 AI 交流以获得帮助",
                priority=FeaturePriority.CORE,
                estimated_effort=5,
                value_score=10,
                risk_score=5
            )
        ]

        # 期望型需求(One-dimensional):满足度与功能成正比
        performance_features = [
            Feature(
                id="F003",
                name="上下文记忆",
                description="AI 能记住对话历史",
                user_story="作为用户,我希望 AI 记住之前的对话",
                priority=FeaturePriority.CORE,
                estimated_effort=4,
                value_score=9,
                risk_score=4
            ),
            Feature(
                id="F004",
                name="多轮对话",
                description="支持复杂的多轮交互",
                user_story="作为用户,我需要进行深入的多轮对话",
                priority=FeaturePriority.IMPORTANT,
                estimated_effort=6,
                value_score=8,
                risk_score=6
            )
        ]

        # 兴奋型需求(Attractive):超出期望的功能
        excitement_features = [
            Feature(
                id="F005",
                name="语音输入",
                description="用户可以通过语音与 AI 交互",
                user_story="作为用户,我希望可以用语音输入",
                priority=FeaturePriority.NICE_TO_HAVE,
                estimated_effort=8,
                value_score=7,
                risk_score=7
            )
        ]

        features = basic_features + performance_features + excitement_features
        self.features = features

        return features

    def prioritize_for_mvp(self) -> List[Feature]:
        """为 MVP 优先排序功能"""
        # 计算价值/努力比率
        def calculate_score(feature: Feature) -> float:
            # 价值分数 / (努力 * 风险)
            return feature.value_score / (feature.estimated_effort * (feature.risk_score / 10))

        # 只选择核心功能
        core_features = [f for f in self.features if f.priority == FeaturePriority.CORE]

        # 按价值/努力比率排序
        core_features.sort(key=calculate_score, reverse=True)

        return core_features

    def define_user_journey(self) -> List[Dict]:
        """定义核心用户旅程"""
        journey = [
            {
                'step': 1,
                'action': '用户访问产品',
                'touchpoint': '着陆页',
                'goal': '了解产品价值',
                'pain_point': '不清楚产品能做什么',
                'solution': '清晰的价值主张和演示视频'
            },
            {
                'step': 2,
                'action': '用户注册',
                'touchpoint': '注册表单',
                'goal': '快速创建账户',
                'pain_point': '注册流程复杂',
                'solution': '简化注册,支持社交登录'
            },
            {
                'step': 3,
                'action': '首次使用',
                'touchpoint': '引导流程',
                'goal': '理解如何使用',
                'pain_point': '不知道从哪里开始',
                'solution': '交互式教程和示例'
            },
            {
                'step': 4,
                'action': '核心功能使用',
                'touchpoint': 'AI 对话界面',
                'goal': '完成主要任务',
                'pain_point': 'AI 响应不准确',
                'solution': '优化提示和模型'
            },
            {
                'step': 5,
                'action': '获得价值',
                'touchpoint': '结果页面',
                'goal': '看到明显效果',
                'pain_point': '价值不明显',
                'solution': '可视化结果和节省时间'
            }
        ]

        self.core_user_journey = journey
        return journey

    def define_success_criteria(self) -> List[Dict]:
        """定义成功标准"""
        criteria = [
            {
                'category': '用户激活',
                'metric': '注册后 24 小时内完成首次核心操作',
                'target': '> 60%',
                'measurement': '用户行为分析'
            },
            {
                'category': '用户留存',
                'metric': '7 日留存率',
                'target': '> 40%',
                'measurement': '用户登录数据'
            },
            {
                'category': '用户满意度',
                'metric': 'NPS 分数',
                'target': '> 30',
                'measurement': '用户调查'
            },
            {
                'category': '技术性能',
                'metric': 'AI 响应时间',
                'target': '< 3 秒',
                'measurement': '性能监控'
            },
            {
                'category': '技术质量',
                'metric': 'AI 准确率',
                'target': '> 85%',
                'measurement': '人工评估'
            }
        ]

        self.success_criteria = criteria
        return criteria

    def generate_mvp_spec(self) -> Dict:
        """生成 MVP 规格说明"""
        core_features = self.prioritize_for_mvp()

        total_effort = sum(f.estimated_effort for f in core_features)

        return {
            'product': self.product_name,
            'mvp_features': [
                {
                    'id': f.id,
                    'name': f.name,
                    'user_story': f.user_story,
                    'effort_days': f.estimated_effort
                }
                for f in core_features
            ],
            'total_effort_days': total_effort,
            'estimated_weeks': round(total_effort / 5, 1),  # 假设每周 5 个工作日
            'user_journey': self.core_user_journey,
            'success_criteria': self.success_criteria
        }

# 使用示例
mvp = MVPDefinition("AI 写作助手")
mvp.define_core_features()
mvp.define_user_journey()
mvp.define_success_criteria()

spec = mvp.generate_mvp_spec()
print(f"MVP: {spec['product']}")
print(f"核心功能数: {len(spec['mvp_features'])}")
print(f"预计开发时间: {spec['estimated_weeks']} 周")
