from typing import Dict, List
import re

class BiasDetector:
    """偏见检测器"""
    
    # 性别偏见词汇
    GENDERED_TERMS = {
        'male': ['he', 'him', 'his', 'man', 'men', 'male', 'gentleman', 'boy'],
        'female': ['she', 'her', 'hers', 'woman', 'women', 'female', 'lady', 'girl']
    }
    
    # 刻板印象关联
    STEREOTYPICAL_ASSOCIATIONS = {
        'gender': {
            'male': ['engineer', 'doctor', 'CEO', 'programmer', 'scientist'],
            'female': ['nurse', 'teacher', 'secretary', 'assistant', 'receptionist']
        },
        'race': {
            # 根据实际需求添加
        }
    }
    
    def detect_gender_bias(self, text: str) -> Dict:
        """检测性别偏见"""
        text_lower = text.lower()
        
        # 统计性别词汇出现频率
        male_count = sum(text_lower.count(term) for term in self.GENDERED_TERMS['male'])
        female_count = sum(text_lower.count(term) for term in self.GENDERED_TERMS['female'])
        
        total = male_count + female_count
        if total == 0:
            return {
                'has_bias': False,
                'bias_score': 0.0,
                'details': '未检测到性别相关词汇'
            }
        
        # 计算偏见分数(偏离50/50的程度)
        male_ratio = male_count / total
        bias_score = abs(male_ratio - 0.5) * 2  # 归一化到 0-1
        
        # 检测刻板印象关联
        stereotypes = self._detect_stereotypes(text)
        
        return {
            'has_bias': bias_score > 0.3 or len(stereotypes) > 0,
            'bias_score': bias_score,
            'male_ratio': male_ratio,
            'female_ratio': 1 - male_ratio,
            'stereotypes': stereotypes,
            'recommendation': self._generate_debiasing_recommendation(bias_score, stereotypes)
        }
    
    def _detect_stereotypes(self, text: str) -> List[str]:
        """检测刻板印象"""
        text_lower = text.lower()
        detected = []
        
        for gender, professions in self.STEREOTYPICAL_ASSOCIATIONS['gender'].items():
            gender_terms = self.GENDERED_TERMS[gender]
            
            for profession in professions:
                # 检查是否同时出现性别词和职业词
                if profession in text_lower:
                    for term in gender_terms:
                        # 使用正则检查词边界
                        if re.search(rf'\b{term}\b.*\b{profession}\b|\b{profession}\b.*\b{term}\b', 
                                   text_lower):
                            detected.append(f"{gender}-{profession} 关联")
        
        return detected
    
    def _generate_debiasing_recommendation(self, bias_score: float, stereotypes: List[str]) -> str:
        """生成去偏见建议"""
        recommendations = []
        
        if bias_score > 0.3:
            recommendations.append("考虑平衡性别代词的使用,或使用性别中性语言(如 'they', '他们')")
        
        if stereotypes:
            recommendations.append(f"检测到刻板印象关联: {', '.join(stereotypes)}。考虑使用更多样化的例子")
        
        return '; '.join(recommendations) if recommendations else "未检测到显著偏见"
    
    def debias_text(self, text: str) -> str:
        """自动去偏见(简单实现)"""
        debiased = text
        
        # 替换性别化职业词汇为中性词汇
        gender_neutral_replacements = {
            'policeman': 'police officer',
            'fireman': 'firefighter',
            'chairman': 'chairperson',
            'mankind': 'humankind',
            'manpower': 'workforce',
        }
        
        for biased, neutral in gender_neutral_replacements.items():
            debiased = re.sub(rf'\b{biased}\b', neutral, debiased, flags=re.IGNORECASE)
        
        return debiased

bias_detector = BiasDetector()

test_text = """
The engineer said he would fix the problem.
The nurse said she would check on the patient.
The CEO announced his new strategy.
"""

bias_report = bias_detector.detect_gender_bias(test_text)
print(f"存在偏见: {bias_report['has_bias']}")
print(f"偏见分数: {bias_report['bias_score']:.2f}")
print(f"男性比例: {bias_report['male_ratio']:.2%}")
print(f"女性比例: {bias_report['female_ratio']:.2%}")
print(f"刻板印象: {bias_report['stereotypes']}")
print(f"建议: {bias_report['recommendation']}")

debiased_text = bias_detector.debias_text("The chairman and policeman arrived.")
print(f"去偏见后: {debiased_text}")
