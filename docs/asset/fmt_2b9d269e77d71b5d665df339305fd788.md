```
def chat_base(messages):
    response = client.chat.completions.create(
        model="deepseek-reasoner",  
        messages=messages,
        tools=tools,
    )

    if response.choices[0].finish_reason == "tool_calls":
        while True:
            messages = create_function_response_messages(messages, response)
            response = client.chat.completions.create(
                model="deepseek-reasoner",  
                messages=messages,
                tools=tools,
            )
            if response.choices[0].finish_reason != "tool_calls":
                break
    return response
```
