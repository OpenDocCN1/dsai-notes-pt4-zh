                    'solution': solution['solution'],
                    'steps': solution['implementation'],
                    'impact': solution['expected_improvement'],
                    'effort': self._estimate_effort(solution)
                })

        step_count = len(solution['implementation'])
        if step_count <= 2:
            return '低 (1-3 天)'
        elif step_count <= 4:
            return '中 (1 周)'
        else:
            return '高 (2-3 周)'
    
    def _calculate_roi(self, item: Dict) -> float:
        """计算 ROI"""
        impact_score = {
            '低': 1,
            '中': 2,
            '高': 3
        }
        
        effort_score = {
            '低 (1-3 天)': 1,
            '中 (1 周)': 2,
            '高 (2-3 周)': 3
        }
        
        # 从 impact 字符串中提取数字
        import re
        impact_match = re.search(r'(\d+)%', item['impact'])
        impact = int(impact_match.group(1)) if impact_match else 50
        
        effort = effort_score.get(item['effort'], 2)
        
        return impact / effort
