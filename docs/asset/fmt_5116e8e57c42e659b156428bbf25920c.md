# Define the parametric equations for the heart shape

# Create the figure and plot the curve
plt.figure(figsize=(6, 6))
plt.plot(x, y, 'r')
plt.title('Heart Shape')
plt.xlabel('x-axis')
plt.ylabel('y-axis')
plt.grid(True)
plt.axis('equal')

# Show the plot
