## `MOVSB`

## `CMPSB`

比较两个大小

+ `0Ah`是输入字符串，要求数组偏移地址在`dx`
+ 调用后`buf[2]`之后才是输入的内容，`buf[1]`是长度
+ 先读到`buf`里再`memcpy`到`s`

```
data segment
buf db 81, 0, 81 dup(0)
s db 100 dup(0)
t db 100 dup(0)
slen db 0
tlen db 0
equal_msg db "Equal", 0Dh, 0Ah, '$'
unequal_msg db "Not equal", 0Dh, 0Ah, '$'
data ends

code segment
assume cs:code, ds:data
input:
   mov ah, 0Ah
   mov dx, offset buf
   int 21h       ; 输入一个长度≤81字符(含回车)的字符串
                 ; 调用前:
                 ; buf[0]=字符串最大长度(含回车)
                 ; 调用后:
                 ; buf[1]=字符串长度(不含回车)
                 ; buf[2]起为输入的字符串内容
   mov ah, 2
   mov dl, 0Dh
   int 21h       ; 输出回车
   mov dl, 0Ah
   int 21h       ; 输出换行
   xor cx, cx
   mov cl, buf[1]
   push cx
   lea si, buf[2]
   rep movsb     ; 把输入的字符串复制到目标地址es:di
   pop cx
   mov byte ptr es:[di], 0; 目标字符串填入结束标志
   mov ax, cx    ; 返回AX=字符串长度
   ret

main:
   mov ax, data
   mov ds, ax
   mov es, ax
   cld
   mov di, offset s
   call input    ; 输入一个字符串保存到s
   mov [slen], al; slen = s的长度
   mov di, offset t
   call input    ; 输入一个字符串保存到t
   mov [tlen], al; tlen = t的长度
   xor cx, cx
   mov cl, [slen]
   cmp cl, [tlen]
   jne not_equal ; 若字符串长度不等则输出"Not equal"
prepare_to_cmp:
   or cx, cx
   jz exit       ; 若字符串长度为0则结束程序
   mov si, offset s
   mov di, offset t
   repe cmpsb    ; 比较字符串
   je is_equal   ; 若最后一次比较相等则一定全等
not_equal:
   mov dx, offset unequal_msg
   jmp output
is_equal:
   mov dx, offset equal_msg
output:
   mov ah, 9
   int 21h
exit:
   mov ah, 4Ch
   int 21h
code ends
end main
```

## `SCASB`

用`repne scasb`求字符串长度

+ `Inc`
+ `not`

```
data segment
s db "abracadabra", 0
slen dw 0
data ends

code segment
assume cs:code, ds:data
main:
   mov ax, data
   mov ds, ax
   mov es, ax
   mov di, offset s; es:di->s[0]
   mov cx, 0FFFFh  ; CX=最大搜索次数
   xor al, al      ; AL=00h
   cld             ; DF=0
   repne scasb     ; 若本次比较不等则继续搜索
   inc cx          ; 假装末尾的00h没搜过
   not cx          ; CX=0FFFFh-CX=已搜过的字符个数(不含末尾的00h)
   mov [slen], cx  ; 保存字符串长度
   mov ah, 4Ch
   int 21h
code ends
end main
```

## `STOSB`

把数组`s`清零

```
.386
data segment use16
s db 2003 dup('A')
slen = $ - offset s
data ends

code segment use16
assume cs:code, ds:data
main:
   mov ax, data
   mov ds, ax
   mov es, ax
   mov di, offset s; ES:DI->s[0]
   mov cx, slen    ; CX = 数组s中包含的字符个数 = 2003
   xor eax, eax    ; EAX = 0
   push cx
   shr cx, 2       ; CX = CX / 4 = 500
   rep stosd       ; 先把s[0]至s[1999]清零
   pop cx
   and cx, 3       ; CX = CX % 4 = 3
   rep stosb       ; 再把s[2000]至s[2002]清零
   mov ah, 4Ch
   int 21h
code ends
end main
```

## `LODSB`

过滤数组的空格

+ 先`lodsb`再`stosb`，遇到要跳过的可以跳过

```
data segment
s db "Ada Lovelace is often regarded as "
  db "the first computer programmer."
slen = $ - offset s
t db slen dup('T')
data ends

code segment
assume cs:code, ds:data
main:
   mov ax, data
   mov ds, ax
   mov es, ax
   mov si, offset s; ds:si->s[0]
   mov di, offset t; es:di->t[0]
   mov cx, slen    ; CX = length of s
   cld             ; DF = 0
   jcxz done       ; jump if cx is zero
again:
   lodsb           ; AL=ds:[si], si++
   cmp al, ' '
   je skip
   stosb           ; es:[di]=AL, di++
skip:
   loop again      ; dec cx
                   ; jnz again
done:
   xor al, al      ; AL = 00h
   stosb           ; es:[di]=AL, di++
   mov ah, 4Ch
   int 21h
code ends
end main
```

## 用`call`和`retn`实现`puts`

简单的函数调用

+ 注意判断为0的时候结束
+ 注意堆栈结构

```
data segment
s db "Hello$World!", 0Dh, 0Ah, 00h
data ends

code segment
assume cs:code, ds:data
;Input:          ; 用寄存器传递参数
;AL=char to output
putchar:         ; 用标号定义一个函数
   push dx
   mov ah, 2
   mov dl, al
   int 21h
   pop dx
   retn

;Input:          ; 用堆栈传递参数
;dword ptr ss:[sp+2]->string to output
puts proc        ; 用函数名 proc ... 函数名 endp定义一个函数
   push bp       ;   ~~~~~~~~~~~     ~~~~~~~~~~~
   mov bp, sp    ; 构造堆栈框架
                 ; word ptr ss:[bp+0] = old bp
                 ; word ptr ss:[bp+2] = offset back
                 ; word ptr ss:[bp+4] = offset s
                 ; word ptr ss:[bp+6] = data
   push ds
   push si
   cld
   lds si, [bp+4]; SI = offset s
                 ; DS = data
puts_next_char:
   lodsb
   or al, al
   jz puts_done
   call putchar
   jmp puts_next_char
puts_done:
   pop si
   pop ds
   pop bp
   retn 4        ; back_addr = ss:[sp] = offset back
                 ; sp = sp + 6, 弹出返回地址及main压入的两个字
                 ; IP = back_addr = offset back, 返回到back:
puts endp

main:
   mov ax, data
   push ax       ; 先压入s的段地址
   mov ax, offset s
   push ax       ; 后压入s的偏移地址
   call puts     ; 调用函数puts()输出s
back:
   mov ah, 4Ch
   int 21h
code ends
end main
```

## `Callfar`

注意远端的调用

```
data segment
s db "Hello$World!", 0Dh, 0Ah, 00h
data ends

faraway segment
assume cs:faraway, ds:data
;Input:          ; 用寄存器传递参数
;AL=char to output
putchar label far; 用标号定义一个函数
   push dx
   mov ah, 2
   mov dl, al
   int 21h
   pop dx
   retf

;Input:          ; 用堆栈传递参数
;dword ptr ss:[sp+4]->string to output
puts proc far    ; 用函数名 proc far ... 函数名 endp定义一个函数
   push bp       ;   ~~~~~~~~~~~~~~~     ~~~~~~~~~~~
   mov bp, sp    ; 构造堆栈框架
                 ; word ptr ss:[bp+0] = old bp
                 ; word ptr ss:[bp+2] = offset back
                 ; word ptr ss:[bp+4] = code
                 ; word ptr ss:[bp+6] = offset s
                 ; word ptr ss:[bp+8] = data
   push ds
   push si
   cld
   lds si, [bp+6]; SI = offset s
                 ; DS = data
puts_next_char:
   lodsb
   or al, al
   jz puts_done
   call far ptr putchar
   jmp puts_next_char
puts_done:
   pop si
   pop ds
   pop bp
   retf 4        ; back_ip = ss:[sp] = offset back
                 ; back_cs = ss:[sp+2] = code
                 ; sp = sp + 8, 弹出返回地址及main压入的两个字
                 ; IP = back_ip = offset back\
                 ; CS = back_cs = code       / 返回code:back
puts endp
faraway ends

code segment
assume cs:code, ds:data
prompt db "far call demo", 0Dh, 0Ah, 00h
prompt_len = $ - offset prompt - 1
main:
   mov cx, prompt_len
   mov bx, 0
show_prompt:
   mov al, cs:prompt[bx]
   call far ptr putchar
   inc bx
   loop show_prompt
   mov ax, data
   push ax          ; 先压入s的段地址
   mov ax, offset s
   push ax          ; 后压入s的偏移地址
   call far ptr puts; 调用函数puts()输出s
back:
   mov ah, 4Ch
   int 21h
code ends
end main
```
