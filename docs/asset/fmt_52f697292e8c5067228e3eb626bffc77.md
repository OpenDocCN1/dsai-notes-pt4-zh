from typing import List, Dict
from datetime import datetime, timedelta
from enum import Enum

class FeedbackSource(Enum):
    """反馈来源"""
    USER_TESTING = "用户测试"
    ANALYTICS = "数据分析"
    SUPPORT_TICKETS = "客服工单"
    USER_INTERVIEWS = "用户访谈"
    SURVEYS = "问卷调查"

class Priority(Enum):
    """优先级"""
    CRITICAL = "紧急"  # 阻塞性问题
    HIGH = "高"  # 严重影响体验
    MEDIUM = "中"  # 改进体验
    LOW = "低"  # 锦上添花

@dataclass
class Feedback:
    """反馈"""
    id: str
    source: FeedbackSource
    description: str
    priority: Priority
    affected_users: int
    reported_date: datetime

@dataclass
class Iteration:
    """迭代"""
    number: int
    start_date: datetime
    end_date: datetime
    goals: List[str]
    completed_items: List[str]
    metrics: Dict

class FeedbackIterationLoop:
    """反馈迭代循环"""
    
    def __init__(self, product_name: str):
        self.product_name = product_name
        self.feedback_items = []
        self.iterations = []
        self.backlog = []
    
    def collect_feedback(self) -> List[Feedback]:
        """收集反馈"""
        # 从多个渠道收集反馈
        channels = {
            FeedbackSource.USER_TESTING: [
                "注册流程太复杂",
                "AI 响应速度慢",
                "不知道如何开始使用"
            ],
            FeedbackSource.ANALYTICS: [
                "70% 用户在注册页流失",
                "平均会话时长只有 2 分钟",
                "移动端使用率低"
            ],
            FeedbackSource.SUPPORT_TICKETS: [
                "忘记密码功能不工作",
                "生成的内容不准确",
                "无法保存草稿"
            ]
        }
        
        for source, items in channels.items():
            for item in items:
                feedback = Feedback(
                    id=f"FB-{len(self.feedback_items) + 1:03d}",
                    source=source,
                    description=item,
                    priority=self._assess_priority(item),
                    affected_users=0,  # 实际应用中从数据获取
                    reported_date=datetime.now()
                )
                self.feedback_items.append(feedback)
        
        return self.feedback_items
    
    def _assess_priority(self, description: str) -> Priority:
        """评估优先级"""
        # 简化实现:基于关键词
        critical_keywords = ['不工作', '崩溃', '无法', '阻塞']
        high_keywords = ['流失', '慢', '不准确']
        
        desc_lower = description.lower()
        
        if any(kw in desc_lower for kw in critical_keywords):
            return Priority.CRITICAL
        elif any(kw in desc_lower for kw in high_keywords):
            return Priority.HIGH
        else:
            return Priority.MEDIUM
    
    def prioritize_backlog(self) -> List[Feedback]:
        """优先排序待办列表"""
        # 使用 RICE 评分框架
        # RICE = (Reach × Impact × Confidence) / Effort
        
        def calculate_rice_score(feedback: Feedback) -> float:
            reach = feedback.affected_users
            impact_map = {
                Priority.CRITICAL: 3,
                Priority.HIGH: 2,
                Priority.MEDIUM: 1,
                Priority.LOW: 0.5
            }
            impact = impact_map[feedback.priority]
            confidence = 0.8  # 80% 置信度
            effort = 1  # 简化:假设都是 1 周工作量
            
            return (reach * impact * confidence) / effort
        
        # 按 RICE 分数排序
        self.backlog = sorted(
            self.feedback_items,
            key=calculate_rice_score,
            reverse=True
        )
        
        return self.backlog
    
    def plan_iteration(
        self,
        iteration_number: int,
        duration_days: int = 7
    ) -> Iteration:
        """规划迭代"""
        start_date = datetime.now()
        end_date = start_date + timedelta(days=duration_days)
        
        # 选择本次迭代要解决的问题
        # 遵循 80/20 原则:解决 20% 的问题,影响 80% 的用户
        
        iteration_items = []
        capacity = 3  # 一周解决 3 个问题
        
        for feedback in self.backlog[:capacity]:
            iteration_items.append(feedback.description)
        
        goals = [
            f"解决 {len(iteration_items)} 个高优先级问题",
            "提升用户满意度",
            "减少流失率"
        ]
        
        iteration = Iteration(
            number=iteration_number,
            start_date=start_date,
            end_date=end_date,
            goals=goals,
            completed_items=[],
            metrics={}
        )
        
        self.iterations.append(iteration)
        return iteration
    
    def execute_iteration(self, iteration: Iteration) -> None:
        """执行迭代"""
        print(f"
🚀 迭代 {iteration.number} 开始")
        print(f"时间: {iteration.start_date.strftime('%Y-%m-%d')} - {iteration.end_date.strftime('%Y-%m-%d')}")
        print(f"目标:")
        for goal in iteration.goals:
            print(f"  - {goal}")
        
        # 实际开发工作在这里进行
        # 使用 AI 工具加速开发
        
        print("
开发进度:")
        print("  [Day 1-2] 修复关键 bug")
        print("  [Day 3-4] 优化用户流程")
        print("  [Day 5-6] 性能优化")
        print("  [Day 7] 测试和部署")
    
    def measure_iteration_impact(self, iteration: Iteration) -> Dict:
        """衡量迭代影响"""
        # 对比迭代前后的关键指标
        metrics = {
            'before': {
                'user_satisfaction': 3.2,
                'completion_rate': 65,
                'avg_session_time': 2.1,
                'churn_rate': 30
            },
            'after': {
                'user_satisfaction': 3.8,
                'completion_rate': 78,
                'avg_session_time': 4.5,
                'churn_rate': 22
            },
            'improvement': {}
        }
        
        # 计算改进幅度
        for key in metrics['before']:
            before = metrics['before'][key]
            after = metrics['after'][key]
            
            if key == 'churn_rate':
                # 流失率越低越好
                improvement = ((before - after) / before * 100)
            else:
                # 其他指标越高越好
                improvement = ((after - before) / before * 100)
            
            metrics['improvement'][key] = round(improvement, 1)
        
        iteration.metrics = metrics
        return metrics
    
    def generate_iteration_report(self, iteration: Iteration) -> Dict:
        """生成迭代报告"""
        metrics = self.measure_iteration_impact(iteration)
        
        return {
            'iteration': iteration.number,
            'duration': f"{(iteration.end_date - iteration.start_date).days} 天",
            'goals': iteration.goals,
            'completed': len(iteration.completed_items),
            'metrics': metrics,
            'key_improvements': [
                f"用户满意度提升 {metrics['improvement']['user_satisfaction']}%",
                f"完成率提升 {metrics['improvement']['completion_rate']}%",
                f"流失率降低 {metrics['improvement']['churn_rate']}%"
            ],
            'next_steps': [
                "继续收集用户反馈",
                "规划下一个迭代",
                "扩大测试用户群"
            ]
        }

# 使用示例
feedback_loop = FeedbackIterationLoop("AI 写作助手")

# 收集反馈
feedback_loop.collect_feedback()

# 优先排序
feedback_loop.prioritize_backlog()

# 规划迭代
iteration1 = feedback_loop.plan_iteration(1, duration_days=7)

# 执行迭代
feedback_loop.execute_iteration(iteration1)

# 生成报告
report = feedback_loop.generate_iteration_report(iteration1)
print(f"
📊 迭代报告:")
print(f"迭代: {report['iteration']}")
print(f"时长: {report['duration']}")
print(f"关键改进:")
for improvement in report['key_improvements']:
    print(f"  ✅ {improvement}")
