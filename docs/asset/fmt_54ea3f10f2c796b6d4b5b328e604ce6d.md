class ResourceACL:
    def __init__(self):
        # 资源访问控制列表
        self.acl = {}
    
    def grant_access(self, resource_id, user_id, permissions):
        """授予用户对资源的访问权限"""
        if resource_id not in self.acl:
            self.acl[resource_id] = {}
        
        self.acl[resource_id][user_id] = set(permissions)
    
    def revoke_access(self, resource_id, user_id):
        """撤销用户对资源的访问权限"""
        if resource_id in self.acl and user_id in self.acl[resource_id]:
            del self.acl[resource_id][user_id]
    
    def check_access(self, resource_id, user_id, permission):
        """检查用户是否有权访问资源"""
        if resource_id not in self.acl:
            return False
        
        if user_id not in self.acl[resource_id]:
            return False
        
        return permission in self.acl[resource_id][user_id]

# 使用示例
acl = ResourceACL()

# 授予权限
acl.grant_access("document_123", "alice", ["read", "write"])
acl.grant_access("document_123", "bob", ["read"])

# 检查权限
if acl.check_access("document_123", "alice", "write"):
    print("Alice 可以编辑文档")

if not acl.check_access("document_123", "bob", "write"):
    print("Bob 只能查看文档")

import logging
import json
from datetime import datetime
from typing import Dict, Any

class SecurityAuditLogger:
    def __init__(self, log_file="security_audit.log"):
        self.logger = logging.getLogger("SecurityAudit")
        self.logger.setLevel(logging.INFO)
        
        # 文件处理器
        file_handler = logging.FileHandler(log_file)
        file_handler.setLevel(logging.INFO)
        
        # JSON 格式化器
        formatter = logging.Formatter(
            '%(message)s'
        )
        file_handler.setFormatter(formatter)
        
        self.logger.addHandler(file_handler)
    
    def log_event(self, event_type: str, user_id: str, 
                  details: Dict[str, Any], status: str = "success"):
        """记录安全事件"""
        log_entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "event_type": event_type,
            "user_id": user_id,
            "status": status,
            "details": details,
            "ip_address": self._get_client_ip(),
            "user_agent": self._get_user_agent()
        }
        
        self.logger.info(json.dumps(log_entry))
    
    def log_authentication(self, user_id: str, method: str, 
                          success: bool, reason: str = None):
        """记录认证事件"""
        self.log_event(
            event_type="authentication",
            user_id=user_id,
            details={
                "method": method,
                "reason": reason
            },
            status="success" if success else "failure"
        )
    
    def log_authorization(self, user_id: str, resource: str, 
                         action: str, granted: bool):
        """记录授权事件"""
        self.log_event(
            event_type="authorization",
            user_id=user_id,
            details={
                "resource": resource,
                "action": action,
                "granted": granted
            },
            status="granted" if granted else "denied"
        )
    
    def log_data_access(self, user_id: str, resource_type: str, 
                       resource_id: str, operation: str):
        """记录数据访问事件"""
        self.log_event(
            event_type="data_access",
            user_id=user_id,
            details={
                "resource_type": resource_type,
                "resource_id": resource_id,
                "operation": operation
            }
        )
    
    def log_configuration_change(self, user_id: str, 
                                component: str, changes: Dict):
        """记录配置变更事件"""
        self.log_event(
            event_type="configuration_change",
            user_id=user_id,
            details={
                "component": component,
                "changes": changes
            }
        )
    
    def log_security_incident(self, user_id: str, 
                            incident_type: str, severity: str, 
                            description: str):
        """记录安全事件"""
        self.log_event(
            event_type="security_incident",
            user_id=user_id,
            details={
                "incident_type": incident_type,
                "severity": severity,
                "description": description
            },
            status="alert"
        )
    
    def _get_client_ip(self):
        """获取客户端 IP 地址"""
        # 实际实现中从请求上下文获取
        return "127.0.0.1"
    
    def _get_user_agent(self):
        """获取用户代理字符串"""
        # 实际实现中从请求上下文获取
        return "OpenClaw/1.0"

audit_logger = SecurityAuditLogger()
