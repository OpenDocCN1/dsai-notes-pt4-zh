
```python
import os
import gradio as gr
from openai import OpenAI, APIError, RateLimitError, AuthenticationError
import traceback
import json
import pprint # 用于更美观地打印复杂数据结构

# --- 安全地获取 API Key ---
api_key = os.getenv("AOAI_API_KEY")

# --- 初始化 OpenAI 客户端 ---
client = None
if api_key:
    try:
        client = OpenAI(
            api_key=api_key,
            base_url="https://api.huaqloud.com/v1" # 请确保这是你的 AOAI API endpoint
            # 如果你的 AOAI 服务需要，可能还需要 api_version 参数，例如:
            # api_version="2023-12-01-preview"
        )
        print("AOAI API 客户端初始化成功。")
    except Exception as e:
        print(f"初始化 OpenAI 客户端时出错: {e}")
        client = None
else:
    print("警告：环境变量 AOAI_API_KEY 未设置。")
    client = None

# --- 定义系统消息和历史限制 ---
SYSTEM_MESSAGE = {"role": "system", "content": "你是一个简洁而友好的聊天机器人。"}
MAX_HISTORY_TURNS = 10 # 保留最近10轮对话（10用户+10助手）

# --- 定义 Gradio 的核心处理函数 ---
def predict(message: str, history_from_gradio: list[dict[str, any]]):
    if not client:
        yield "**错误**：AOAI API 客户端未初始化。请检查 API Key 设置。"
        return

    print("\n----------- history_from_gradio (原始，由Gradio传入) -----------")
    pprint.pprint(history_from_gradio)
    print("--------------------------------------------------------------")

    messages_for_api = [SYSTEM_MESSAGE.copy()] # 总是从系统消息开始

    # 清理 history_from_gradio，只保留 role 和 content
    for turn_data in history_from_gradio:
        # Gradio 的 history 元素可能包含额外的键，如 'metadata', 'options'
        # 我们只需要 'role' 和 'content'
        role = turn_data.get("role")
        content = turn_data.get("content")

        if role and content is not None: # 确保 role 和 content 都有效
            messages_for_api.append({"role": role, "content": content})
        else:
            # 兼容旧版 Gradio 可能的 (user, assistant) 元组格式 (虽然 type="messages" 不应如此)
            if isinstance(turn_data, (list, tuple)) and len(turn_data) == 2:
                user_msg, assistant_msg = turn_data
                if user_msg:
                    messages_for_api.append({"role": "user", "content": str(user_msg)})
                if assistant_msg:
                    messages_for_api.append({"role": "assistant", "content": str(assistant_msg)})
            else:
                print(f"警告: 从 history_from_gradio 中跳过格式不正确的条目: {turn_data}")

    messages_for_api.append({"role": "user", "content": message}) # 添加当前用户输入

    # (可选) 限制历史记录长度，防止 Token 过多
    if len(messages_for_api) > 1 + MAX_HISTORY_TURNS * 2: # 1 是 system message
        relevant_conversation = messages_for_api[1:] # 对话历史（不含系统消息）
        messages_for_api = [messages_for_api[0]] + relevant_conversation[-(MAX_HISTORY_TURNS * 2):]

    print(f"\n--- 向 API 发送的消息 (共 {len(messages_for_api)} 条) ---")
    print(json.dumps(messages_for_api, indent=2, ensure_ascii=False))
    print("--------------------------------------------------")

    try:
        stream_response = client.chat.completions.create(
            model="gpt-4o-mini", # 确保你的 AOAI endpoint 支持这个模型名
            messages=messages_for_api, # 现在 messages_for_api 是清理过的
            stream=True,
            temperature=0.7,
            max_tokens=1500 # 根据需要调整
        )

        assistant_response_content = ""
        print("机器人 (流式): ", end="")
        for chunk in stream_response:
            if chunk.choices and chunk.choices[0].delta and chunk.choices[0].delta.content:
                content_part = chunk.choices[0].delta.content
                assistant_response_content += content_part
                print(content_part, end="", flush=True)
                yield assistant_response_content # 流式返回给 Gradio
        
        print("\n--- 流式输出结束 ---")

    except AuthenticationError as e:
        print(f"\n认证错误: {e}")
        yield f"**错误**：认证失败。检查 API Key 和 API Base URL。"
    except RateLimitError as e:
        print(f"\n速率限制错误: {e}")
        yield f"**错误**：请求过于频繁，请稍后再试。"
    except APIError as e:
        print(f"\nAPI 错误: Status Code={e.status_code}, Response={e.response}")
        error_detail = f"原始错误信息: {e.message}" # 默认错误信息
        if e.response:
            try:
                error_content = e.response.json()
                # 尝试提取 OpenAI 风格的错误消息
                specific_error = error_content.get("error")
                if isinstance(specific_error, dict):
                    error_detail = specific_error.get("message", str(specific_error))
                elif isinstance(specific_error, str): # 有时 error 直接是字符串
                    error_detail = specific_error
                elif "detail" in error_content: # 有些 API 返回 'detail'
                     error_detail = str(error_content.get("detail"))
                else: #  fallback
                    error_detail = str(error_content)

            except json.JSONDecodeError: # 如果响应不是JSON
                error_detail = e.response.text
            except Exception as json_e: # 其他解析错误
                 error_detail = f"无法解析API错误响应: {json_e}, 原始响应: {e.response.text[:500]}"

        yield f"**错误**：API 返回错误 (代码: {e.status_code})。详情: {error_detail}"
    except Exception as e:
        print(f"\n发生意外错误: {type(e).__name__} - {e}")
        traceback.print_exc()
        yield f"**发生意外错误**: {type(e).__name__}。请检查服务器日志。"

# --- 创建 Gradio 聊天界面组件 ---
chatbot_component = gr.Chatbot(
    label="聊天窗口",
    height=700,
    show_copy_button=True,
)

# --- 使用 gr.ChatInterface 创建完整的聊天应用 UI ---
chatbot_interface = gr.ChatInterface(
    fn=predict,
    title="🚀 AI 聊天助手 (Gradio版)",
    description="你好！我是你的AI聊天伙伴。输入你的问题，开始对话吧！",
    chatbot=chatbot_component, # 使用自定义的 chatbot
    type="messages",          # 确保 history 是 OpenAI 格式
    examples=[
        ["你好！"],
        ["给我讲个关于程序员的笑话。"],
        ["解释一下什么是人工智能。"]
    ],
    cache_examples=False, # 开发时设为 False 方便调试
    textbox=gr.Textbox(placeholder="和我说点什么吧...", container=False, scale=7),
    # 如果你的 Gradio 版本较新，可以尝试启用这些自定义按钮文本
    # retry_btn="🔄 重试",
    # undo_btn="↩️ 撤销",
    # clear_btn="🗑️ 清除对话"
)

# --- 启动 Gradio 应用 ---
if __name__ == "__main__":
    if not client:
        print("\n" + "="*40)
        print("‼️ 严重警告：未能初始化 AOAI API 客户端。")
        print("   聊天功能将无法工作！")
        print("   请确保正确设置并导出 AOAI_API_KEY 环境变量，")
        print("   并检查 API Base URL 是否正确。")
        print("="*40 + "\n")
    else:
        print("\n" + "="*40)
        print("✅ AOAI API 客户端已准备就绪。启动 Gradio 应用...")
        print("   访问 http://127.0.0.1:7860 (或 Gradio 启动时指定的其他地址)")
        print("="*40 + "\n")

    chatbot_interface.launch() # 默认 server_name="127.0.0.1", server_port=7860
    # 若需局域网访问: chatbot_interface.launch(server_name="0.0.0.0")
```

## 代码关键点解析

1. `import gradio as gr`: 导入 Gradio 库。

2. `predict(message, history)` 函数：这是 Gradio 应用的核心逻辑。

   + 它接收用户当前输入 `message` 和由 Gradio 自动管理的 `history`（我们配置了 `type='messages'`，所以 `history` 直接是 OpenAI 的消息格式列表）。

   + 构建 `messages`：我们添加了一个系统提示（system role），然后将 `history` 扩展进去，最后（隐式地，由 Gradio 处理）用户的最新 `message` 也会包含在内（体现在 `history` 中）。这保证了模型能理解对话的上下文。

   + 历史截断：添加了简单的逻辑来限制 `messages` 列表的最大长度，防止 API 调用因 Token 过多而失败或费用过高。

   + 调用 API （流式）：使用 `client.chat.completions.create` 并设置 `stream=True`。这与我们在 3.3 节学到的流式调用一致。
