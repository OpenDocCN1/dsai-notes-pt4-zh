class PIPLCompliance:
    def __init__(self):
        self.data_categories = {
            'sensitive': ['生物识别', '宗教信仰', '健康', '金融账户', '行踪轨迹'],
            'general': ['姓名', '电话', '电子邮件', '地址']
        }
    
    def classify_data(self, data_type):
        """分类个人信息"""
        if data_type in self.data_categories['sensitive']:
            return 'sensitive', '需要单独同意'
        elif data_type in self.data_categories['general']:
            return 'general', '需要明示同意'
        else:
            return 'unknown', '需要评估'
    
    def obtain_consent(self, user_id, data_types, purposes):
        """获取用户同意"""
        # PIPL 要求明示同意,敏感信息需要单独同意
        consent_requirements = []
        
        for data_type in data_types:
            category, requirement = self.classify_data(data_type)
            if category == 'sensitive':
                consent_requirements.append({
                    'data_type': data_type,
                    'requirement': '单独同意',
                    'consent_text': f"我同意处理我的{data_type}信息,用于{purposes}"
                })
        
        return consent_requirements
    
    def conduct_pia(self, processing_activity):
        """进行个人信息保护影响评估"""
        # PIPL 要求对以下情况进行评估:
        # 1. 处理敏感个人信息
        # 2. 利用个人信息进行自动化决策
        # 3. 委托处理、向他人提供、公开个人信息
        # 4. 向境外提供个人信息
        # 5. 其他对个人权益有重大影响的处理活动
        
        pia = {
            'processing_activity': processing_activity,
            'necessity_assessment': '必要性和比例原则评估',
            'risk_assessment': '对个人权益的影响评估',
            'security_measures': '安全保护措施',
            'conducted_at': datetime.utcnow()
        }
        
        return pia
    
    def handle_cross_border_transfer(self, data_type, destination_country):
        """处理跨境数据传输"""
        # PIPL 对跨境传输有严格要求
        requirements = []
        
        # 1. 通过国家网信部门安全评估
        requirements.append('安全评估')
        
        # 2. 经专业机构认证
        requirements.append('专业机构认证')
        
        # 3. 签订标准合同
        requirements.append('标准合同')
        
        # 4. 其他国家网信部门规定的条件
        requirements.append('其他条件')
        
        return {
            'data_type': data_type,
            'destination': destination_country,
            'requirements': requirements,
            'user_consent_required': True
        }

# 使用示例
pipl = PIPLCompliance()

# 分类数据

class DataClassification:
    def __init__(self):
        # 根据《数据安全法》进行分类分级
        self.classification_levels = {
            'public': {
                'level': 0,
                'description': '公开数据',
                'protection': '基本保护'
            },
            'internal': {
                'level': 1,
                'description': '内部数据',
                'protection': '内部访问控制'
            },
            'confidential': {
                'level': 2,
                'description': '机密数据',
                'protection': '加密存储和传输'
            },
            'secret': {
                'level': 3,
                'description': '秘密数据',
                'protection': '严格访问控制和审计'
            },
            'top_secret': {
                'level': 4,
                'description': '绝密数据',
                'protection': '最高级别保护措施'
            }
        }
    
    def classify_data(self, data_description):
        """对数据进行分类分级"""
        # 实际实现中,这里会使用更复杂的逻辑
        # 可能包括机器学习模型
        
        keywords = {
            'top_secret': ['国家秘密', '军事', '外交'],
            'secret': ['商业秘密', '核心技术', '重要客户'],
            'confidential': ['个人敏感信息', '财务数据', '内部文件'],
            'internal': ['员工信息', '内部通讯', '工作文档'],
            'public': ['公开发布', '营销材料', '公告']
        }
        
        for level, words in keywords.items():
            if any(word in data_description for word in words):
                return level, self.classification_levels[level]
        
        return 'internal', self.classification_levels['internal']
    
    def apply_protection(self, data, classification_level):
        """应用相应的保护措施"""
        level_info = self.classification_levels[classification_level]
        
        protection_measures = []
        
        if level_info['level'] >= 1:
            protection_measures.append('访问控制')
        
        if level_info['level'] >= 2:
            protection_measures.extend(['加密存储', '加密传输'])
        
        if level_info['level'] >= 3:
            protection_measures.extend(['审计日志', '多因素认证'])
        
        if level_info['level'] >= 4:
            protection_measures.extend(['物理隔离', '专人管理'])
        
        return protection_measures

# 使用示例
classifier = DataClassification()

# 分类数据
data_desc = "包含用户生物识别信息的数据库"
level, info = classifier.classify_data(data_desc)
print(f"分类级别: {level}")
print(f"描述: {info['description']}")

# 应用保护措施
measures = classifier.apply_protection(data_desc, level)
print(f"保护措施: {', '.join(measures)}")
