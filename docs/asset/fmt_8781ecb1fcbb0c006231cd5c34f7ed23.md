import json
from datetime import datetime
from typing import Dict, Any, List

class AgentBehaviorMonitor:
    """AI Agent 行为监控系统"""
    
    def __init__(self):
        self.action_log = []
        self.alert_threshold = {
            'file_access': 5,      # 5分钟内文件访问次数
            'api_calls': 10,       # 5分钟内 API 调用次数
            'privilege_escalation': 1,  # 任何权限提升尝试
        }
    
    def log_action(self, action_type: str, details: Dict[str, Any]) -> None:
        """记录 agent 动作"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'action_type': action_type,
            'details': details
        }
        self.action_log.append(log_entry)
        
        # 实时异常检测
        if self._detect_anomaly(action_type, details):
            self._trigger_alert(log_entry)
    
    def _detect_anomaly(self, action_type: str, details: Dict[str, Any]) -> bool:
        """检测异常行为"""
        # 检查权限提升尝试
        if action_type == 'command_execution':
            dangerous_commands = ['sudo', 'su', 'chmod +x', 'chown']
            command = details.get('command', '')
            if any(cmd in command for cmd in dangerous_commands):
                return True
        
        # 检查敏感文件访问
        if action_type == 'file_access':
            sensitive_paths = ['/etc/passwd', '/etc/shadow', '~/.ssh/', '~/.aws/']
            path = details.get('path', '')
            if any(sensitive in path for sensitive in sensitive_paths):
                return True
        
        # 检查频率异常
        recent_actions = self._get_recent_actions(minutes=5)
        action_count = sum(1 for a in recent_actions if a['action_type'] == action_type)
        threshold = self.alert_threshold.get(action_type, float('inf'))
        if action_count > threshold:
            return True
        
        return False
    
    def _get_recent_actions(self, minutes: int = 5) -> List[Dict]:
        """获取最近的动作"""
        from datetime import timedelta
        cutoff_time = datetime.now() - timedelta(minutes=minutes)
        return [
            action for action in self.action_log
            if datetime.fromisoformat(action['timestamp']) > cutoff_time
        ]
    
    def _trigger_alert(self, log_entry: Dict) -> None:
        """触发安全警报"""
        alert = {
            'severity': 'HIGH',
            'message': f"检测到异常 agent 行为: {log_entry['action_type']}",
            'details': log_entry,
            'timestamp': datetime.now().isoformat()
        }
        print(f"🚨 安全警报: {json.dumps(alert, indent=2, ensure_ascii=False)}")
        # 实际部署中应发送到 SIEM 系统或告警平台

# 使用示例

class OutputFilter:
    """输出内容过滤器"""
    
    SENSITIVE_PATTERNS = {
        'api_key': r'(api[_-]?key|apikey)["\']?\s*[:=]\s*["\']?([a-zA-Z0-9_\-]{20,})',
        'aws_key': r'AKIA[0-9A-Z]{16}',
        'private_key': r'-----BEGIN (RSA|DSA|EC) PRIVATE KEY-----',
        'password': r'(password|passwd|pwd)["\']?\s*[:=]\s*["\']?([^\s"\']{8,})',
        'email': r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
        'phone': r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b',
        'ssn': r'\b\d{3}-\d{2}-\d{4}\b',
        'credit_card': r'\b\d{4}[- ]?\d{4}[- ]?\d{4}[- ]?\d{4}\b',
    }
    
    @classmethod
    def filter_sensitive_data(cls, output: str, redact: bool = True) -> Tuple[str, List[str]]:
        """
        过滤输出中的敏感数据
        
        Args:
            output: 原始输出
            redact: 是否编辑敏感信息(True)或仅检测(False)
        
        Returns:
            (filtered_output, detected_types): 过滤后的输出和检测到的敏感类型
        """
        detected_types = []
        filtered_output = output
        
        for data_type, pattern in cls.SENSITIVE_PATTERNS.items():
            matches = re.findall(pattern, filtered_output, re.IGNORECASE)
            if matches:
                detected_types.append(data_type)
                if redact:
                    filtered_output = re.sub(
                        pattern,
                        f'[已编辑:{data_type.upper()}]',
                        filtered_output,
                        flags=re.IGNORECASE
                    )
        
        return filtered_output, detected_types
