
```
import os
import json
from openai import OpenAI, APIError

# --- (复用之前的 API Key 获取和 client 初始化代码) ---
api_key = os.getenv("AOAI_API_KEY")
if not api_key:
    raise ValueError("请设置环境变量 AOAI_API_KEY")

client = OpenAI(api_key=api_key, base_url="https://api.huaqloud.com/v1")

# --- 步骤 1: 定义我们的“工具”（模拟的天气查询函数） ---
def get_current_weather(location, unit="celsius"):
    """ (模拟函数) 获取指定地点的当前天气 """
    print(f"--- 模拟工具执行：正在查询 {location} 的天气 ({unit}) ---")
    # 在真实应用中，这里会调用天气API
    if "北京" in location:
        return json.dumps({"location": location, "temperature": "2", "unit": unit, "forecast": "晴朗"})
    elif "东京" in location:
        return json.dumps({"location": location, "temperature": "8", "unit": unit, "forecast": "多云"})
    elif "伦敦" in location:
        return json.dumps({"location": location, "temperature": "5", "unit": unit, "forecast": "小雨"})
    else:
        return json.dumps({"location": location, "temperature": "未知", "unit": unit, "forecast": "未知"})

# --- 步骤 1 (续): 定义工具的描述，供模型理解 ---
tools = [
    {
        "type": "function",
        "function": {
            "name": "get_current_weather",
            "description": "获取指定地点的当前天气信息。如果用户没有指定单位，默认为摄氏度。",
            "parameters": {
                "type": "object",
                "properties": {
                    "location": {
                        "type": "string",
                        "description": "需要查询天气的城市或地点，例如：北京, 东京",
                    },
                    "unit": {
                        "type": "string",
                        "enum": ["celsius", "fahrenheit"],
                        "description": "温度单位，摄氏度(celsius)或华氏度(fahrenheit)"
                    },
                },
                "required": ["location"], # location 参数是必需的
            },
        }
    }
]

print("\n\n--- 场景六：使用工具 (Function Calling / Tool Use) ---")

# --- 初始对话消息 ---
messages = [{"role": "user", "content": "你好，请问现在伦敦的天气怎么样？"}]
print(f"用户: {messages[0]['content']}")

try:
    # --- 步骤 2: 发起第一次 API 请求，包含工具定义 ---
    print("\n--- 第一次 API 调用 (提供工具) ---")
    response = client.chat.completions.create(
        model="gpt-4o-mini", # 确保模型支持 Tool Use
        messages=messages,
        tools=tools,
        tool_choice="auto",  # 让模型自动决定是否使用工具
    )

    response_message = response.choices[0].message
    # print("第一次响应的完整消息:", response_message) # 调试用

    # --- 步骤 3 & 4: 检查模型是否请求调用工具，并执行 ---
    tool_calls = response_message.tool_calls
    if tool_calls:
        print("\n--- 模型请求调用工具 ---")
        # 将模型的请求添加到消息历史中 (必须！)
        messages.append(response_message)

        # 在实际应用中，可能有多个 tool_calls，需要遍历处理
        for tool_call in tool_calls:
            function_name = tool_call.function.name
            function_args = json.loads(tool_call.function.arguments) # 解析参数字符串为字典
            tool_call_id = tool_call.id # 获取 tool_call_id

            print(f"  工具名称: {function_name}")
            print(f"  工具参数: {function_args}")
            print(f"  工具调用 ID: {tool_call_id}")

            # --- 执行我们定义的函数 ---
            if function_name == "get_current_weather":
                function_response = get_current_weather(
                    location=function_args.get("location"),
                    unit=function_args.get("unit", "celsius") # 处理可选参数
                )
                print(f"--- 工具执行结果: {function_response} ---")

                # --- 步骤 5: 将工具执行结果添加到消息历史中 ---
                messages.append(
                    {
                        "tool_call_id": tool_call_id,
                        "role": "tool",
                        "content": function_response, # 函数返回的结果
                    }
                )
            else:
                print(f"警告：模型请求调用未知的工具 '{function_name}'")
                # 可以选择返回一个错误信息给模型
                messages.append(
                    {
                        "tool_call_id": tool_call_id,
                        "role": "tool",
                        "content": json.dumps({"error": f"Unknown tool: {function_name}"}),
                    }
                )

        # --- 步骤 5 (续): 发起第二次 API 请求，包含工具结果 ---
        print("\n--- 第二次 API 调用 (提供工具结果) ---")
        # print("发送给API的消息历史:", messages) # 调试用
        second_response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            # 注意：第二次调用通常不需要再传 tools 或 tool_choice
        )

        # --- 步骤 6: 获取并打印最终的自然语言回复 ---
        final_response = second_response.choices[0].message.content
        print(f"\n模型最终回复:\n{final_response}")

    else:
        # --- 如果模型没有调用工具，直接打印它的回复 ---
        print("\n--- 模型未调用工具，直接回复 ---")
        final_response = response_message.content
        print(f"模型回复:\n{final_response}")

except APIError as e:
    print(f"\nAPI 调用出错: {e.status_code} - {e.message}")
except json.JSONDecodeError as e:
    print(f"\n错误：解析模型返回的参数时出错: {e}")
    print(f"原始参数字符串: {tool_call.function.arguments if 'tool_call' in locals() else '未获取'}")
except Exception as e:
    print(f"\n发生意外错误: {e}")
```

## 代码解释

### `get_current_weather` 函数
这是一个模拟函数。关键在于它接收参数（`location`、`unit`）并返回一个 JSON 字符串作为结果，模拟真实 API 的行为。

### `tools` 列表
定义了 `get_current_weather` 工具。

- `description` 清晰地说明了它的用途和默认行为。
- `parameters` 使用 JSON Schema 定义了 `location`（必需）和 `unit`（可选，枚举值）参数。

### 第一次 API 调用
发送用户消息和 `tools` 定义，`tool_choice="auto"`。

### 检查 `tool_calls`
判断 `response_message.tool_calls` 是否存在。如果存在，说明模型想用工具。

### 处理 `tool_calls`
- 将包含 `tool_calls` 的 `response_message` 添加到 `messages` 历史记录中。
- 遍历 `tool_calls`（虽然本例只有一个）。
- 提取函数名、参数（`json.loads` 解析）和 `tool_call_id`。
- 调用本地的 `get_current_weather` 函数。
- 将结果封装成 `role: tool` 的消息，包含对应的 `tool_call_id` 和 `content`（函数返回值），并添加到 `messages` 历史。

### 第二次 API 调用
将更新后的 `messages`（包含用户问题、模型工具请求、工具执行结果）再次发送给 API。这次不需要 `tools` 或 `tool_choice`。

### 最终回复
第二次调用的响应 `content` 就是模型结合了工具结果后的自然语言回答。

### 错误处理
增加了对 `APIError` 错误和 `JSONDecodeError` 错误的处理。

## 工具使用的主要应用场景

- **实时信息查询：** 天气、股票、新闻、体育比分等。
- **外部服务交互：** 发送邮件、创建日历事件、控制智能家居设备、在电商平台下单。
- **数据库/知识库查询：** 从结构化数据库或内部知识库中检索信息。
- **复杂计算：** 执行精确的数学计算或调用专门的计算库。
- **代码执行：** （需极度谨慎）在沙箱环境中执行代码片段。
- **多模态交互：** 调用图像生成或分析工具。

## 注意事项

- **成本与延迟：** Function Calling 通常需要至少两次 API 调用，会增加成本和响应时间。
- **工具描述是关键：** 模型能否在合适的时机选择正确的工具，很大程度上取决于您提供的 `description` 是否清晰、准确、无歧义。
- **参数 Schema：** 准确定义 `parameters` 的 JSON Schema 很重要，有助于模型生成格式正确的参数。
- **执行函数的健壮性：** 您自己实现的工具函数需要有良好的错误处理机制。如果工具执行失败，应将错误信息作为结果返回给模型，模型可能会据此告知用户。
- **安全性：** 绝对不要让模型能够调用可以执行任意代码或访问敏感系统的函数，除非有严格的沙箱和权限控制。要仔细审查模型生成的参数，防止注入攻击。
- **并发工具调用：** 模型可能一次请求调用多个工具（`tool_calls` 列表包含多个元素）。您的代码需要能处理这种情况，可以串行执行，也可以并行执行（如果工具之间无依赖）。
