class BudgetController:
    def __init__(self, daily_budget=10.0):
        self.daily_budget = daily_budget
        self.daily_spend = 0
        self.request_count = 0
    
    def check_budget(self, estimated_cost):
        """检查是否超出预算"""
        if self.daily_spend + estimated_cost > self.daily_budget:
            raise Exception(f"Daily budget exceeded: ${self.daily_spend:.2f} / ${self.daily_budget}")
        return True
    
    def estimate_cost(self, input_tokens, output_tokens, model):
        """估算请求成本"""
        pricing = {
            "haiku": {"input": 1, "output": 5},
            "sonnet": {"input": 3, "output": 15},
            "opus": {"input": 5, "output": 25},
        }
        
        model_type = "haiku" if "haiku" in model else ("sonnet" if "sonnet" in model else "opus")
        cost = (input_tokens / 1_000_000 * pricing[model_type]["input"]) + \
               (output_tokens / 1_000_000 * pricing[model_type]["output"])
        return cost
    
    def log_request(self, input_tokens, output_tokens, model):
        """记录请求并更新预算"""
        cost = self.estimate_cost(input_tokens, output_tokens, model)
        self.daily_spend += cost
        self.request_count += 1
        
        print(f"Request #{self.request_count}: ${cost:.4f} | Daily total: ${self.daily_spend:.2f} / ${self.daily_budget}")

# 使用示例

## 5.4 多模型混合使用

在实际应用中，单一模型往往无法满足所有需求。多模型混合使用策略可以在性能、成本和速度之间实现最优平衡，让每个模型发挥其最大优势。

### 混合使用的核心理念

“用最合适的工具做最合适的事” —— 根据任务特性动态选择模型，而不是一刀切地使用单一模型。

class LayeredModelRouter:
    def __init__(self):
        self.models = {
            "simple": "claude-haiku-4-5",
            "medium": "claude-sonnet-4-6",
            "complex": "claude-opus-4-6"
        }
    
    def classify_complexity(self, task):
        """分析任务复杂度"""
        indicators = {
            "simple": [
                "是什么", "定义", "简单解释", "yes/no",
                len(task) < 100,  # 短问题通常简单
            ],
            "complex": [
                "抽象推理", "多步骤推理", "跨领域知识",
                "为什么", "如何", len(task) > 500,
            ]
        }
        
        # 简单规则分类
        if any(keyword in task for keyword in ["抽象推理", "多步骤推理"]):
            return "complex"
        elif any(keyword in task for keyword in ["是什么", "定义"]):
            return "simple"
        else:
            return "medium"
    
    def route_request(self, task):
        """路由到合适的模型"""
        complexity = self.classify_complexity(task)
        model = self.models[complexity]
        
        print(f"Task complexity: {complexity}, using model: {model}")
        
        return self.call_claude(model, task)
    
    def call_claude(self, model, task):
        return client.messages.create(
            model=model,
            max_tokens=1024,
            messages=[{"role": "user", "content": task}]
        )

### 模式 2: 两阶段处理

第一阶段用快速模型生成初稿，第二阶段用高级模型精炼：

```
class TwoStageProcessor:
    def __init__(self):
        self.draft_model = "claude-haiku-4-5"
        self.refine_model = "claude-sonnet-4-6"
    
    def generate_content(self, task, require_refinement=True):
        """两阶段内容生成"""
        
        # 阶段 1：快速生成草稿（Haiku）
        print("Stage 1: Generating draft with Haiku...")
        draft = client.messages.create(
            model=self.draft_model,
            max_tokens=1024,
            messages=[{"role": "user", "content": task}]
        )
        
        draft_content = draft.content[0].text
        
        if not require_refinement:
            return draft_content
        
        # 阶段 2：精炼和改进（Sonnet）
        print("Stage 2: Refining with Sonnet...")
        refined = client.messages.create(
            model=self.refine_model,
            max_tokens=1024,
            messages=[
                {
                    "role": "user",
                    "content": f"""以下是一个初稿，请改进它：

初稿：
{draft_content}

请改进：
1. 提高表达的准确性和专业性
2. 补充遗漏的重要信息
3. 优化结构和逻辑
4. 修正任何错误

直接输出改进后的版本，不要解释。"""
                }
            ]
        )
        
        return refined.content[0].text

# 使用示例
processor = TwoStageProcessor()

# 对于重要内容，使用两阶段处理
article = processor.generate_content(
    "写一篇关于量子计算的科普文章",
    require_refinement=True
)

# 对于一般内容，只用第一阶段
quick_summary = processor.generate_content(
    "总结这篇新闻",
    require_refinement=False
)
```

**成本分析**： 两阶段处理的成本约为直接使用 Sonnet 的 60-70%，但质量可能更好（因为有两次迭代）。
