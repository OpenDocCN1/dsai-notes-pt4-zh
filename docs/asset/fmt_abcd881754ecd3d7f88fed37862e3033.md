```python
import re
from collections import Counter
from datetime import datetime, timedelta

class LogAnalyzer:
    def __init__(self, log_file="security_audit.log"):
        self.log_file = log_file
    
    def detect_brute_force(self, threshold=5, time_window=300):
        """检测暴力破解攻击"""
        failed_attempts = {}
        alerts = []
        
        with open(self.log_file, 'r') as f:
            for line in f:
                try:
                    log_entry = json.loads(line)
                    
                    if (log_entry['event_type'] == 'authentication' and 
                        log_entry['status'] == 'failure'):
                        
                        user_id = log_entry['user_id']
                        timestamp = datetime.fromisoformat(log_entry['timestamp'])
                        
                        if user_id not in failed_attempts:
                            failed_attempts[user_id] = []
                        
                        failed_attempts[user_id].append(timestamp)
                        
                        # 检查时间窗口内的失败次数
                        recent_failures = [
                            t for t in failed_attempts[user_id]
                            if (timestamp - t).total_seconds() <= time_window
                        ]
                        
                        if len(recent_failures) >= threshold:
                            alerts.append({
                                'type': 'brute_force',
                                'user_id': user_id,
                                'attempts': len(recent_failures),
                                'time_window': time_window
                            })
                
                except json.JSONDecodeError:
                    continue
        
        return alerts
    
    def detect_privilege_escalation(self):
        """检测权限提升尝试"""
        alerts = []
        
        with open(self.log_file, 'r') as f:
            for line in f:
                try:
                    log_entry = json.loads(line)
                    
                    if (log_entry['event_type'] == 'authorization' and 
                        log_entry['status'] == 'denied'):
                        
                        details = log_entry['details']
                        if details['action'] in ['admin', 'sudo', 'manage_users']:
                            alerts.append({
                                'type': 'privilege_escalation_attempt',
                                'user_id': log_entry['user_id'],
                                'resource': details['resource'],
                                'action': details['action']
                            })
                
                except json.JSONDecodeError:
                    continue
        
        return alerts
    
    def generate_report(self, days=7):
        """生成安全报告"""
        cutoff_date = datetime.utcnow() - timedelta(days=days)
        
        event_counts = Counter()
        user_activity = Counter()
        failed_auths = 0
        
        with open(self.log_file, 'r') as f:
            for line in f:
                try:
                    log_entry = json.loads(line)
                    timestamp = datetime.fromisoformat(log_entry['timestamp'])
                    
                    if timestamp < cutoff_date:
                        continue
                    
                    event_counts[log_entry['event_type']] += 1
                    user_activity[log_entry['user_id']] += 1
                    
                    if (log_entry['event_type'] == 'authentication' and 
                        log_entry['status'] == 'failure'):
                        failed_auths += 1
                
                except json.JSONDecodeError:
                    continue
        
        report = {
            'period': f'Last {days} days',
            'total_events': sum(event_counts.values()),
            'event_breakdown': dict(event_counts),
            'top_users': user_activity.most_common(10),
            'failed_authentications': failed_auths,
            'alerts': {
                'brute_force': self.detect_brute_force(),
                'privilege_escalation': self.detect_privilege_escalation()
            }
        }
        
        return report

# 使用示例
analyzer = LogAnalyzer()

# 检测暴力破解
brute_force_alerts = analyzer.detect_brute_force()
if brute_force_alerts:
    print("⚠️ 检测到暴力破解攻击:")
    for alert in brute_force_alerts:
        print(f"  用户: {alert['user_id']}, 尝试次数: {alert['attempts']}")

# 生成报告
report = analyzer.generate_report(days=7)
print(json.dumps(report, indent=2, ensure_ascii=False))
```

```python
import boto3
import json

class AWSSecretsManager:
    def __init__(self, region_name='us-east-1'):
        self.client = boto3.client('secretsmanager', region_name=region_name)
    
    def store_secret(self, secret_name, secret_value):
        """存储密钥"""
        try:
            self.client.create_secret(
                Name=secret_name,
                SecretString=json.dumps(secret_value)
            )
        except self.client.exceptions.ResourceExistsException:
            # 密钥已存在,更新它
            self.client.update_secret(
                SecretId=secret_name,
                SecretString=json.dumps(secret_value)
            )
    
    def retrieve_secret(self, secret_name):
        """检索密钥"""
        response = self.client.get_secret_value(SecretId=secret_name)
        return json.loads(response['SecretString'])
    
    def delete_secret(self, secret_name):
        """删除密钥"""
        self.client.delete_secret(
            SecretId=secret_name,
            ForceDeleteWithoutRecovery=True
        )

# 使用示例

    'openclaw/api-keys',
    {
        'openai_key': 'sk-xxx',
        'anthropic_key': 'sk-ant-xxx'
    }
)

# 检索密钥
```
