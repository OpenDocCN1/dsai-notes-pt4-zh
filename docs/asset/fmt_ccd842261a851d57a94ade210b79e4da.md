```python
def load_context():
    """会话开始时自动加载记忆"""

    # 加载隐性知识
    memory = read_file(".claude/memory/MEMORY.md")

    # 加载昨日的笔记
    yesterday = get_yesterday()
    daily_note = read_file(f".claude/memory/memory/{yesterday}.md")

    # 加载相关主题的知识图谱
    topics = scan_topics()
    relevant = find_relevant_topics(topics)

    return {
        "memory": memory,
        "daily_note": daily_note,
        "topics": relevant
    }
```
