    # 构建Docker镜像
    subprocess.run([
        "docker", "build", "-t", "customer-service-agent:latest", "."
    ], check=True)
    
    # 推送到华为云容器镜像服务
    subprocess.run([

    ], check=True)
    
    subprocess.run([
        "docker", "push",
