## 第一个程序：字符串输出与中断21h服务

```
;----------请把以下代码保存到src\main.asm------------------------------
;----------选中main.sh及src文件夹->右键->压缩成submit.zip提交----------
.386
data segment use16
sth db 10 dup(0)
s db 10 dup('S'), 0; 测评时数组s的长度及内容和此样本不会一样,
                   ; 但s一定是一个符合C语言标准的字符串
data ends

code segment use16
assume cs:code, ds:data
main:
   mov ax, data
   mov ds, ax
;在#1_begin和#1_end之间补充代码实现puts(s)的效果,即除了输出字符串s本身内容外,还需要再输出0Dh及0Ah
;#1_begin--------------------------------------
   lea si, s
loop1:
   mov dl, [si]
   cmp dl, 0
   je loopend
   mov ah, 2
   int 21h
   inc si
   jmp loop1

loopend:
   mov dl, 0Dh
   mov ah, 2
   int 21h
   mov dl, 0Ah
   mov ah , 2
   int 21h
;#1_end----------------------------------------
   mov ah, 4Ch
   int 21h
code ends
end main
;----------请把以上代码保存到src\main.asm------------------------------
```

`int21h/AH=09h`用于输出一个`$`结尾的字符串但不输出`$`本身

中断`21h`里的`9`号服务是显示字符`串`，显示`DS:DX`指向的字符串，直到遇到`$`

1. `mov ah, 9`：把`9`这个值移动到`ah`寄存器。在DOS中断`21h`的服务中，`9`号服务是用于显示字符串的。它会显示`DS:DX`指向的字符串，直到遇到`$`字符为止。

1. `mov dx, offset s`：把`s`的地址移动到`dx`寄存器。`s`是一个字符串变量，在这里，我们把它的地址存入`dx`寄存器，以便在下一步使用DOS服务来显示它。

1. `int 21h`：调用DOS中断`21h`。由于`ah`寄存器设置为`9`，所以这个调用会打印出`DS:DX`指向的字符串，直到遇到`$`字符为止。

## 计算1-100和

1. 函数传参

1. 地址

1. `loop`

```
;本题要求:
;完成3项程序填空,调用函数f()求1+2+3+...+n之和并保存到变量sum中,
;注意sum不会超过0FFFFh,故在计算过程中不需要考虑加法溢出
;==========请把在完成填空后把本文件保存到src\main.asm==================
;==========选中main.sh及src文件夹->右键->压缩成submit.zip提交==========
.386
data segment use16
sth db 10h dup(0)
n dw 3
sum dw 0
data ends

code segment use16
assume cs:code, ds:data
;函数f()原型:
;   short int f(short int n, short int *psum);
;函数f()功能:
;   计算1+2+3+...n的和, 并把和保存到*psum中。例如当n=3时, *psum=6。
;提示:
;   f()的参数通过堆栈传递, 其中n就是[bp+4], psum就是[bp+6]。
;注意:
;   在函数f()中不可直接引用data段中的变量n及sum, 否则会编译失败
f proc near
   push bp
   mov bp, sp
   mov bx, 0
   mov cx, 0
;#1_begin--------
   mov cx, [bp+4]
;#1_end----------
   mov bx, [bp+6]
   xor ax, ax
next:
   add ax, cx
   loop next
;#2_begin------
   mov [bx], ax
;#2_end--------
   pop bp
   ret
f endp

main:
   mov ax, data
   mov ds, ax
   mov ax, offset sum; ax = sum的偏移地址
;#3_begin------------
   push ax
;#3_end--------------
   push [n]     ; 传递n的值给函数f()
   call f
   add sp, 4    ; 清除堆栈上的2个参数
   mov ax, [sum]; 跟踪时请检查当n=3时,ax应该等于6
   mov ah, 4Ch
   int 21h
code ends
end main
;==========请在完成填空后把本文件保存到src\main.asm================
```

## 16位符号数n转化成十进制字符串，并填充`$`

1. 有符号，注意负数

1. `Neg`

1. 负数先加上 `-`，再到正数

1. `loop`

```
;==========请把以下代码保存到src\main.asm==============================
;==========选中main.sh及src文件夹->右键->压缩成submit.zip提交==========
.386
data segment use16
sth db 10h dup(0)
n dw 0FFF0h   
s db 10h dup(0)
data ends

code segment use16
assume cs:code, ds:data
;void f(short int n, char s[])
;注意f()的参数通过堆栈传递,其中n就是[bp+4], s就是[bp+6]。
;在函数f()中不可直接引用data段中的变量n及数组s, 否则会产生编译错误。
f proc near
   push bp
   mov bp, sp
;请在#1_begin和#1_end之间补充代码实现以下功能:
;把16位符号数n转化成十进制字符串保存到数组s中,并在字符串后面填充'$'
;例如,当n=0FFF0h时, s的内容为: '-', '1', '6', '$'
;#1_begin-----------=========------------------
   mov ax, [bp+4]         ; 加载n到ax
   mov bx, [bp+6]         ; 加载s的地址到bx
   xor cx, cx
   
   ; 检查n是否为负数
   test ax, ax
   jns positive_number
   
   ; 是负数, 在字符串开头添加'-';
   mov byte ptr [bx], '-';
   neg ax                 ; 取绝对值
   inc bx                 ; 移动s的地址
   
positive_number:
   ; 转换数字到字符串
   mov si, 10             ; 设置除数为10
.reverse_loop:
   xor dx, dx             ; 清除dx
   div si    ; ax = ax / 10, dx = ax % 10
   push dx                ; 将余数压入栈中
   inc cx
   test ax, ax            ; 检查商是否为0
   jnz .reverse_loop
   
   ; 将数字从栈中弹出并存储为字符
.store_loop:
   pop dx                 ; 弹出数字
   add dx, '0'            ; 转换为字符
   mov byte ptr [bx], dl  ; 存储字符
   inc bx                 ; 移动到下一个字符位置
   loop .store_loop
   
   ; 在字符串后添加 '$';
   mov byte ptr [bx], '$';

;#1_end-------------=========------------------
   pop bp
   ret
f endp

main:
   mov ax, data
   mov ds, ax
   mov ax, offset s
   push ax
   push [n]
   call f
   add sp, 4
   mov ah, 9
   mov dx, offset s
   int 21h
   mov ah, 4Ch
   int 21h
code ends
end main
;==========请把以上代码保存到src\main.asm==============================
```

## `sqrt`开方+转换成10进制输出

+ 注意浮点数

+ 转换成10进制和上面类似，运用到`pop`，`push`

```
.386
code segment use16
assume cs:code
;Input: EAX
;Output: EAX = sqrt(EAX)
sqrt proc near
push bp
mov bp, sp
sub sp, 4             ; 创建一个32位的动态变量
mov [bp-4], eax       ; 把EAX保存到该动态变量中
fild dword ptr [bp-4] ; 把[bp-4]中的整数转化成小数载入到st(0)
fsqrt                 ; 对st(0)开方结果仍旧保存到st(0)
fistp dword ptr [bp-4]; 把st(0)中的值转化成整数并保存到[bp-4]中再弹出st(0)
mov eax, [bp-4]       ; EAX=函数的返回值
mov sp, bp
pop bp
ret
sqrt endp

;Input: EAX
;Output: output EAX to stdout in decimal format
val2dec proc near
   push eax
   push ebx
   push ecx
   push edx
   mov cx, 0
div_again:
   xor edx, edx
   mov ebx, 10
   div ebx
   add dl, '0';
   push dx
   inc cx
   or eax, eax
   jnz div_again
pop_again:
   pop dx
   mov ah, 2
   int 21h
   loop pop_again
   pop edx
   pop ecx
   pop ebx
   pop eax
   ret
val2dec endp

main:
   mov eax, 2147483647
   call sqrt          ; 用EAX传递参数给函数sqrt, 函数返回值也在EAX中
   call val2dec       ; 用EAX传递参数给函数val2dec, 该函数输出"46341"
   mov ah, 4Ch
   int 21h
code ends
end main
```

## 把32位的整数转成16进制输出

+ `Rol`

+ `and`会改变

+ 用`int 21h`的`2`号功能输出

```
.386
data segment use16
abc dd 2147483647
data ends

code segment use16
assume cs:code, ds:data
main:
   mov ax, seg abc
   mov ds, ax
   mov eax, abc
   mov cx, 8
again:
   rol eax, 4
   push eax
   and eax, 0Fh
   cmp al, 10
   jb is_digit
is_alpha:
   sub al, 10
   add al, 'A';
   jmp finish_4bits
is_digit:
   add al, '0';
finish_4bits:
   mov ah, 2
   mov dl, al
   int 21h
   pop eax
   sub cx, 1
   jnz again
   mov ah, 4Ch
   int 21h
code ends
end main
```

## 用变量传参+`strlen`+转换成16进制

+ `hex`部分和上面类似

```
.386
data segment use16
s db "Hello,world!", 0Dh, 0Ah, 00h
p dw 0
n dw 0
data ends

code segment use16
assume cs:code, ds:data
;Input: [p]->string
;Output: [n]=string's length
strlen proc near
   push ax
   push cx
   push si
   xor cx, cx
   mov si, [p]          ; p是函数strlen()的参数
strlen_next_char:
   lodsb
   or al, al
   jz strlen_done
   inc cx
   jmp strlen_next_char
strlen_done:
   mov [n], cx          ; n是函数strlen()的返回值
   pop si
   pop cx
   pop ax
   ret
strlen endp

;Input: [n]
;Output: Output n in hex to stdout
hex proc near
   push ax
   push cx
   push dx
   mov dx, [n]          ; n是函数hex()的参数
   mov cx, 4
hex_next_4_bits:
   rol dx, 4
   push dx
   and dl, 0Fh
   cmp dl, 10
   jae is_alpha
is_digit:
   add dl, '0';
   jmp hex_output
is_alpha:
   sub dl, 10-'A';       ; DL = DL - (10 - 'A') = DL - 10 + 'A';
hex_output:
   mov ah, 2
   int 21h
   pop dx
   loop hex_next_4_bits
   pop dx
   pop cx
   pop ax
   ret
hex endp

main:
   mov ax, data
   mov ds, ax
   cld
   mov [p], offset s    ; 用变量p传递s的偏移地址给函数strlen()
   call strlen          ; strlen()的返回值在变量n中
   call hex             ; 用变量n传递参数给函数hex(), hex()输出000E
   mov ah, 4Ch
   int 21h
code ends
end main
```

## 16位整数转换成二进制

+ `shl`每次左移一个

+ 巧用`adc`

```
data segment
abc dw 32767
data ends
code segment
assume cs:code, ds:data
main:
   mov ax, data
   mov ds, ax
   mov ax, [abc]
   mov cx, 16
again:
   shl ax, 1
;  mov dl, '0';
;  adc dl, 0
   jc is_one
is_zero:
   mov dl, '0';
   jmp output
is_one:
   mov dl, '1';
output:
   push ax
   mov ah, 2
   int 21h
   pop ax
   sub cx, 1
   jnz again
   mov ah, 4Ch
   int 21h
code ends
end main
```
