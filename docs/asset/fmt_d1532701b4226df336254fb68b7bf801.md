
### 14.4.2 内容溯源和审计

**生成内容元数据追踪：**

```
from enum import Enum
from dataclasses import dataclass
from typing import Optional, Callable
import hashlib
from datetime import datetime

class RiskLevel(Enum):
    """风险级别"""
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4

class ReviewStatus(Enum):
    """审查状态"""
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"
    NEEDS_REVISION = "needs_revision"

@dataclass
class GeneratedContent:
    """生成的内容"""
    content: str
    content_type: str  # 'marketing', 'legal', 'medical', 'general'
    model: str
    prompt: str
    timestamp: datetime
    risk_level: RiskLevel
    review_status: ReviewStatus = ReviewStatus.PENDING
    reviewer: Optional[str] = None
    review_notes: Optional[str] = None
    content_hash: Optional[str] = None
    
    def __post_init__(self):
        """生成内容哈希"""
        if not self.content_hash:
            self.content_hash = hashlib.sha256(
                self.content.encode()
            ).hexdigest()[:16]

class ContentReviewWorkflow:
    """内容审查工作流"""
    
    # 内容类型风险映射
    RISK_MAPPING = {
        'legal': RiskLevel.CRITICAL,      # 法律内容 - 必须专家审查
        'medical': RiskLevel.CRITICAL,    # 医疗内容 - 必须专家审查
        'financial': RiskLevel.HIGH,      # 金融建议 - 专家审查
        'marketing': RiskLevel.MEDIUM,    # 营销内容 - 编辑审查
        'internal_doc': RiskLevel.MEDIUM, # 内部文档 - 编辑审查
        'general': RiskLevel.LOW,         # 一般内容 - 轻度审查
        'social_media': RiskLevel.LOW,    # 社交媒体 - 轻度审查
    }
    
    def __init__(self):
        self.content_registry = []
        self.review_callbacks = {
            RiskLevel.CRITICAL: self._expert_review,
            RiskLevel.HIGH: self._expert_review,
            RiskLevel.MEDIUM: self._editor_review,
            RiskLevel.LOW: self._automated_review,
        }
    
    def submit_content(
        self,
        content: str,
        content_type: str,
        model: str,
        prompt: str
    ) -> GeneratedContent:
        """提交生成的内容进行审查"""
        
        # 确定风险级别
        risk_level = self.RISK_MAPPING.get(content_type, RiskLevel.MEDIUM)
        
        # 创建内容对象
        gen_content = GeneratedContent(
            content=content,
            content_type=content_type,
            model=model,
            prompt=prompt,
            timestamp=datetime.now(),
            risk_level=risk_level
        )
        
        # 注册内容
        self.content_registry.append(gen_content)
        
        # 根据风险级别路由到相应的审查流程
        review_func = self.review_callbacks[risk_level]
        review_func(gen_content)
        
        return gen_content
    
    def _automated_review(self, content: GeneratedContent) -> None:
        """自动审查(低风险内容)"""
        print(f"[自动审查] 内容 {content.content_hash}")
        
        # 运行自动检查
        checks = {
            'toxicity': self._check_toxicity(content.content),
            'factuality': self._check_factuality(content.content),
            'bias': self._check_bias(content.content),
        }
        
        # 如果所有检查通过,自动批准
        if all(checks.values()):
            content.review_status = ReviewStatus.APPROVED
            content.reviewer = "automated_system"
            content.review_notes = "通过所有自动检查"
            print(f"  ✅ 自动批准")
        else:
            # 升级到人工审查
            content.risk_level = RiskLevel.MEDIUM
            self._editor_review(content)
    
    def _editor_review(self, content: GeneratedContent) -> None:
        """编辑审查(中等风险内容)"""
        print(f"[编辑审查] 内容 {content.content_hash} 已加入编辑审查队列")
        print(f"  内容类型: {content.content_type}")
        print(f"  风险级别: {content.risk_level.name}")
        print(f"  需要编辑在 24 小时内审查")
        
        # 实际实现中会发送通知给编辑团队
        # 这里模拟审查
        content.review_notes = "等待编辑审查"
    
    def _expert_review(self, content: GeneratedContent) -> None:
        """专家审查(高风险/关键内容)"""
        print(f"[专家审查] 内容 {content.content_hash} 已加入专家审查队列")
        print(f"  ⚠️ 内容类型: {content.content_type}")
        print(f"  ⚠️ 风险级别: {content.risk_level.name}")
        print(f"  需要领域专家在 48 小时内审查")
        print(f"  在专家批准前,内容不得发布")
        
        # 实际实现中会发送紧急通知给专家
        content.review_notes = "等待专家审查 - 高优先级"
    
    def _check_toxicity(self, content: str) -> bool:
        """检查毒性"""
        # 实际实现中调用毒性检测 API
        return True
    
    def _check_factuality(self, content: str) -> bool:
        """检查事实性"""
        # 实际实现中调用事实核查系统
        return True
    
    def _check_bias(self, content: str) -> bool:
        """检查偏见"""
        # 实际实现中调用偏见检测器
        return True
    
    def approve_content(
        self,
        content_hash: str,
        reviewer: str,
        notes: Optional[str] = None
    ) -> bool:
        """批准内容"""
        content = self._find_content(content_hash)
        if content:
            content.review_status = ReviewStatus.APPROVED
            content.reviewer = reviewer
            content.review_notes = notes or "已批准发布"
            print(f"✅ 内容 {content_hash} 已由 {reviewer} 批准")
            return True
        return False
    
    def reject_content(
        self,
        content_hash: str,
        reviewer: str,
        reason: str
    ) -> bool:
        """拒绝内容"""
        content = self._find_content(content_hash)
        if content:
            content.review_status = ReviewStatus.REJECTED
            content.reviewer = reviewer
            content.review_notes = f"拒绝原因: {reason}"
            print(f"❌ 内容 {content_hash} 已由 {reviewer} 拒绝")
            return True
        return False
    
    def _find_content(self, content_hash: str) -> Optional[GeneratedContent]:
        """查找内容"""
        for content in self.content_registry:
            if content.content_hash == content_hash:
                return content
        return None
    
    def get_pending_reviews(self, risk_level: Optional[RiskLevel] = None) -> List[GeneratedContent]:
        """获取待审查内容"""
        pending = [
            c for c in self.content_registry
            if c.review_status == ReviewStatus.PENDING
        ]
        
        if risk_level:
            pending = [c for c in pending if c.risk_level == risk_level]
        
        return pending

# 使用示例
workflow = ContentReviewWorkflow()

# 提交不同类型的内容
contents = [
    ("这是一条营销推文...", "marketing", "claude-4-sonnet", "写一条产品推文"),
    ("患者应服用阿司匹林...", "medical", "gpt-4", "提供医疗建议"),
    ("根据合同第5条...", "legal", "claude-4-sonnet", "解释法律条款"),
    ("今天天气不错", "general", "gpt-3.5-turbo", "闲聊"),
]

for content, content_type, model, prompt in contents:
    gen_content = workflow.submit_content(content, content_type, model, prompt)
    print()

# 查看待审查内容
print("\n=== 待审查内容摘要 ===")
for risk_level in RiskLevel:
    pending = workflow.get_pending_reviews(risk_level)
    if pending:
        print(f"{risk_level.name}: {len(pending)} 项")
```
