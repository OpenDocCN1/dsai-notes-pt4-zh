def initialize_aoai_client():
    """初始化统一的 AOAI OpenAI 客户端"""
    global aoai_client
    if AOAI_API_KEY and AOAI_BASE_URL:
        try:
            aoai_client = OpenAI(api_key=AOAI_API_KEY, base_url=AOAI_BASE_URL)
            print("AOAI 客户端初始化成功。")
        except Exception as e:
            print(f"错误: 初始化 AOAI 客户端失败: {e}")
            aoai_client = None
    else:
        print("警告：未配置 AOAI_API_KEY 或 AOAI_BASE_URL。AOAI 功能将不可用。")
        aoai_client = None

def initialize_chromadb():
    """初始化 ChromaDB 客户端并加载或创建 Collection"""
    global collection
    if not pathlib.Path(CHROMA_DB_PATH).exists():
        print(f"ChromaDB 路径 '{CHROMA_DB_PATH}' 不存在，将自动创建。")
    
    try:
        # 对于新版 chromadb (>=0.4.0)，PersistentClient 是推荐方式
        # 对于旧版 (如 0.3.x)，可能是 chromadb.Client(Settings(...))
        # 假设使用的是支持 PersistentClient 的版本
        chroma_client = chromadb.PersistentClient(path=CHROMA_DB_PATH)
        collection = chroma_client.get_or_create_collection(
            name=COLLECTION_NAME,
            metadata={"hnsw:space": DISTANCE_METRIC} # 指定距离度量
        )
        print(f"已加载或创建 ChromaDB Collection '{COLLECTION_NAME}'.")
    except Exception as e:
        print(f"致命错误：无法加载或创建 Collection '{COLLECTION_NAME}'. 错误: {e}")
        traceback.print_exc()
        collection = None

## 文件加载与分块函数

```python
def load_and_chunk_data(data_dir: pathlib.Path):
    """加载、解析、分块支持的文件"""
    all_chunks = []
    all_metadatas = []
    all_ids = []
    chunk_id_counter = 0

    if not data_dir.is_dir():
        print(f"错误: 数据目录 '{data_dir}' 不存在或不是一个目录。")
        return [], [], []

    print(f"开始扫描目录 '{data_dir}' 下的 {SUPPORTED_FILE_EXTENSIONS} 文件...")
    source_files_processed = set()

    for file_path in data_dir.rglob('*'):
        if file_path.is_file() and file_path.suffix.lower() in SUPPORTED_FILE_EXTENSIONS:
            print(f"  处理文件: {file_path.relative_to(data_dir)}")
            file_content = None
            try:
                if file_path.suffix.lower() in ['.txt', '.md']:
                    with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                        file_content = f.read()
                elif file_path.suffix.lower() == '.pdf':
                    doc_text = ""
                    with fitz.open(file_path) as pdf_doc:
                        for page_num, page in enumerate(pdf_doc):
                            # 添加页码信息，使上下文更明确
                            doc_text += f"[源: {file_path.name}, 页码 {page_num + 1}]\n" + page.get_text("text") + "\n\n"
                    file_content = doc_text
                elif file_path.suffix.lower() == '.docx':
                    doc = docx.Document(file_path)
                    paragraphs = [para.text for para in doc.paragraphs if para.text.strip()]
                    file_content = f"[源: {file_path.name}]\n" + "\n\n".join(paragraphs) # 添加文件名信息

                if file_content is None or not file_content.strip():
                    print(f"  警告: 未能从 {file_path.name} 提取有效文本内容。")
                    continue

                source_files_processed.add(str(file_path.relative_to(data_dir)))
                
                # 简单的基于换行符的切分策略
                potential_chunks = file_content.split('\n\n') # 主要按双换行分段
                current_chunk_text = ""

                for chunk_part in potential_chunks:
                    trimmed_part = chunk_part.strip()
                    if trimmed_part:
                        if not current_chunk_text:
                           current_chunk_text = trimmed_part
                        # 尝试合并短文本块，避免过于碎片化
                        elif len(current_chunk_text) + len(trimmed_part) < 800 and len(current_chunk_text) < 500 :
                           current_chunk_text += "\n" + trimmed_part # 用单换行连接同一逻辑块内的部分
                        else:
                            if len(current_chunk_text) >= CHUNK_MIN_LENGTH:
                                metadata = {"source": str(file_path.relative_to(data_dir)), "file_name": file_path.name}
                                chunk_unique_id = f"{file_path.stem}_{chunk_id_counter}"
                                all_chunks.append(current_chunk_text)
                                all_metadatas.append(metadata)
                                all_ids.append(chunk_unique_id)
                                chunk_id_counter += 1
                            current_chunk_text = trimmed_part
                
                # 处理最后一个累积的块
                if current_chunk_text and len(current_chunk_text) >= CHUNK_MIN_LENGTH:
                    metadata = {"source": str(file_path.relative_to(data_dir)), "file_name": file_path.name}
                    chunk_unique_id = f"{file_path.stem}_{chunk_id_counter}"
                    all_chunks.append(current_chunk_text)
                    all_metadatas.append(metadata)
                    all_ids.append(chunk_unique_id)
                    chunk_id_counter += 1

            except Exception as e:
                print(f"  处理文件 {file_path.name} 时出错: {e}")
                traceback.print_exc()

    if not all_chunks:
        print(f"警告: 未能从目录 '{data_dir}' 加载任何有效的文本块。")
    else:
        print(f"成功加载 {len(all_chunks)} 个文本块，来源 {len(source_files_processed)} 个文件。")

    return all_chunks, all_metadatas, all_ids
```

## 嵌入函数

```python
def get_embeddings_with_aoai(texts_to_embed):
    """使用 AOAI API 获取文本列表的嵌入向量"""
    global aoai_client
    if not aoai_client:
        print("错误: AOAI 客户端未初始化。")
        return [None] * len(texts_to_embed)

    if isinstance(texts_to_embed, str):
        texts_to_embed = [texts_to_embed]

    all_embeddings = []
    try:
        response = aoai_client.embeddings.create(model=EMBEDDING_MODEL, input=texts_to_embed)
        
        # 确保按输入顺序返回嵌入，并检查维度
        embedding_map = {item.index: item.embedding for item in response.data}
        for i in range(len(texts_to_embed)):
            emb = embedding_map.get(i)
            if emb and len(emb) == EMBEDDING_DIMENSION:
                all_embeddings.append(emb)
            else:
                all_embeddings.append(None) # 维度不匹配或获取失败
                if emb: # 如果获取到但是维度不对
                    print(f"警告: 文本索引 {i} 的嵌入维度 ({len(emb)}) 与期望维度 ({EMBEDDING_DIMENSION}) 不符。")
        
        print(f"  成功为 {len(texts_to_embed)} 个文本请求了嵌入向量。")

    except AuthenticationError as e:
         print(f"认证错误: AOAI API Key 无效或权限不足。 {e}")
         return [None] * len(texts_to_embed)
    except APIError as e:
        print(f"API错误: 调用 AOAI Embeddings API 失败: {e.status_code} - {e.message}")
        try:
            error_details = e.response.json() # 尝试解析 JSON 响应体
            print(f"  错误详情 (API): {error_details}")
        except:
            print(f"  原始错误响应 (API): {e.response.text}")
        return [None] * len(texts_to_embed)
    except Exception as e:
        print(f"未知错误: 获取嵌入向量时发生错误: {e}")
        traceback.print_exc()
        return [None] * len(texts_to_embed)

    if any(emb is None for emb in all_embeddings):
        print(f"警告: 未能成功获取所有 {len(texts_to_embed)} 个文本的有效嵌入向量。")

    return all_embeddings
```

## 数据索引函数

```python
def index_knowledge_base(chunks, metadatas, ids):
    """获取嵌入并索引数据到 ChromaDB"""
    global collection, aoai_client # aoai_client 用于 get_embeddings_with_aoai
    if not collection:
        print("错误: ChromaDB Collection 未初始化，无法索引。")
        return False
    if not chunks:
        print("错误: 没有可供索引的文本块。")
        return False
    if not aoai_client:
        print("错误: AOAI 客户端未初始化，无法生成嵌入向量进行索引。")
        return False

    print(f"开始为 {len(chunks)} 个文本块生成嵌入向量 (使用 {EMBEDDING_MODEL})...")
    embeddings = get_embeddings_with_aoai(chunks)

    valid_embeddings = []
    valid_chunks = []
    valid_metadatas = []
    valid_ids = []

    for i, emb in enumerate(embeddings):
        if emb is not None and len(emb) == EMBEDDING_DIMENSION:
            valid_embeddings.append(emb)
            valid_chunks.append(chunks[i])
            valid_metadatas.append(metadatas[i])
            valid_ids.append(ids[i])
        else:
            print(f"  跳过索引失败的块 ID: {ids[i]} (未能获取有效嵌入)。")
    
    if not valid_embeddings:
        print("错误: 未能为任何文本块生成有效的嵌入向量。索引操作中止。")
        print(f"请检查 AOAI API Key、网络连接、模型名称 ({EMBEDDING_MODEL}) 和期望维度 ({EMBEDDING_DIMENSION})。")
        return False
    
    print(f"获取到 {len(valid_embeddings)} 个有效嵌入向量。开始添加到 ChromaDB...")

    try:
        collection.add(
            embeddings=valid_embeddings,
            documents=valid_chunks,
            metadatas=valid_metadatas,
            ids=valid_ids
        )
        print(f"成功将 {len(valid_ids)} 个条目索引到 Collection '{COLLECTION_NAME}'。")
        return True
    except Exception as e:
        print(f"错误: 添加数据到 ChromaDB 时出错: {e}")
        traceback.print_exc()
        return False
```

## RAG 核心回答函数

```python
def get_rag_answer(query: str) -> str:
    """处理用户查询，执行 RAG 流程并返回答案"""
    global collection, aoai_client
    if not query or not query.strip():
        return "请输入您的问题。"
    if collection is None:
        return "错误：知识库向量数据库未初始化。"
    if aoai_client is None: # 统一检查 aoai_client
        return "错误：AOAI 客户端未初始化 (请检查 API Key 和 Base URL 设置)。"

    print(f"\n--- 收到查询: {query} ---")

    # 1. & 2. 查询嵌入
    print(f"  生成查询嵌入向量 (使用 {EMBEDDING_MODEL})...")
    query_embedding_list = get_embeddings_with_aoai(query) # 单个查询也用列表
    if not query_embedding_list or query_embedding_list[0] is None:
        print("  错误: 未能生成查询嵌入。")
        return "抱歉，处理您的问题时遇到错误（无法生成查询嵌入）。"
    query_embedding = query_embedding_list[0]
    print("  查询嵌入向量已生成。")

    # 3. 检索
    print(f"  在 ChromaDB 中检索 Top {N_RESULTS_RETRIEVAL} 相关文档...")
    try:
        results = collection.query(
            query_embeddings=[query_embedding],
            n_results=N_RESULTS_RETRIEVAL,
            include=['documents', 'metadatas', 'distances']
        )
    except Exception as e:
        print(f"  错误: 查询 ChromaDB 时出错: {e}")
        traceback.print_exc()
        return "抱歉，在知识库中检索信息时遇到错误。"

    retrieved_chunks = []
    retrieved_sources_info = [] # 存储 (距离, 来源, 内容片段)

    if results and results.get('documents') and results['documents'][0]:
        docs = results['documents'][0]
        metadatas = results['metadatas'][0] if results.get('metadatas') and results['metadatas'] else [{}] * len(docs)
        distances = results['distances'][0] if results.get('distances') and results['distances'] else [float('inf')] * len(docs)

        print(f"  初步检索到 {len(docs)} 个文档。应用距离阈值 (< {DISTANCE_THRESHOLD})...")

        for i in range(len(docs)):
            doc_content = docs[i]
            dist = distances[i]
            meta = metadatas[i] if i < len(metadatas) else {} # 安全访问元数据
            source_file = meta.get('file_name', meta.get('source', '未知来源')) # 优先用 file_name

            print(f"    - 文档 {i}: 距离={dist:.4f}, 来源='{source_file}'")
            if dist < DISTANCE_THRESHOLD:
                print(f"      -> 相关 (距离 < {DISTANCE_THRESHOLD})")
                retrieved_chunks.append(doc_content)
                retrieved_sources_info.append(f"[来源: {source_file}, 距离: {dist:.4f}]")
            else:
                print(f"      -> 不够相关 (距离 >= {DISTANCE_THRESHOLD})")
        
        if retrieved_chunks:
            context_str = "\n\n---\n\n".join(retrieved_chunks)
            sources_display = "\n".join(retrieved_sources_info)
            print(f"  最终筛选出 {len(retrieved_chunks)} 个相关文本块。相关来源信息:\n{sources_display}")
        else:
            print("  警告: 没有文档满足距离阈值。")
            context_str = "虽然检索到一些信息，但它们与问题的相关性不够高。"
            if docs: # 如果有检索到但不满足阈值的
                 context_str += f"\n(最近似的几条信息来源: {[m.get('file_name', m.get('source','未知')) for m in metadatas[:N_RESULTS_RETRIEVAL]]})"

    else:
        print("  警告: ChromaDB 未返回任何文档。")
        context_str = "根据现有知识库，未找到与您问题直接相关的信息。"

    # 4. 构建 Prompt
    prompt = f"""
你是一个专业的“易速鲜花”客服助手。请根据下面提供的“背景知识”来回答“用户问题”。
请主要依据背景知识作答。如果背景知识中没有足够的信息来回答问题，请明确说明你的回答并非完全基于提供的背景知识，或者说明无法从背景知识中找到答案。

背景知识:
---
{context_str}
---

用户问题: {query}

请给出清晰、准确、友好的回答:
"""
    # print(f"  Prompt 构建完成:\n{prompt[:1000]}...") # 调试时可以打印部分 prompt

    # 5. 调用 AOAI LLM 生成答案
    print(f"  调用 AOAI API ({AOAI_CHAT_MODEL}) 生成回答...")
    final_answer = "抱歉，我在思考时遇到了一些问题，暂时无法回答。"
    try:
        response = aoai_client.chat.completions.create(
            model=AOAI_CHAT_MODEL,
            messages=[{"role": "user", "content": prompt}], # 可以考虑加入 system prompt
            max_tokens=1000,
            temperature=0.3,
            stream=False
        )
        if response.choices and response.choices[0].message and response.choices[0].message.content:
            final_answer = response.choices[0].message.content.strip()
            print("  成功从 AOAI 获取回答。")
        else:
            print("  警告: AOAI API 返回的响应中没有有效的回答。")
            final_answer = "抱歉，模型响应异常，未能生成回答。"
    except AuthenticationError as e:
         print(f"认证错误: AOAI API Key 无效或权限不足。 {e}")
         final_answer = "错误：AOAI API Key 配置无效。"
    except APIError as e:
        print(f"API错误: 调用 AOAI Chat API 失败: {e.status_code} - {e.message}")
        if e.status_code == 429: # Too Many Requests
            final_answer = "抱歉，请求过于频繁，请稍后再试。"
        else:
            try:
                error_details = e.response.json()
                final_answer = f"抱歉，与语言模型服务通信时出错 (代码: {e.status_code} - {error_details.get('error', {}).get('message', '无详细信息')})。"
            except:
                 final_answer = f"抱歉，与语言模型服务通信时出错 (代码: {e.status_code})。"
    except Exception as e:
        print(f"未知错误: 调用 AOAI API 时发生错误: {e}")
        traceback.print_exc()
        final_answer = "抱歉，处理您的请求时发生内部错误。"

    print(f"--- 回答已生成 ---")
    return final_answer
```
