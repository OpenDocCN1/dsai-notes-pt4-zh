
# 文本掩码与多语言翻译处理

## 正则表达式模式与掩码工具

```python
CODE_BLOCK = re.compile(r"```.*?```", re.S)
INLINE_CODE = re.compile(r"`[^`]+`")
MATH_INLINE = re.compile(r"\$[^$\n]+\$")
MATH_BLOCK  = re.compile(r"\$\$.*?\$\$", re.S)
NUMBERS     = re.compile(r"\b\d+(?:\.\d+)?\b")

def mask_text(text: str):
    """将代码、公式、数字替换为占位符，返回(替换后文本, 占位列表)"""
    masks = []
    def repl(pat, tag, s):
        def _r(m):
            idx = len(masks)
            key = f"<{tag}_{idx}>"
            masks.append((key, m.group(0)))
            return key
        return pat.sub(_r, s)
    s = text
    for pat, tag in [(CODE_BLOCK,"CODE"), (MATH_BLOCK,"MATHB"), (MATH_INLINE,"MATHI"),
                     (INLINE_CODE,"INCODE"), (NUMBERS,"NUM")]:
        s = repl(pat, tag, s)
    return s, masks

def unmask_text(text: str, masks):
    """将占位符还原回原文"""
    for key, val in reversed(masks):
        text = text.replace(key, val)
    return text

def zh_trans_with_mask(s: Optional[str]) -> Optional[str]:
    if not isinstance(s, str) or not s.strip():
        return s
    masked, ms = mask_text(s)
    zh = translate_fn(masked)
    zh = unmask_text(zh, ms)
    return zh
```

## 单条样本处理

```python
# -------------------------------------------------------
# 2) 单条样本处理：遵循你的“English保留，非English翻译+messages合成”规则
# -------------------------------------------------------
def process_one(example: dict) -> dict:
    """
    输入：原始样本
    输出：处理后的样本（English原样；非English -> 简体中文 + 拼接messages）
    messages 的 system.content 规则：
      content = "reasoning language: 简体中文\n\n{developer_zh}"
    """

    # 取出语言标记
    lang = example.get("reasoning_language", "")

    # English 行：整条样本保持原样
    if isinstance(lang, str) and lang.strip().lower() == "english":
        # 保持不变（深拷贝防止原地改动）
        return copy.deepcopy(example)

    # 非 English：进行翻译
    dev_src   = example.get("developer", "")
    user_src  = example.get("user", "")
    ana_src   = example.get("analysis", "")
    final_src = example.get("final", "")

    dev_zh   = zh_trans_with_mask(dev_src)
    user_zh  = zh_trans_with_mask(user_src)
    ana_zh   = zh_trans_with_mask(ana_src)
    final_zh = zh_trans_with_mask(final_src)

    # 构造 system content（即使 developer 为空，也要放入“reasoning language: 简体中文”）
    sys_lines = ["reasoning language: 简体中文"]
    if isinstance(dev_zh, str) and dev_zh.strip():
        sys_lines += ["", dev_zh]
    system_content = "\n".join(sys_lines)

    # 重新拼 messages
    messages = []
    messages.append({"role": "system", "content": system_content, "thinking": None})
    if isinstance(user_zh, str) and user_zh.strip():
        messages.append({"role": "user", "content": user_zh, "thinking": None})
    if isinstance(ana_zh, str) and ana_zh.strip():
        messages.append({"role": "assistant", "content": ana_zh, "thinking": None})
    if isinstance(final_zh, str) and final_zh.strip():
        messages.append({"role": "assistant", "content": final_zh, "thinking": None})

    # 产出样本：顶层字段翻译 + 固定把 reasoning_language 改为“简体中文”
    out = copy.deepcopy(example)   # 如果你不想保留原 messages，可不拷贝，直接新建 dict
    out["reasoning_language"] = "简体中文"
    out["developer"] = dev_zh
    out["user"] = user_zh
    out["analysis"] = ana_zh
    out["final"] = final_zh
    out["messages"] = messages

    return out
```

## 数据集主流程与断点续传

```python
# -------------------------------------------------------
# 3) 主流程：断点续传 + tqdm 进度 + 抽样打印
# -------------------------------------------------------
def process_dataset_with_resume(
    split: str = "train",
    out_path: str = "Multilingual-Thinking-zh.jsonl",
    print_every: int = 20,
    reuse_cache_offline: bool = True
):
    """
    - 读取HuggingFaceH4/Multilingual-Thinking（走缓存）
    - English行原样保留；非English行按规则翻译并重建messages
    - 结果逐行追加写JSONL，支持断点续传
    """
    # 加载数据（复用缓存，离线可用）
    ds = load_dataset(
        "HuggingFaceH4/Multilingual-Thinking",
        split=split,
        download_mode="reuse_dataset_if_exists" if reuse_cache_offline else None
    )

    # 断点续传
    start_idx = 0
    if os.path.exists(out_path):
        with open(out_path, "r", encoding="utf-8") as f:
            start_idx = sum(1 for _ in f)
    print(f"从第 {start_idx} 条开始处理...（总计 {len(ds)} 条）")

    with open(out_path, "a", encoding="utf-8") as f:
        for i, rec in tqdm(enumerate(ds), total=len(ds), initial=start_idx, desc="翻译进度"):
            if i < start_idx:
                continue

            ex_out = process_one(rec)
            f.write(json.dumps(ex_out, ensure_ascii=False) + "\n")

            if print_every and (i % print_every == 0):
                # 抽样打印：展示关键字段 100 字
                lang_before = rec.get("reasoning_language", "")
                lang_after  = ex_out.get("reasoning_language", "")
                print(f"\n# 样本 {i}")
                print("language:", lang_before, "->", lang_after)
                print("developer:", str(ex_out.get("developer",""))[:100])
                print("user     :", str(ex_out.get("user",""))[:100])
                print("analysis :", str(ex_out.get("analysis",""))[:100])
                print("final    :", str(ex_out.get("final",""))[:100])

    print("✅ 处理完成 ->", out_path)
```

## 调用入口

```python
process_dataset_with_resume(
    split="train",
    out_path="Multilingual-Thinking-zh.jsonl",
    print_every=20
)
```
