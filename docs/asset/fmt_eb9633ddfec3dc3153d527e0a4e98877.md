```python
from transformers import pipeline
import torch

class LocalToxicityDetector:
    """本地毒性检测器"""
    
    def __init__(self, model_name: str = "unitary/toxic-bert"):
        """
        初始化本地毒性检测模型
        
        常用模型:
        - unitary/toxic-bert
        - martin-ha/toxic-comment-model
        - s-nlp/roberta_toxicity_classifier
        """
        self.device = 0 if torch.cuda.is_available() else -1
        self.classifier = pipeline(
            "text-classification",
            model=model_name,
            device=self.device,
            top_k=None  # 返回所有标签的分数
        )
    
    def detect(self, text: str, threshold: float = 0.7) -> Dict:
        """
        检测文本毒性
        
        Returns:
            {
                'is_toxic': bool,
                'scores': dict,
                'max_toxicity': float,
                'toxic_labels': list
            }
        """
        results = self.classifier(text)[0]
        
        # 转换为字典格式
        scores = {item['label']: item['score'] for item in results}
        
        # 找出超过阈值的标签
        toxic_labels = [
            label for label, score in scores.items()
            if score > threshold and label != 'non-toxic'
        ]
        
        max_toxicity = max(
            score for label, score in scores.items()
            if label != 'non-toxic'
        )
        
        return {
            'is_toxic': len(toxic_labels) > 0,
            'scores': scores,
            'max_toxicity': max_toxicity,
            'toxic_labels': toxic_labels
        }
    
    def batch_detect(self, texts: List[str], threshold: float = 0.7) -> List[Dict]:
        """批量检测"""
        return [self.detect(text, threshold) for text in texts]

# 使用示例
detector = LocalToxicityDetector()

test_texts = [
    "这是一段正常的对话内容。",
    "你这个白痴,去死吧!",  # 有毒内容示例
]

for text in test_texts:
    result = detector.detect(text, threshold=0.7)
    print(f"
文本: {text}")
    print(f"是否有毒: {result['is_toxic']}")
    print(f"最高毒性分数: {result['max_toxicity']:.2f}")
    if result['toxic_labels']:
        print(f"有毒标签: {result['toxic_labels']}")
```

```python
import requests
from typing import List, Dict, Optional
from dataclasses import dataclass

@dataclass
class FactCheckResult:
    """事实核查结果"""
    claim: str
    is_verified: bool
    confidence: float
    sources: List[str]
    contradictions: List[str]

class FactChecker:
    """事实核查系统"""
    
    def __init__(self, search_api_key: str):
        self.search_api_key = search_api_key
    
    def check_claim(self, claim: str, min_sources: int = 3) -> FactCheckResult:
        """
        核查单个声明
        
        Args:
            claim: 要核查的声明
            min_sources: 最少验证源数量
        
        Returns:
            FactCheckResult对象
        """
        # 1. 搜索相关信息
        search_results = self._search_claim(claim)
        
        # 2. 提取支持和反对的证据
        supporting = []
        contradicting = []
        
        for result in search_results[:10]:
            evidence_type = self._classify_evidence(claim, result['snippet'])
            if evidence_type == 'support':
                supporting.append(result['url'])
            elif evidence_type == 'contradict':
                contradicting.append(result['url'])
        
        # 3. 计算置信度
        total_evidence = len(supporting) + len(contradicting)
        if total_evidence == 0:
            confidence = 0.0
            is_verified = False
        else:
            confidence = len(supporting) / total_evidence
            is_verified = len(supporting) >= min_sources and confidence > 0.7
        
        return FactCheckResult(
            claim=claim,
            is_verified=is_verified,
            confidence=confidence,
            sources=supporting,
            contradictions=contradicting
        )
    
    def _search_claim(self, claim: str) -> List[Dict]:
        """搜索声明相关信息"""
        # 实际实现中调用搜索 API(Google, Bing, etc.)
        # 这里返回模拟数据
        return [
            {'url': 'https://example.com/article1', 'snippet': '...'},
            {'url': 'https://example.com/article2', 'snippet': '...'},
        ]
    
    def _classify_evidence(self, claim: str, evidence: str) -> str:
        """
        分类证据是支持还是反对声明
        
        实际实现中使用 NLI (Natural Language Inference) 模型
        如 roberta-large-mnli 或 deberta-v3-large-mnli
        """
        # 简化实现
        return 'support'  # 或 'contradict' 或 'neutral'
    
    def detect_hallucination(self, generated_text: str) -> Dict:
        """
        检测 AI 生成文本中的幻觉
        
        策略:
        1. 提取可验证的声明
        2. 对每个声明进行事实核查
        3. 标记未验证的声明
        """
        # 1. 提取声明
        claims = self._extract_claims(generated_text)
        
        # 2. 核查每个声明
        results = []
        for claim in claims:
            result = self.check_claim(claim)
            results.append(result)
        
        # 3. 汇总
        unverified_claims = [r for r in results if not r.is_verified]
        hallucination_score = len(unverified_claims) / len(results) if results else 0
        
        return {
            'hallucination_score': hallucination_score,
            'total_claims': len(results),
            'verified_claims': len(results) - len(unverified_claims),
            'unverified_claims': [r.claim for r in unverified_claims],
            'details': results
        }
    
    def _extract_claims(self, text: str) -> List[str]:
        """
        从文本中提取可验证的声明
        
        实际实现可使用:
        - 依存句法分析
        - ClaimBuster 等专门工具
        - LLM 辅助提取
        """
        # 简化实现:按句子分割
        import re
        sentences = re.split(r'[。!?]', text)
        return [s.strip() for s in sentences if len(s.strip()) > 10]

# 使用示例
fact_checker = FactChecker(search_api_key="your_api_key")

generated_text = """
OpenClaw 是由 Peter Steinberger 创建的开源 AI Agent 框架。
该项目在 2026 年 1 月获得了超过 179,000 个 GitHub stars。
OpenClaw 可以自动驾驶汽车并控制核电站。
"""

hallucination_report = fact_checker.detect_hallucination(generated_text)

print(f"幻觉分数: {hallucination_report['hallucination_score']:.2%}")
print(f"总声明数: {hallucination_report['total_claims']}")
print(f"已验证: {hallucination_report['verified_claims']}")
print(f"未验证的声明: {hallucination_report['unverified_claims']}")
```
