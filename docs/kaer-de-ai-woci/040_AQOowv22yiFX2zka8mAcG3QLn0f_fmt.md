A third-person camera that follows behind and slightly above a basic airplane model （the model can be created using basic geometry, but

class Snake:
    def __init__(self):
        self.body = [(GRID_SIZE // 2, GRID_SIZE // 2)]
        self.direction = (1, 0)
        self.grow_count = 0
        self.effects = {}
        self.decoy_positions = []
        self.segment_colors = []
        
    def move(self, reverse=False):
        head = self.body[0]
        direction = self.direction
        
        if reverse:
            direction = (-direction[0], -direction[1])
            
        new_head = (head[0] + direction[0], head[1] + direction[1])
        
        # Wrap around screen
        new_head = (new_head[0] % GRID_SIZE, new_head[1] % (WINDOW_HEIGHT // CELL_SIZE))
        
        self.body.insert(0, new_head)
        
        if self.grow_count > 0:
            self.grow_count -= 1
            self.segment_colors.insert(0, self.get_segment_color(len(self.body)))
        else:
            self.body.pop()
            if self.segment_colors:
                self.segment_colors.pop()
                
    def change_direction(self, new_direction):
        if (new_direction[0] * -1, new_direction[1] * -1) != self.direction:
            self.direction = new_direction
            
    def grow(self, amount=1):
        self.grow_count += amount
        
    def get_segment_color(self, index):
        # Biomechanical evolution based on length
        base_color = GREEN
        evolution_factor = min(index / 20, 1.0)
        
        # Shift from green to cyan as snake grows
        r = int(base_color[0] * (1 - evolution_factor))
        g = int(base_color[1])
        b = int(base_color[2] * evolution_factor + 255 * evolution_factor)
        
        return (r, g, min(b, 255))
        
    def draw(self, surface, time_factor=1.0):
        # Draw decoy if double vision is active
        if 'double_vision' in self.effects:
            for i, pos in enumerate(self.decoy_positions):
                alpha = 100 - (i * 2)
                segment_surf = pygame.Surface((CELL_SIZE, CELL_SIZE), pygame.SRCALPHA)
                pygame.draw.rect(segment_surf, (*CYAN[:3], alpha), (0, 0, CELL_SIZE, CELL_SIZE))
                surface.blit(segment_surf, (pos[0] * CELL_SIZE, pos[1] * CELL_SIZE))
        
        # Draw main snake
        for i, segment in enumerate(self.body):
            # Get color based on evolution
            if i < len(self.segment_colors):
                color = self.segment_colors[i]
            else:
                color = self.get_segment_color(i)
                self.segment_colors.append(color)
            
            # Animated texture effect for evolved segments
            if i > 10:
                pulse = math.sin(time.time() * 5 + i * 0.5) * 20
                color = (min(color[0] + int(pulse), 255), 
                        min(color[1] + int(pulse), 255), 
                        min(color[2] + int(pulse), 255))
            
            # Draw segment with gradient effect
            segment_rect = pygame.Rect(segment[0] * CELL_SIZE, segment[1] * CELL_SIZE, 
                                     CELL_SIZE, CELL_SIZE)
            pygame.draw.rect(surface, color, segment_rect)
            
            # Inner highlight for 3D effect

            else:
                inner_color = tuple(min(c + 50, 255) for c in color)
                pygame.draw.rect(surface, inner_color, segment_rect.inflate(-4, -4))

            size = random.randint(2, 6)
            self.particles.append(Particle(x * CELL_SIZE + CELL_SIZE // 2,

            self.game_over = True

            if self.snake.effects[effect] <= 0:
                del self.snake.effects[effect]
                if effect == 'double_vision':
                    self.snake.decoy_positions = []
                    
        # Update particles
        for particle in self.particles[:]:
            particle.update()
            if particle.lifetime <= 0:
                self.particles.remove(particle)
                
        # Update trails
        for trail in list(self.trails):
            trail.update()
            if trail.lifetime <= 0:
                self.trails.remove(trail)
                
        # Update obstacles
        for obstacle in self.obstacles:
            obstacle.update(self.time_factor)
            
        # Update terrain zones
        for zone in self.terrain_zones:

        for effect, duration in self.snake.effects.items():
            effect_text = font.render(f"{effect.replace('_', ' ').title()}: {int(duration/30)}s", 
                                    True, YELLOW)
            surface.blit(effect_text, (10, y_offset))
            y_offset += 30
            
        # Game over
        if self.game_over:
            game_over_font = pygame.font.Font(None, 72)
            game_over_text = game_over_font.render("GAME OVER", True, RED)
            text_rect = game_over_text.get_rect(center=(WINDOW_WIDTH // 2, WINDOW_HEIGHT // 2))
            
            # Glow effect
            for i in range(3):
                glow_surf = pygame.Surface((text_rect.width + i*20, text_rect.height + i*10), pygame.SRCALPHA)
                glow_surf.fill((255, 0, 0, 50-i*15))
                surface.blit(glow_surf, (text_rect.x - i*10, text_rect.y - i*5))
                
            surface.blit(game_over_text, text_rect)
            
            restart_font = pygame.font.Font(None, 36)
            restart_text = restart_font.render("Press SPACE to restart", True, WHITE)
            restart_rect = restart_text.get_rect(center=(WINDOW_WIDTH // 2, WINDOW_HEIGHT // 2 + 50))
            surface.blit(restart_text, restart_rect)

def main():
    game = Game()
    running = True
    move_timer = 0
    move_delay = 5
    
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_SPACE and game.game_over:
                    game = Game()
                elif not game.game_over:
                    if event.key == pygame.K_UP:
                        game.snake.change_direction((0, -1))
                    elif event.key == pygame.K_DOWN:
                        game.snake.change_direction((0, 1))
                    elif event.key == pygame.K_LEFT:
                        game.snake.change_direction((-1, 0))
                    elif event.key == pygame.K_RIGHT:
                        game.snake.change_direction((1, 0))
        
        # Update game at controlled intervals
        move_timer += clock.get_time()
        if move_timer >= move_delay * (10 / game.time_factor):
            move_timer = 0
            game.update()
        
        # Draw everything
        game.draw(screen)
        pygame.display.flip()
        clock.tick(60)
    
    pygame.quit()

if __name__ == "__main__":
    main()
