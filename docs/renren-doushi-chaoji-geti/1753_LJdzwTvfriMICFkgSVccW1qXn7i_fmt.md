### 分词

`分词` 执行中文分词处理。

```elisp
(defun 分词 (text)
  "中文分词处理"
  (let ((词表 '()))
    (with-temp-buffer
      (insert text)
      (goto-char (point-min))
      (while (re-search-forward "\\cc+" nil t)
        (push (match-string 0) 词表)))
    (nreverse 词表)))
```

### 计算词语权重

(defun 计算词语权重 (词语 文档集)
  "计算词语的TF-IDF权重"
  (let* ((总文档数 (length 文档集))
         (包含词语文档数 (length (删除重复元素
                                (mapcar (lambda (doc)
                                        (when (string-match-p 词语 doc) doc))
                                      文档集))))
         (词频 (count 词语 (分词 (car 文档集)) :test 'string=)))
    (* 词频 (log (/ 总文档数 (1+ 包含词语文档数))))))

### 提取关键概念

`提取关键概念` 从段落中提取关键概念。

```elisp
(defun 提取关键概念 (段落列表)
  "从段落中提取关键概念"
  (let ((词频表 (make-hash-table :test 'equal))
        (关键词表 '()))
    (dolist (段落 段落列表)
      (let ((词语列表 (分词 段落)))
        (dolist (词语 词语列表)
          (puthash 词语
                   (1+ (or (gethash 词语 词频表) 0))
                   词频表))))
    (maphash (lambda (词语 频率)
               (when (and (> (length 词语) 1)
                         (> 频率 1))
                 (push (cons 词语 (计算词语权重 词语 段落列表))
                       关键词表)))
             词频表)
    (mapcar #'car
            (sort 关键词表
                  (lambda (a b) (> (cdr a) (cdr b)))))))
```

### 高效去重

`高效去重` 使用哈希表实现 O(n) 去重。

```elisp
(defun 高效去重 (列表)
  "使用哈希表实现O(n)去重"
  (let ((哈希表 (make-hash-table :test 'equal))
        (结果 '()))
    (dolist (元素 列表)
      (unless (gethash 元素 哈希表)
        (push 元素 结果)
        (puthash 元素 t 哈希表)))
    (nreverse 结果)))
```
